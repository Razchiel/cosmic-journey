// ============================================================
// GET /api/get-wishes?sort=newest|rating&category=all|<id>&cursor=<opaque>&limit=12
// Hanya status 'approved' dan is_private = false. Field status/is_private/ip_hash tidak pernah keluar.
// Kontrak respons sama dengan getWishes() di src/services/wishes.js:
//   { items, nextCursor, total, aggregate: { average, count } }
// ============================================================
import { CATEGORY_IDS, db, fail, json, totalFromRange } from '../lib/shared.js'

const PAGE_DEFAULT = 12
const PAGE_MAX = 50
const OFFSET_MAX = 2000
const COLUMNS = 'id,display_name,is_anonymous,category,message,rating,created_at'

const ORDER = {
    newest: 'created_at.desc,id.desc',
    rating: 'rating.desc,created_at.desc,id.desc',
}

// Cursor "opaque" = offset dalam base64url. Klien mengirim balik apa adanya.
const encodeCursor = (n) => Buffer.from(String(n)).toString('base64url')
function decodeCursor(c) {
    if (!c) return 0
    if (c.length > 24) return null
    const n = Number(Buffer.from(c, 'base64url').toString())
    return Number.isInteger(n) && n >= 0 && n <= OFFSET_MAX ? n : null
}

const CACHE_HEADERS = {
    // Browser: 30 dtk. CDN Netlify: 60 dtk + stale-while-revalidate (hemat kuota Supabase & function).
    'Cache-Control': 'public, max-age=30',
    'Netlify-CDN-Cache-Control': 'public, s-maxage=60, stale-while-revalidate=120',
}

export default async (req) => {
    if (req.method !== 'GET') return fail('validation', 'Method not allowed.', 405)

    const params = new URL(req.url).searchParams
    const sort = params.get('sort') || 'newest'
    const category = params.get('category') || 'all'
    const offset = decodeCursor(params.get('cursor'))
    const limit = Math.min(PAGE_MAX, Math.max(1, parseInt(params.get('limit'), 10) || PAGE_DEFAULT))

    if (!ORDER[sort]) return fail('validation', 'Unknown sort.', 400)
    if (category !== 'all' && !CATEGORY_IDS.includes(category)) return fail('validation', 'Unknown category.', 400)
    if (offset === null) return fail('validation', 'Invalid cursor.', 400)

    try {
        const q = new URLSearchParams({
            select: COLUMNS,
            status: 'eq.approved',
            is_private: 'eq.false',
            order: ORDER[sort],
            limit: String(limit),
            offset: String(offset),
        })
        if (category !== 'all') q.set('category', `eq.${category}`)

        const [listRes, statsRes] = await Promise.all([
            db(`wishes?${q}`, { headers: { Prefer: 'count=exact' } }),
            db('rpc/wish_stats', { method: 'POST', body: {} }),
        ])

        // 416 = offset melewati jumlah baris → anggap halaman kosong
        let rows = []
        let total = 0
        if (listRes.status === 416) {
            total = totalFromRange(listRes)
        } else if (listRes.ok) {
            rows = await listRes.json()
            total = totalFromRange(listRes)
        } else {
            throw new Error(`list failed (${listRes.status})`)
        }

        if (!statsRes.ok) throw new Error(`stats failed (${statsRes.status})`)
        const stats = (await statsRes.json())?.[0] || {}

        const items = rows.map((r) => ({
            id: r.id,
            displayName: r.display_name,
            isAnonymous: r.is_anonymous,
            category: r.category,
            message: r.message,
            rating: r.rating,
            createdAt: r.created_at,
        }))

        const nextCursor = offset + items.length < total && items.length > 0
            ? encodeCursor(offset + items.length)
            : null

        return json(
            {
                items,
                nextCursor,
                total,
                aggregate: { count: Number(stats.count) || 0, average: Number(stats.average) || 0 },
            },
            200,
            CACHE_HEADERS
        )
    } catch (err) {
        console.error('[get-wishes]', err?.message || err)
        return fail('server', 'The observatory is quiet right now. Please try again soon.', 500)
    }
}

// ============================================================
// POST /api/submit-wish
// Validasi ketat server-side → honeypot → rate limit + cek duplikat → simpan.
// Default status 'pending'. AUTO_APPROVE=true → 'approved' (hanya pesan publik tanpa email/link/telepon).
// Pesan private tersimpan tetapi TIDAK PERNAH dikembalikan oleh get-wishes.
// ============================================================
import {
    CATEGORY_IDS,
    MESSAGE_MIN,
    MESSAGE_MAX,
    NAME_MAX,
    cleanMessage,
    cleanName,
    db,
    fail,
    hasBadWords,
    hashIp,
    json,
    makeCallsign,
    needsReview,
} from '../lib/shared.js'

const LIMIT_PER_HOUR = 10
const LIMIT_PER_DAY = 30
const MAX_BODY = 4000
const HOUR_MS = 3600 * 1000
const DAY_MS = 24 * HOUR_MS

// Satu query untuk dua keperluan: rate limit (jumlah per jam / per hari) dan cek duplikat (24 jam terakhir)
async function recentFromIp(ipHash) {
    const q = new URLSearchParams({
        select: 'message,created_at',
        ip_hash: `eq.${ipHash}`,
        created_at: `gte.${new Date(Date.now() - DAY_MS).toISOString()}`,
        order: 'created_at.desc',
        limit: '20',
    })
    const res = await db(`wishes?${q}`)
    if (!res.ok) throw new Error(`recent check failed (${res.status})`)
    const rows = await res.json()
    return Array.isArray(rows) ? rows : []
}

// Normalisasi untuk membandingkan pesan: huruf kecil, tanpa aksen, tanpa tanda baca, spasi dirapikan
const normalize = (s) =>
    String(s)
        .toLowerCase()
        .normalize('NFKD')
        .replace(/[\u0300-\u036f]/g, '')
        .replace(/[^\p{L}\p{N}]+/gu, ' ')
        .trim()

const toRecord = (r) => ({
    id: r.id,
    displayName: r.display_name,
    isAnonymous: r.is_anonymous,
    category: r.category,
    message: r.message,
    rating: r.rating,
    isPrivate: r.is_private,
    createdAt: r.created_at,
    status: r.status,
})

export default async (req, context) => {
    if (req.method !== 'POST') return fail('validation', 'Method not allowed.', 405)

    // ── Baca body ──
    let data
    try {
        const raw = await req.text()
        if (raw.length > MAX_BODY) return fail('validation', 'That message is too large.', 413)
        data = JSON.parse(raw)
    } catch (_) {
        return fail('validation', 'We could not read that request.', 400)
    }
    if (!data || typeof data !== 'object' || Array.isArray(data)) {
        return fail('validation', 'We could not read that request.', 400)
    }

    // ── Validasi & pembersihan ──
    const message = cleanMessage(data.message)
    if (message.length < MESSAGE_MIN || message.length > MESSAGE_MAX) {
        return fail('validation', `Your message needs ${MESSAGE_MIN}–${MESSAGE_MAX} characters.`, 400)
    }
    const rating = data.rating
    if (typeof rating !== 'number' || !Number.isInteger(rating) || rating < 1 || rating > 5) {
        return fail('validation', 'Please choose a rating from 1 to 5 stars.', 400)
    }
    const category = data.category
    if (typeof category !== 'string' || !CATEGORY_IDS.includes(category)) {
        return fail('validation', 'Please choose a category.', 400)
    }
    const isPrivate = data.isPrivate === true
    if (!isPrivate && data.consent !== true) {
        return fail('validation', 'Please agree to show your message in the Gallery, or send it privately.', 400)
    }

    let displayName = cleanName(data.displayName).slice(0, NAME_MAX)
    const isAnonymous = data.isAnonymous === true || !displayName
    if (!displayName) displayName = makeCallsign()

    if (hasBadWords(message) || hasBadWords(displayName)) {
        return fail('validation', 'Please keep your words kind, the stars are listening. Try rephrasing your message.', 400)
    }

    // ── Honeypot: bot mengisi field tersembunyi → pura-pura sukses, tidak disimpan ──
    if (typeof data.website === 'string' && data.website.trim() !== '') {
        return json(
            {
                id: `ghost-${Date.now().toString(36)}`,
                displayName, isAnonymous, category, message, rating, isPrivate,
                createdAt: new Date().toISOString(),
                status: 'pending',
            },
            201,
            { 'Cache-Control': 'no-store' }
        )
    }

    try {
        // ── Rate limit + cek duplikat per IP-hash ──
        const ipHash = hashIp(context?.ip || req.headers.get('x-nf-client-connection-ip'))
        if (ipHash) {
            const recent = await recentFromIp(ipHash)
            const hourAgo = Date.now() - HOUR_MS
            const lastHour = recent.filter((r) => Date.parse(r.created_at) >= hourAgo).length

            if (lastHour >= LIMIT_PER_HOUR || recent.length >= LIMIT_PER_DAY) {
                return fail('rate_limit', 'You have sent a few wishes already. Please come back a little later.', 429)
            }

            // Pesan yang sama persis dari IP yang sama dalam 24 jam → tolak ramah.
            // Pesan sama dari orang lain (IP berbeda) tetap boleh.
            const mine = normalize(message)
            if (recent.some((r) => normalize(r.message) === mine)) {
                return fail('validation', 'You already sent this wish. Try a new one.', 409)
            }
        }

        // ── Status awal ──
        const autoApprove =
            process.env.AUTO_APPROVE === 'true' &&
            !isPrivate &&
            !needsReview(message) &&
            !needsReview(displayName)

        // ── Simpan ──
        const res = await db('wishes', {
            method: 'POST',
            headers: { Prefer: 'return=representation' },
            body: {
                display_name: displayName,
                is_anonymous: isAnonymous,
                category,
                message,
                rating,
                is_private: isPrivate,
                status: autoApprove ? 'approved' : 'pending',
                ip_hash: ipHash,
            },
        })
        if (!res.ok) {
            let code = ''
            try { code = (await res.json())?.code || '' } catch (_) { /* abaikan */ }
            throw new Error(`insert failed (${res.status} ${code})`)
        }
        const rows = await res.json()
        return json(toRecord(rows[0]), 201, { 'Cache-Control': 'no-store' })
    } catch (err) {
        console.error('[submit-wish]', err?.message || err) // tanpa isi pesan / IP
        return fail('server', 'The observatory is quiet right now. Please try again soon.', 500)
    }
}

import { useEffect } from 'react'
import styles from './FactPopup.module.css'

export default function FactPopup({ fact, onClose }) {
  useEffect(() => {
    if (!fact) return
    const timer = setTimeout(onClose, 6000)
    return () => clearTimeout(timer)
  }, [fact, onClose])

  return (
    <div className={`${styles.popup} ${fact ? styles.show : ''}`} role="status">
      {fact && (
        <>
          <div className={styles.header}>
            <span className={styles.title}>{fact.title}</span>
            {fact.subtitle && <span className={styles.subtitle}>{fact.subtitle}</span>}
            <button className={styles.close} onClick={onClose} aria-label="Close">×</button>
          </div>
          <p className={styles.text}>{fact.text}</p>
          {fact.extra && <p className={styles.extra}>{fact.extra}</p>}
        </>
      )}
    </div>
  )
}

import { useRef, useState, useEffect, useMemo, memo } from 'react'
import { Canvas, useFrame } from '@react-three/fiber'
import * as THREE from 'three'
import { sfx } from '../hooks/useSfx.js'
import styles from './HomeWarp.module.css'

// Timeline (ms)
const T_LAUNCH = 1800    // roket mulai naik, warp mulai
const T_COUNT = 10000    // hitung mundur "3" (lalu 2, 1 tiap 900 ms)
const T_FLASH = 12700    // kilatan menuju Bumi
export const HOME_MS = 13200 // dipakai App sebelum pindah ke Intro
// Overlay mulai memudar saat Intro tampil di belakangnya.
// App melepas overlay di HOME_MS + HOME_END_PAD (800 ms): fade 400 ms ini harus selesai sebelum itu.
const T_LEAVE = HOME_MS + 400

const COUNT = 360
const COUNT_REDUCED = 160
const clamp01 = (v) => Math.min(1, Math.max(0, v))

// Hanya starfield (garis bintang saat warp)
function Scene({ t0, reduce }) {
    const count = reduce ? COUNT_REDUCED : COUNT
    const stars = useRef()
    const mat = useRef()
    const dummy = useMemo(() => new THREE.Object3D(), [])
    const ps = useMemo(() => Array.from({ length: count }, () => ({
        x: (Math.random() - 0.5) * 40,
        y: (Math.random() - 0.5) * 60,
        z: Math.random() * 13 - 14,
    })), [count])

    useFrame((state, delta) => {
        if (!t0.current || !stars.current) return
        const t = performance.now() - t0.current
        const k = clamp01((t - T_LAUNCH) / 6000)
        const s = k * k * (3 - 2 * k)

        // Bintang makin terang mengikuti hitung mundur, puncaknya saat kilatan (0,8 → 1)
        if (mat.current) {
            const b = clamp01((t - T_COUNT) / (T_FLASH - T_COUNT))
            mat.current.opacity = 0.8 + 0.2 * b * b
        }

        // Reduced motion: tanpa getaran kamera, garis bintang pendek dan lambat
        if (!reduce) {
            const amp = s * s * 0.14
            state.camera.position.x = (Math.random() - 0.5) * amp
            state.camera.position.y = (Math.random() - 0.5) * amp
        }

        const speed = 0.2 + s * (reduce ? 12 : 60)
        const stretch = 1 + s * (reduce ? 8 : 110)
        ps.forEach((p, i) => {
            p.y -= speed * delta
            if (p.y < -30) { p.y = 30; p.x = (Math.random() - 0.5) * 40 }
            dummy.position.set(p.x, p.y, p.z)
            dummy.scale.set(1, stretch, 1)
            dummy.updateMatrix()
            stars.current.setMatrixAt(i, dummy.matrix)
        })
        stars.current.instanceMatrix.needsUpdate = true
    })

    return (
        <instancedMesh ref={stars} args={[null, null, count]} frustumCulled={false}>
            <boxGeometry args={[0.025, 0.025, 0.025]} />
            <meshBasicMaterial ref={mat} color="#e0f0ff" transparent opacity={0.8} />
        </instancedMesh>
    )
}

// Roket 2D: mengapung & bergoyang → api menyala → berdiri tegak & naik perlahan
function Rocket2D({ t0, reduce }) {
    const body = useRef()
    const flame = useRef()

    useEffect(() => {
        let raf = 0
        const j = reduce ? 0 : 1   // 0 = tanpa goyang, getar, dan kedip api
        const loop = () => {
            const t = performance.now() - t0.current
            const lk = clamp01((t - T_LAUNCH) / (T_FLASH - T_LAUNCH - 800))
            const rise = lk * lk * (3 - 2 * lk)                       // naik 0..1
            const up = clamp01((t - T_LAUNCH) / 1400)
            const upright = up * up * (3 - 2 * up)                    // tegak 0..1
            const ig = clamp01((t - T_LAUNCH) / 900)                  // api menyala 0..1
            const sway = Math.sin(t * 0.0016) * 10 * (1 - upright) * j
            const bob = Math.sin(t * 0.0021) * 10 * (1 - upright) * j
            const shake = ig * (Math.random() - 0.5) * 1.6 * j
            const y = bob - rise * window.innerHeight * 0.3
            if (body.current) {
                body.current.style.transform =
                    `translate(${shake}px, ${y}px) rotate(${sway + (1 - upright) * 6 * j}deg)`
            }
            if (flame.current) {
                const len = ig * (0.75 + rise * 0.6) + (Math.random() - 0.5) * 0.16 * ig * j
                flame.current.style.transform = `scale(${1 + (Math.random() - 0.5) * 0.1 * ig * j}, ${Math.max(len, 0.001)})`
            }
            raf = requestAnimationFrame(loop)
        }
        raf = requestAnimationFrame(loop)
        return () => cancelAnimationFrame(raf)
    }, [t0, reduce])

    return (
        <div className={styles.rocketWrap} aria-hidden="true">
            <div ref={body} className={styles.rocket}>
                <svg viewBox="0 0 80 150" className={styles.rocketSvg}>
                    <g ref={flame} className={styles.flame} style={{ transformOrigin: '40px 108px' }}>
                        <path d="M32 108 C32 128 38 142 40 152 C42 142 48 128 48 108Z" fill="#ff8a30" opacity="0.9" />
                        <path d="M36 108 C36 120 39 130 40 138 C41 130 44 120 44 108Z" fill="#fff2b0" />
                    </g>
                    <path d="M20 74 L4 104 L20 96Z" fill="#ff5a5a" />
                    <path d="M60 74 L76 104 L60 96Z" fill="#ff5a5a" />
                    <path d="M30 98 L50 98 L47 109 L33 109Z" fill="#444b5e" />
                    <path d="M40 4 C58 22 62 50 60 98 L20 98 C18 50 22 22 40 4Z" fill="#e8ecf8" />
                    <path d="M40 4 C50 13 56 25 58 38 L22 38 C24 25 30 13 40 4Z" fill="#ff5a5a" />
                    <circle cx="40" cy="58" r="9" fill="#6ec5ff" stroke="#aab4cc" strokeWidth="3" />
                    <circle cx="37" cy="55" r="2.5" fill="#fff" opacity="0.7" />
                </svg>
            </div>
        </div>
    )
}

// Konstelasi nama pengguna terbentuk di langit saat roket naik
function NameConstellation({ name, points }) {
    if (!points?.length) return null
    const xs = points.map((p) => p.x)
    const ys = points.map((p) => p.y)
    const pad = 14
    const minX = Math.min(...xs) - pad
    const minY = Math.min(...ys) - pad
    const w = Math.max(...xs) - Math.min(...xs) + pad * 2
    const h = Math.max(...ys) - Math.min(...ys) + pad * 2
    const d = points.map((p, i) => `${i ? 'L' : 'M'}${p.x} ${p.y}`).join(' ')
    return (
        <div className={styles.sky}>
            <svg viewBox={`${minX} ${minY} ${w} ${h}`} className={styles.skySvg} aria-hidden="true">
                <path d={d} pathLength="1" className={styles.skyLine} />
                {points.map((p, i) => (
                    <g key={i} className={styles.skyStar} style={{ animationDelay: `${0.6 + i * 0.5}s` }}>
                        <circle cx={p.x} cy={p.y} r="6" className={styles.skyHalo} />
                        <circle cx={p.x} cy={p.y} r="2" fill="#fff" />
                    </g>
                ))}
            </svg>
            <p className={styles.skyLabel}>✦ {name}’s star</p>
            <p className={styles.skyNote}>It will keep watch over your home, always.</p>
        </div>
    )
}

// onBrighten: dipanggil sekali saat hitung mundur dimulai (Bumi mendekat).
// App menaikkan StarField ke depth 0; easing ~2 dtk membuatnya terang penuh saat kilatan.
function HomeWarp({ active, name = '', points = [], onBrighten }) {
    const [phase, setPhase] = useState('float') // float | launch | count | flash
    const [count, setCount] = useState(0)
    const [leaving, setLeaving] = useState(false)
    const t0 = useRef(performance.now())
    const overlayRef = useRef(null)
    const brightenRef = useRef(onBrighten)
    brightenRef.current = onBrighten

    const reduce = useMemo(
        () => typeof window !== 'undefined' && window.matchMedia('(prefers-reduced-motion: reduce)').matches,
        []
    )

    useEffect(() => {
        if (!active) return
        t0.current = performance.now()
        setPhase('float')
        setCount(0)
        setLeaving(false)
        const ids = [
            setTimeout(() => { setPhase('launch'); sfx.warp((T_FLASH - T_LAUNCH) / 1000 + 0.4) }, T_LAUNCH),
            ...[3, 2, 1].map((n, i) =>
                setTimeout(() => {
                    setPhase('count'); setCount(n); sfx.click()
                    if (i === 0) brightenRef.current?.()
                }, T_COUNT + i * 900)
            ),
            setTimeout(() => { setPhase('flash'); sfx.chime(4) }, T_FLASH),
            setTimeout(() => setLeaving(true), T_LEAVE),
        ]
        return () => ids.forEach(clearTimeout)
    }, [active])

    // Fokus pindah ke overlay: tombol "Return home" di belakangnya tidak lagi menerima Enter/Spasi
    useEffect(() => {
        if (active) overlayRef.current?.focus({ preventScroll: true })
    }, [active])

    if (!active) return null

    const nearEarth = phase === 'count' || phase === 'flash'

    return (
        <div
            ref={overlayRef}
            className={`${styles.overlay} ${leaving ? styles.overlayOut : ''}`}
            role="dialog"
            aria-modal="true"
            aria-label="Returning home"
            tabIndex={-1}
            onKeyDown={(e) => { if (e.key === 'Tab') e.preventDefault() }}
        >
            <div className={styles.canvasWrap} aria-hidden="true">
                <Canvas dpr={[1, 1.5]} camera={{ position: [0, 0, 10], fov: 60 }}>
                    <Scene t0={t0} reduce={reduce} />
                </Canvas>
            </div>

            {phase !== 'float' && <NameConstellation name={name} points={points} />}
            <Rocket2D t0={t0} reduce={reduce} />

            <div className={`${styles.earthHorizon} ${nearEarth ? styles.earthHorizonOn : ''}`} aria-hidden="true" />

            <div className={styles.caption} role="status" aria-live="polite">
                {phase === 'float' && <p key="f" className={styles.line}>Adrift between the stars…</p>}
                {phase === 'launch' && <p key="l" className={styles.line}>Engines awake. Taking you home{name ? `, ${name}` : ''}.</p>}
                {phase === 'count' && (
                    <>
                        <p className={styles.small}>Arriving at Earth in</p>
                        <div key={count} className={styles.count}>{count}</div>
                    </>
                )}
            </div>

            <div className={`${styles.flash} ${phase === 'flash' ? styles.flashOn : ''}`} aria-hidden="true" />
        </div>
    )
}

import styles from './MessageBox.module.css'

export default function MessageBox({ message, onClose }) {
  const [displayed, setDisplayed] = useState('')
  const [done, setDone] = useState(false)

  useEffect(() => {
    if (!message) { setDisplayed(''); setDone(false); return }
    setDisplayed('')
    setDone(false)

    let i = 0
    const interval = setInterval(() => {
      if (i < message.length) {
        setDisplayed(message.slice(0, i + 1))
        i++
      } else {
        setDone(true)
        clearInterval(interval)
      }
    }, 24)

    return () => clearInterval(interval)
  }, [message])

  const handleBackdrop = (e) => {
    if (e.target === e.currentTarget) onClose()
  }

  if (!message) return null

  return (
    <div className={styles.overlay} onClick={handleBackdrop} role="dialog" aria-modal="true" aria-label="Message from the universe">
      <div className={styles.card}>
        <div className={styles.deco}>✦</div>
        <p className={styles.label}>A message from the universe</p>
        <p className={styles.text}>
          {displayed}
          {!done && <span className={styles.cursor} />}
        </p>
        {done && (
          <button className={styles.closeBtn} onClick={onClose}>
            Return to the cosmos
          </button>
        )}
      </div>
    </div>
  )
}

function getAudioMetrics(data) {
  if (!data || data.length === 0) return { bass: 0, avg: 0, mid: 0 }
  const bassEnd = Math.floor(data.length * 0.06)
  const midEnd = Math.floor(data.length * 0.35)
  let bass = 0, mid = 0, total = 0
  for (let i = 0; i < data.length; i++) {
    const v = data[i] / 255
    total += v
    if (i < bassEnd) bass += v
    else if (i < midEnd) mid += v
  }
  return { bass: bass / bassEnd, mid: mid / (midEnd - bassEnd), avg: total / data.length }
}

function fmt(sec) {
  if (!sec || isNaN(sec)) return '0:00'
  const m = Math.floor(sec / 60)
  const s = Math.floor(sec % 60).toString().padStart(2, '0')
  return `${m}:${s}`
}

function BarVisualizer({ getAudioData, playing, color, onMetrics }) {
  const canvasRef = useRef(null)
  const rafRef = useRef(null)
  const t0 = useRef(Date.now())
  const freqRef = useRef(new Uint8Array(128))

  useEffect(() => {
    const canvas = canvasRef.current
    if (!canvas) return
    const ctx = canvas.getContext('2d')
    const W = canvas.width, H = canvas.height
    const BINS = 28
    const freqData = freqRef.current

    const draw = () => {
      ctx.clearRect(0, 0, W, H)
      const t = (Date.now() - t0.current) / 1000

      if (!getAudioData || !playing) {
        // Idle sine wave
        ctx.strokeStyle = color[0] + '55'
        ctx.lineWidth = 1.5
        ctx.beginPath()
        for (let x = 0; x < W; x++) {
          const y = H / 2 + Math.sin(x * 0.18 + t * 1.2) * 3.5
            + Math.sin(x * 0.08 + t * 0.7) * 2
          x === 0 ? ctx.moveTo(x, y) : ctx.lineTo(x, y)
        }
        ctx.stroke()
        if (onMetrics) onMetrics(null)
        rafRef.current = requestAnimationFrame(draw)
        return
      }

      getAudioData(freqData)
      if (onMetrics) onMetrics(freqData)

      const step = Math.floor(freqData.length / BINS)
      const barW = (W / BINS) - 1

      const grad = ctx.createLinearGradient(0, H, 0, 0)
      grad.addColorStop(0, color[0] + 'dd')
      grad.addColorStop(0.6, color[1] + 'aa')
      grad.addColorStop(1, '#ffffff44')

      for (let i = 0; i < BINS; i++) {
        let sum = 0
        for (let j = 0; j < step; j++) sum += (freqData[i * step + j] || 0)
        const v = (sum / step) / 255
        const barH = Math.max(2, v * H * 1.15)
        const x = i * (barW + 1)
        ctx.fillStyle = grad
        ctx.beginPath()
        if (ctx.roundRect) ctx.roundRect(x, H - barH, barW, barH, 2)
        else ctx.rect(x, H - barH, barW, barH)
        ctx.fill()
      }
      rafRef.current = requestAnimationFrame(draw)
    }

    rafRef.current = requestAnimationFrame(draw)
    return () => cancelAnimationFrame(rafRef.current)
  }, [getAudioData, playing, color, onMetrics])

  return <canvas ref={canvasRef} width={196} height={32} className={styles.barCanvas} />
}

            vv.addEventListener('scroll', sync)
        }

        return () => {
            mountedRef.current = false
            clearTimeout(timerRef.current)
            document.removeEventListener('keydown', onKey)
            if (vv) {
                vv.removeEventListener('resize', sync)
                vv.removeEventListener('scroll', sync)
            }
        }
    }, [])

    // Fokus terjebak di dalam dialog (portal ada di akhir body)
    const onKeyDown = (e) => {
        if (e.key !== 'Tab') return
        const active = document.activeElement
        if (e.shiftKey && (active === closeRef.current || active === dialogRef.current)) {
            e.preventDefault()
            submitRef.current?.focus()
        } else if (!e.shiftKey && active === submitRef.current) {
            e.preventDefault()
            closeRef.current?.focus()
        }
    }

    // ── Submit ──
    const handleSubmit = async (e) => {
        e.preventDefault()
        if (status !== 'idle') return
        setAttempted(true)

        if (errors.message) return messageRef.current?.focus()
        if (errors.rating) return ratingRef.current?.focus()
        if (errors.consent) return consentRef.current?.focus()

        setStatus('sending')
        setServerError('')

        const anonymous = draft.anonymous || !draft.name.trim()
        const payload = {
            displayName: anonymous ? callsign : draft.name.trim(),
            isAnonymous: anonymous,
            category: draft.category,
            message: draft.message.trim(),
            rating: draft.rating,
            isPrivate: draft.isPrivate,
            consent: draft.consent,
            website: honeypot, // honeypot: harus kosong
        }

        try {
            const record = await submitWish(payload)
            if (!mountedRef.current) return
            sfx.reveal()
            cbRef.current.onSaved?.(record)
            setStatus('rise')
            timerRef.current = setTimeout(() => cbRef.current.onDone?.(), reduce ? RISE_MS_REDUCED : RISE_MS)
        } catch (err) {
            if (!mountedRef.current) return
            setStatus('idle')
            setServerError(
                err?.code === 'validation' || err?.code === 'rate_limit'
                    ? err.message
                    : 'Your wish got lost in the dark for a moment. Please try again.'
            )
        }
    }

    const busy = status !== 'idle'
    const rising = status === 'rise'

    const overlayStyle = viewport
        ? { height: `${viewport.h}px`, top: `${viewport.top}px` }
        : undefined

    const content = (
        <div className={`${styles.overlay} ${rising ? styles.overlayRise : ''}`} style={overlayStyle}>
            <div
                ref={dialogRef}
                className={`${styles.box} ${rising ? styles.boxOut : ''}`}
                role="dialog"
                aria-modal="true"
                aria-labelledby={`${uid}-title`}
                aria-describedby={`${uid}-notice`}
                tabIndex={-1}
                onKeyDown={onKeyDown}
            >
                <header className={styles.header}>
                    <div>
                        <p className={styles.eyebrow}>Wishing Star</p>
                        <h3 id={`${uid}-title`} className={styles.title}>
                            Whisper <em>your wish</em>
                        </h3>
                    </div>
                    <button
                        ref={closeRef}
                        type="button"
                        className={styles.close}
                        onClick={handleClose}
                        disabled={busy}
                        aria-label="Close wish form"
                    >
                        ✕
                    </button>
                </header>

                <form className={styles.form} onSubmit={handleSubmit} noValidate>
                    <div className={styles.body}>
                        <p id={`${uid}-notice`} className={styles.notice}>
                            {draft.isPrivate
                                ? 'Only the developer will read this message. Still, please don’t share personal information.'
                                : 'Your message will be displayed in the Gallery of Astronauts for other travelers to see. Please don’t include personal information such as email, phone number or address.'}
                        </p>

                        {/* Nama + anonim */}
                        <div className={styles.field}>
                            <label htmlFor={`${uid}-name`} className={styles.label}>
                                Your name <span className={styles.opt}>optional</span>
                            </label>
                            <input
                                id={`${uid}-name`}
                                type="text"
                                className={styles.input}
                                value={draft.anonymous ? callsign : draft.name}
                                onChange={(e) => set({ name: e.target.value })}
                                disabled={draft.anonymous}
                                maxLength={NAME_MAX}
                                placeholder="How should we call you?"

                                    checked={draft.anonymous}

                                <span className={styles.switchTrack} aria-hidden="true">
                                    <span className={styles.switchThumb} />

                            <p className={styles.hintSmall}>

                                    ? `You will fly as ${callsign}.`
                                    : 'Your name will appear on your astronaut card.'}
                            </p>
                        </div>

                        {/* Kategori */}
                        <fieldset className={styles.fieldset}>
                            <legend className={styles.label}>What is this about?</legend>
                            <div className={styles.chips}>
                                {CATEGORIES.map((c) => (
                                    <label key={c.id} className={styles.chip}>
                                        <input
                                            type="radio"

                            <div className={styles.metaRow}>

                        </button>

                </form>
            </div>

            {/* Bintang permintaan naik */}
            {rising && (
                <div className={styles.rise} aria-hidden="true">
                    <span className={styles.riseStar}>✦</span>
                    <p className={styles.riseText}>Your wish is rising to the stars…</p>
                </div>
            )}
            <div role="status" className={styles.srOnly}>
                {rising ? 'Your wish is rising to the stars.' : ''}
            </div>
        </div>
    )

    return createPortal(content, document.body)
}

// Bangun titik rasi bintang dari huruf pertama nama (dipakai tampilan + kartu share)
function makeConstellation(name) {
  let char = name.trim().charAt(0).toUpperCase()
  if (!ALPHABET_POINTS[char]) char = 'O'
  return ALPHABET_POINTS[char].map(([x, y]) => ({
    x: 90 + x * 1.2 + (Math.random() - 0.5) * 8,
    y: 20 + y * 1.2 + (Math.random() - 0.5) * 8,
  }))
}

// ── SHARE CARD: render konstelasi + pesan ke PNG 1080×1350 ─────────────
function wrapLines(g, text, maxWidth) {
  const words = text.split(' ')
  const lines = []
  let line = ''
  for (const w of words) {
    const test = line ? `${line} ${w}` : w
    if (g.measureText(test).width > maxWidth && line) {
      lines.push(line)
      line = w
    } else {
      line = test
    }
  }
  if (line) lines.push(line)
  return lines
}

async function drawShareCard({ name, from, message, points, coords }) {
  try {
    await Promise.all([
      document.fonts.load('italic 300 60px "Cormorant Garamond"'),
      document.fonts.load('300 24px "Outfit"'),
      document.fonts.load('400 24px "Outfit"'),
    ])
  } catch (_) { /* font fallback */ }

  const W = 1080, H = 1350
  const canvas = document.createElement('canvas')
  canvas.width = W
  canvas.height = H
  const g = canvas.getContext('2d')
  const serif = '"Cormorant Garamond", Georgia, serif'
  const sans = '"Outfit", system-ui, sans-serif'
  // Warna teks kecil di kartu: lantai sama dengan --text-muted (#9fb0d4)
  const MUTED = '#9fb0d4'
  const setSpacing = (px) => { if ('letterSpacing' in g) g.letterSpacing = `${px}px` }

  // background
  const bg = g.createLinearGradient(0, 0, 0, H)
  bg.addColorStop(0, '#02021a')
  bg.addColorStop(0.55, '#0b0720')
  bg.addColorStop(1, '#16091f')
  g.fillStyle = bg
  g.fillRect(0, 0, W, H)

  // soft nebula washes
  const wash = (x, y, r, color) => {
    const rg = g.createRadialGradient(x, y, 0, x, y, r)
    rg.addColorStop(0, color)
    rg.addColorStop(1, 'rgba(0,0,0,0)')
    g.fillStyle = rg
    g.fillRect(0, 0, W, H)
  }
  wash(220, 260, 520, 'rgba(80,110,255,0.16)')
  wash(900, 1050, 560, 'rgba(160,80,255,0.14)')

  // star dust
  for (let i = 0; i < 320; i++) {
    const r = Math.random() * 1.6 + 0.3
    g.fillStyle = `rgba(210,225,255,${Math.random() * 0.7 + 0.15})`
    g.beginPath()
    g.arc(Math.random() * W, Math.random() * H, r, 0, Math.PI * 2)
    g.fill()
  }

  // frame
  g.strokeStyle = 'rgba(136,170,255,0.25)'
  g.lineWidth = 2
  g.beginPath()
  g.roundRect ? g.roundRect(40, 40, W - 80, H - 80, 28) : g.rect(40, 40, W - 80, H - 80)
  g.stroke()

  // constellation (fit into 520×340 area centred at y=330)
  const xs = points.map((p) => p.x), ys = points.map((p) => p.y)
  const minX = Math.min(...xs), maxX = Math.max(...xs)
  const minY = Math.min(...ys), maxY = Math.max(...ys)
  const scale = Math.min(520 / ((maxX - minX) || 1), 340 / ((maxY - minY) || 1))
  const pts = points.map((p) => ({
    x: W / 2 + (p.x - (minX + maxX) / 2) * scale,
    y: 330 + (p.y - (minY + maxY) / 2) * scale,
  }))

  g.strokeStyle = 'rgba(136,170,255,0.55)'
  g.lineWidth = 2.5
  g.setLineDash([10, 10])
  g.beginPath()
  pts.forEach((p, i) => (i ? g.lineTo(p.x, p.y) : g.moveTo(p.x, p.y)))
  g.stroke()
  g.setLineDash([])

  pts.forEach((p) => {
    const halo = g.createRadialGradient(p.x, p.y, 0, p.x, p.y, 30)
    halo.addColorStop(0, 'rgba(160,190,255,0.85)')
    halo.addColorStop(1, 'rgba(160,190,255,0)')
    g.fillStyle = halo
    g.fillRect(p.x - 30, p.y - 30, 60, 60)
    g.fillStyle = '#fff'
    g.beginPath()
    g.arc(p.x, p.y, 7, 0, Math.PI * 2)
    g.fill()
  })

  g.textAlign = 'center'

  // label under constellation
  g.fillStyle = MUTED
  g.font = `400 24px ${sans}`
  setSpacing(5)
  g.fillText(`CONSTELLATION ${name.toUpperCase()}`, W / 2, 560)

  // divider
  g.strokeStyle = 'rgba(255,255,255,0.12)'
  g.lineWidth = 1
  g.beginPath()
  g.moveTo(160, 600)
  g.lineTo(W - 160, 600)
  g.stroke()

  // header
  g.fillStyle = MUTED
  g.font = `400 24px ${sans}`
  setSpacing(4)
  g.fillText('DECRYPTED TRANSMISSION', W / 2, 668)
  setSpacing(0)

  // "Dear Name," mengecil sampai muat
  let size = 70
  g.fillStyle = '#aaccff'
  do {
    g.font = `italic 300 ${size}px ${serif}`
    size -= 4
  } while (g.measureText(`Dear ${name},`).width > 880 && size > 30)
  g.fillText(`Dear ${name},`, W / 2, 770)

  // message: font mengecil sampai muat di atas tanda tangan/footer
  g.fillStyle = '#e0e5ff'
  let msgSize = 54
  let lines = []
  while (true) {
    g.font = `italic 300 ${msgSize}px ${serif}`
    lines = wrapLines(g, `“${message}”`, 840)
    if (lines.length * msgSize * 1.45 <= 270 || msgSize <= 32) break
    msgSize -= 2
  }
  const lh = msgSize * 1.45
  lines.forEach((ln, i) => g.fillText(ln, W / 2, 870 + i * lh))

  // signature (untuk "someone else")
  if (from) {
    const sigY = Math.min(870 + (lines.length - 1) * lh + 105, 1185)
    g.fillStyle = MUTED
    g.font = `400 24px ${sans}`
    setSpacing(3)
    g.fillText('WITH COSMIC LOVE,', W / 2, sigY - 44)
    setSpacing(0)
    let sigSize = 56
    g.fillStyle = '#ff99cc'
    do {
      g.font = `italic 300 ${sigSize}px ${serif}`
      sigSize -= 4
    } while (g.measureText(from).width > 700 && sigSize > 28)
    g.fillText(from, W / 2, sigY + 8)
  }

  // footer
  g.fillStyle = MUTED
  g.font = `400 24px ${sans}`
  setSpacing(4)
  g.fillText('COSMIC JOURNEY', W / 2, H - 110)
  g.font = `400 21px ${sans}`
  setSpacing(2)
  g.fillText(`COSMIC COORDS: ${coords}`, W / 2, H - 76)
  setSpacing(0)

  return new Promise((resolve) => canvas.toBlob(resolve, 'image/png'))
}

function timeAgo(iso) {
    const s = Math.max(0, (Date.now() - new Date(iso).getTime()) / 1000)
    if (s < 60) return 'just now'
    const m = Math.floor(s / 60)
    if (m < 60) return `${m}m ago`
    const h = Math.floor(m / 60)
    if (h < 24) return `${h}h ago`
    const d = Math.floor(h / 24)
    if (d < 7) return `${d}d ago`
    const w = Math.floor(d / 7)
    if (d < 30) return `${w}w ago`
    const mo = Math.floor(d / 30)
    if (mo < 12) return `${mo}mo ago`
    return `${Math.floor(d / 365)}y ago`
}

import { memo, useEffect, useId, useRef, useState } from 'react'
import { sfx } from '../hooks/useSfx.js'
import WishForm from '../components/WishForm.jsx'
import styles from './Wishing.module.css'

const rand = (a, b) => a + Math.random() * (b - a)
const randInt = (a, b) => Math.floor(rand(a, b + 1))

const HINT_AFTER = 20      // detik tanpa tangkapan sebelum glow bintang terang naik
const FADE_MS = 650        // bintang biasa memudar setelah di-tap
const BURST_MS = 1400      // ledakan cahaya bintang terang
const BURST_MS_REDUCED = 900

function Wishing({
    visitor,
    onVisitorChange,
    onGallery,
    onSubmitted,
    onBackToConstellation,
    onReturnHome,
}) {
    const uid = useId()
    const wrapRef = useRef(null)
    const canvasRef = useRef(null)
    const catchRef = useRef(null)        // diisi effect canvas: tangkap bintang terang lewat tombol bantu
    const leavingRef = useRef(false)     // true setelah Skip: abaikan onCaught yang terlambat
    const resumeBtnRef = useRef(null)
    const doneBtnRef = useRef(null)

    // Wish sudah terkirim di perjalanan ini (mis. kembali lewat NavDots dari Secret Ending / Gallery):
    // tidak ada permainan lagi supaya tidak terjadi kiriman ganda. Dibaca sekali saat mount.
    const alreadyDone = useRef(Boolean(visitor?.wishSubmitted)).current

    const [round, setRound] = useState(0)          // naik = main ulang (lepas bintang)
    const [caught, setCaught] = useState(false)    // bintang terang sudah tertangkap
    const [formOpen, setFormOpen] = useState(false)
    const [draft, setDraft] = useState(null)       // isi form yang tersimpan saat form ditutup

    // Dipanggil dari loop canvas (lewat ref agar effect tidak perlu dibuat ulang)
    const handleCaught = () => {
        if (leavingRef.current) return
        setCaught(true)
        setFormOpen(true)
    }
    const caughtCbRef = useRef(handleCaught)
    caughtCbRef.current = handleCaught

    useEffect(() => {
        if (alreadyDone) return
        const canvas = canvasRef.current
        const wrap = wrapRef.current
        if (!canvas || !wrap) return
        const ctx = canvas.getContext('2d')
        const reduce = window.matchMedia('(prefers-reduced-motion: reduce)').matches

        // ── Ukuran canvas (DPR dibatasi 1.5 agar hemat di HP) ───────────
        let w = 0, h = 0
        let settled = false   // true = loop sudah berhenti (frame terakhir tampil), resize menggambar ulang
        const resize = () => {
            const r = wrap.getBoundingClientRect()
            w = r.width
            h = r.height
            const dpr = Math.min(window.devicePixelRatio || 1, 1.5)
            canvas.width = Math.max(1, Math.round(w * dpr))
            canvas.height = Math.max(1, Math.round(h * dpr))
            ctx.setTransform(dpr, 0, 0, dpr, 0, 0)
            if (settled) draw()
        }
        resize()
        const ro = new ResizeObserver(resize)
        ro.observe(wrap)

        // ── State permainan (semua di ref/closure, tanpa re-render) ─────
        // Antrean: false = bintang biasa, true = bintang terang. Terang tidak pernah jadi yang pertama.
        const total = reduce ? randInt(5, 6) : randInt(5, 10)
        const queue = Array.from({ length: total - 1 }, () => false)
        queue.splice(randInt(1, queue.length), 0, true)

        const stars = []
        let time = 0
        let nextSpawn = 0.9
        let boost = 0                 // 0..1, glow petunjuk bintang terang
        let phase = 'play'            // 'play' | 'burst'
        let burst = null

        const burstMs = reduce ? BURST_MS_REDUCED : BURST_MS

        const spawn = (bright) => {
            const x = rand(0.06 * w, 0.94 * w)
            let dir = x < w / 2 ? 1 : -1            // arah ke sisi layar yang lebih lebar
            if (!bright && Math.random() < 0.2) dir = -dir
            const room = dir > 0 ? w * 0.97 - x : x - w * 0.03
            let ang = bright ? rand(0.3, 0.7) : rand(0.35, 1.0)

                ang = Math.min(ang, Math.atan(Math.max(room, 40) / (h * 0.75)))
            }

            const base = Math.min(Math.max(h, 360), 900)
            const speed = (bright ? rand(0.15, 0.19) : rand(0.16, 0.26)) * base * (reduce ? 0.6 : 1)
            stars.push({
                x,
                y: rand(-30, -10),
                vx: dir * Math.sin(ang) * speed,
                vy: Math.cos(ang) * speed,
                bright,
                r: bright ? 4.6 : rand(1.6, 2.6),
                tail: bright ? rand(190, 230) : rand(60, 120),
                state: 'fall',          // fall | fade | caught

            const alpha = s.state === 'fade' ? Math.max(0, 1 - s.t / FADE_MS) : 1

            const ty = s.y - uy * s.tail
            const g = ctx.createLinearGradient(tx, ty, s.x, s.y)

            }

            ctx.beginPath()
            ctx.arc(s.x, s.y, s.r, 0, Math.PI * 2)
            ctx.fill()

            // riak kecil saat bintang biasa di-tap

        }

        const drawBurst = () => {
            const p = burst.t / burstMs
            const e = 1 - Math.pow(1 - p, 3)               // ease-out
            const diag = Math.hypot(w, h)
            const A = p < 0.5 ? p * 2 : 1 - (p - 0.5) * 0.9 // naik cepat, mengendap ke ~0.55
            const R = 20 + e * diag * 0.6

            const g = ctx.createRadialGradient(burst.x, burst.y, 0, burst.x, burst.y, R)
            g.addColorStop(0, `rgba(255,255,255,${A})`)
            g.addColorStop(0.25, `rgba(205,222,255,${A * 0.8})`)
            g.addColorStop(1, 'rgba(120,150,255,0)')
            ctx.fillStyle = g
            ctx.fillRect(0, 0, w, h)

            // inti bintang membesar
            ctx.fillStyle = `rgba(255,255,255,${Math.min(1, 0.4 + p)})`
            ctx.beginPath()
            ctx.arc(burst.x, burst.y, 5 + e * 38, 0, Math.PI * 2)
            ctx.fill()

            // gelombang kejut (dilewati saat reduced motion)
            if (!reduce) {
                ctx.strokeStyle = `rgba(190,210,255,${Math.max(0, 0.7 * (1 - p))})`
                ctx.lineWidth = 2
                ctx.beginPath()
                ctx.arc(burst.x, burst.y, e * Math.min(w, h) * 0.55, 0, Math.PI * 2)
                ctx.stroke()
            }
        }

        function draw() {
            ctx.clearRect(0, 0, w, h)
            ctx.save()
            if (burst && !reduce && burst.t < 350) {      // getaran singkat, tanpa saat reduced motion
                const k = 3 * (1 - burst.t / 350)
                ctx.translate((Math.random() - 0.5) * k, (Math.random() - 0.5) * k)
            }
            stars.forEach(drawStar)
            if (burst) drawBurst()
            ctx.restore()
        }

        let raf = 0
        let last = performance.now()
        const frame = (now) => {
            const dt = Math.min((now - last) / 1000, 0.05)
            last = now
            update(dt)
            draw()
            // Setelah ledakan selesai hanya bintang tertangkap yang tersisa: tidak ada yang bergerak lagi.
            // Hentikan loop agar HP tidak menggambar ulang 60fps di belakang form (backdrop blur).
            if (done && stars.every((s) => s.state === 'caught')) {
                settled = true
                raf = 0
                return
            }
            raf = requestAnimationFrame(frame)
        }
        raf = requestAnimationFrame(frame)

        return () => {
            cancelAnimationFrame(raf)
            canvas.removeEventListener('pointerdown', onPointerDown)
            ro.disconnect()
            catchRef.current = null
        }
    }, [round, alreadyDone])

    // Fokus: panel "sudah jadi bintang" saat mount; kartu resume saat form ditutup
    useEffect(() => {
        if (alreadyDone) doneBtnRef.current?.focus({ preventScroll: true })
    }, [alreadyDone])

    useEffect(() => {
        if (caught && !formOpen) resumeBtnRef.current?.focus({ preventScroll: true })
    }, [caught, formOpen])

    const handleSkip = () => {
        sfx.click()
        leavingRef.current = true
        onVisitorChange?.({ skipped: true })   // flag: skip TIDAK membuka Secret Ending
        onGallery?.()
    }

    // Form ditutup tanpa kirim: simpan isi form, tawarkan buka lagi / lepas bintang
    const handleFormClose = (savedDraft) => {
        setDraft(savedDraft)
        setFormOpen(false)
    }

    // Kiriman sukses: simpan ke state App (mySubmission)
    const handleSaved = (record) => {
        onVisitorChange?.({ mySubmission: record, wishSubmitted: true, skipped: false })
    }

    // Animasi bintang naik selesai → Secret Ending
    const handleDone = () => {
        onSubmitted?.()
    }

    const handleRelease = () => {
        sfx.click()
        setFormOpen(false)
        setCaught(false)
        setRound((r) => r + 1)
    }

    const handleReopen = () => {
        sfx.open()
        setFormOpen(true)
    }

    const act = (fn) => () => {
        sfx.click()
        leavingRef.current = true
        fn?.()
    }

    // ── Wish sudah terkirim: tidak ada permainan, hanya pilihan lanjut ──
    if (alreadyDone) {
        return (
            <section id="wishing" className={styles.wishing}>
                <div className={styles.resume} role="region" aria-labelledby={`${uid}-done`}>
                    <div className={styles.card}>
                        <p className={styles.cardStar} aria-hidden="true">✦</p>
                        <p id={`${uid}-done`} className={styles.cardTitle}>Your wish is already a star</p>
                        <p className={styles.cardText}>
                            One wish for each journey keeps every star special.
                            Return home whenever you wish to begin a new one.
                        </p>
                        <div className={styles.cardActions}>
                            <button ref={doneBtnRef} className={styles.cardBtn} onClick={act(onBackToConstellation)}>
                                Stay with my constellation
                            </button>
                            <button className={styles.cardBtn} onClick={act(onGallery)}>
                                View the Gallery
                            </button>
                            <button className={styles.cardBtn} onClick={act(onReturnHome)}>
                                Return home
                            </button>
                        </div>
                    </div>
                </div>
            </section>
        )
    }

    return (
        <section id="wishing" className={styles.wishing}>
            <div ref={wrapRef} className={styles.sky}>
                <canvas
                    ref={canvasRef}
                    className={styles.canvas}
                    role="img"
                    aria-label="Night sky with falling stars. Tap the brightest one to make a wish."
                />
            </div>

            {/* Jalur keyboard / screen reader: terlihat hanya saat difokus */}
            {!caught && (
                <button type="button" className={styles.assist} onClick={() => catchRef.current?.()}>
                    Catch the brightest star
                </button>
            )}

            <div className={styles.caption}>
                <p className={styles.label}>Wishing Star</p>
                <h2 className={styles.title}>
                    Make <em>a wish</em>
                </h2>
                <p className={styles.hint}>Find the brightest falling star</p>
            </div>

            <button className={styles.skip} onClick={handleSkip}>
                Skip to gallery
            </button>

            {/* Form ditutup setelah bintang tertangkap: bisa dibuka lagi atau bintang dilepas */}
            {caught && !formOpen && (
                <div className={styles.resume}>
                    <div
                        className={styles.card}
                        role="dialog"
                        aria-modal="true"
                        aria-labelledby={`${uid}-resume`}
                    >
                        <p className={styles.cardStar} aria-hidden="true">✦</p>
                        <p id={`${uid}-resume`} className={styles.cardTitle}>Your star is still glowing</p>
                        <p className={styles.cardText}>
                            {draft?.message?.trim()
                                ? 'Your words are safe. Pick up where you left off.'
                                : 'Whisper your wish whenever you are ready.'}
                        </p>
                        <div className={styles.cardActions}>
                            <button ref={resumeBtnRef} className={styles.cardBtn} onClick={handleReopen}>
                                Continue my wish
                            </button>
                            <button className={styles.cardBtn} onClick={handleRelease}>
                                Release the star
                            </button>
                        </div>
                    </div>
                </div>
            )}

            {formOpen && (
                <WishForm
                    initialDraft={draft}
                    onClose={handleFormClose}
                    onSaved={handleSaved}
                    onDone={handleDone}
                />
            )}
        </section>
    )
}

// ============================================================
// COSMIC JOURNEY — Wishes service (Fase 6: backend nyata)
// submitWish → POST /api/submit-wish, getWishes → GET /api/get-wishes
// Kontrak fungsi SAMA dengan versi dummy (Fase 2 & 4); UI tidak berubah.
// Tidak ada service key di sini: semua rahasia hanya ada di Netlify Functions.
// ============================================================

export const MESSAGE_MIN = 5
export const MESSAGE_MAX = 280
export const NAME_MAX = 30
export const PAGE_SIZE = 12

// id = nilai yang dikirim ke backend (whitelist server-side di netlify/lib/shared.js)
export const CATEGORIES = [
    { id: 'feedback', label: 'Feedback' },
    { id: 'suggestion', label: 'Suggestion' },
    { id: 'developer', label: 'About the developer' },
    { id: 'wish', label: 'Just a wish' },
]
const CATEGORY_IDS = new Set(CATEGORIES.map((c) => c.id))

const API_BASE = '/api'
const SUBMIT_TIMEOUT = 15000
const GET_TIMEOUT = 10000
const ERROR_CODES = new Set(['validation', 'rate_limit', 'network', 'server'])

// ── Callsign anonim: dibuat sekali per sesi (mis. "Astronaut-7F2") ──
const CALLSIGN_CHARS = '23456789ABCDEFGHJKLMNPQRSTUVWXYZ' // tanpa 0/O/1/I agar tidak membingungkan
let sessionCallsign = null

export function getSessionCallsign() {
    if (sessionCallsign) return sessionCallsign
    const buf = new Uint32Array(3)
    if (typeof window !== 'undefined' && window.crypto?.getRandomValues) {
        window.crypto.getRandomValues(buf)
    } else {
        for (let i = 0; i < buf.length; i++) buf[i] = Math.floor(Math.random() * 4294967296)
    }
    const code = Array.from(buf, (n) => CALLSIGN_CHARS[n % CALLSIGN_CHARS.length]).join('')
    sessionCallsign = `Astronaut-${code}`
    return sessionCallsign
}

// Dipanggil saat pengunjung "pulang ke Bumi" (perjalanan baru = pengunjung baru)
export function resetSessionCallsign() {
    sessionCallsign = null
}

// ── Pembersihan teks (server melakukan hal yang sama dan tetap menjadi penentu) ──
const cleanName = (s) =>
    String(s ?? '').replace(/[\u0000-\u001F\u007F-\u009F]/g, ' ').replace(/\s+/g, ' ').trim()

const cleanMessage = (s) =>
    String(s ?? '')
        .replace(/\r\n?/g, '\n')
        .replace(/[\u0000-\u0008\u000B\u000C\u000E-\u001F\u007F-\u009F]/g, '')
        .replace(/\n{3,}/g, '\n\n')
        .trim()

const wait = (ms) => new Promise((r) => setTimeout(r, ms))

function fail(code, message) {
    const err = new Error(message)
    err.code = code
    return err
}

// ── Pemanggil HTTP: semua kegagalan diubah menjadi Error ber-.code ──
async function request(path, { method = 'GET', body, timeout }) {
    const ctrl = new AbortController()
    const timer = setTimeout(() => ctrl.abort(), timeout)
    try {
        let res
        try {
            res = await fetch(`${API_BASE}/${path}`, {
                method,
                headers: body ? { 'Content-Type': 'application/json' } : undefined,
                body: body ? JSON.stringify(body) : undefined,
                signal: ctrl.signal,
            })
        } catch (_) {
            // offline, DNS, timeout, atau diblokir
            throw fail('network', 'Could not reach the observatory.')
        }

        // Bila backend mati / /api tidak ada, server bisa membalas HTML (bukan JSON) → data = null
        let data = null
        try { data = await res.json() } catch (_) { /* bukan JSON */ }

        if (!res.ok) {
            const e = data?.error
            if (e && ERROR_CODES.has(e.code)) {
                throw fail(e.code, typeof e.message === 'string' ? e.message : 'Something went wrong.')
            }
            throw fail(res.status === 429 ? 'rate_limit' : 'server', 'The observatory is quiet right now.')
        }
        if (!data || typeof data !== 'object') {
            throw fail('server', 'The observatory is quiet right now.')
        }
        return data
    } finally {
        clearTimeout(timer)
    }
}

/**
 * Kontrak (sama seperti sebelumnya):
 *   submitWish({ displayName, isAnonymous, category, message, rating, isPrivate, consent, website })
 *   → Promise<record>  atau throw Error dengan .code: 'validation' | 'rate_limit' | 'network' | 'server'
 *   record = { id, displayName, isAnonymous, category, message, rating, isPrivate, createdAt, status }
 * `website` = field honeypot (harus kosong; server membalas sukses palsu tanpa menyimpan bila terisi).
 * POST TIDAK di-retry otomatis agar tidak terjadi kiriman ganda.
 */
export async function submitWish(payload) {
    // Pemeriksaan cepat di klien (server tetap memvalidasi ulang secara ketat)
    const message = cleanMessage(payload?.message)
    const rating = Number(payload?.rating)
    const isPrivate = Boolean(payload?.isPrivate)

    if (message.length < MESSAGE_MIN || message.length > MESSAGE_MAX) {
        throw fail('validation', `Your message needs ${MESSAGE_MIN}–${MESSAGE_MAX} characters.`)
    }
    if (!Number.isInteger(rating) || rating < 1 || rating > 5) {
        throw fail('validation', 'Please choose a rating from 1 to 5 stars.')
    }
    if (!CATEGORY_IDS.has(payload?.category)) {
        throw fail('validation', 'Please choose a category.')
    }
    if (!isPrivate && !payload?.consent) {
        throw fail('validation', 'Please agree to show your message in the Gallery, or send it privately.')
    }

    const isAnonymous = Boolean(payload?.isAnonymous)
    const displayName = cleanName(payload?.displayName).slice(0, NAME_MAX) || getSessionCallsign()

    const record = await request('submit-wish', {
        method: 'POST',
        timeout: SUBMIT_TIMEOUT,
        body: {
            displayName,
            isAnonymous,
            category: payload.category,
            message,
            rating,
            isPrivate,
            consent: Boolean(payload?.consent),
            website: typeof payload?.website === 'string' ? payload.website : '',
        },
    })

    if (!record.id || typeof record.message !== 'string') {
        throw fail('server', 'The observatory is quiet right now.')
    }
    return record
}

/**
 * Kontrak (sama seperti sebelumnya):
 *   getWishes({ sort = 'newest' | 'rating', category = 'all' | <id kategori>, cursor = null, limit = 12 })
 *   → Promise<{
 *        items: [{ id, displayName, isAnonymous, category, message, rating, createdAt }],
 *        nextCursor: string | null,     // opaque: kirim balik apa adanya; null = sudah habis
 *        total: number,                 // jumlah item yang cocok dengan filter kategori
 *        aggregate: { average: number, count: number }  // semua wish publik, TIDAK terpengaruh filter
 *     }>
 *   atau throw Error dengan .code: 'validation' | 'network' | 'server'
 * GET aman di-retry: satu kali ulang bila jaringan gagal.
 */
export async function getWishes({ sort = 'newest', category = 'all', cursor = null, limit = PAGE_SIZE } = {}) {
    if (sort !== 'newest' && sort !== 'rating') throw fail('validation', 'Unknown sort.')
    if (category !== 'all' && !CATEGORY_IDS.has(category)) throw fail('validation', 'Unknown category.')
    const pageSize = Math.min(50, Math.max(1, Math.floor(Number(limit)) || PAGE_SIZE))

    const q = new URLSearchParams({ sort, category, limit: String(pageSize) })
    if (cursor) q.set('cursor', String(cursor))
    const path = `get-wishes?${q}`

    let data
    try {
        data = await request(path, { timeout: GET_TIMEOUT })
    } catch (err) {
        if (err.code !== 'network') throw err
        await wait(700)
        data = await request(path, { timeout: GET_TIMEOUT })
    }

    if (!Array.isArray(data.items) || !data.aggregate) {
        throw fail('server', 'The observatory is quiet right now.')
    }
    return {
        items: data.items,
        nextCursor: typeof data.nextCursor === 'string' ? data.nextCursor : null,
        total: Number(data.total) || 0,
        aggregate: {
            count: Number(data.aggregate.count) || 0,
            average: Number(data.aggregate.average) || 0,
        },
    }
}
