import { useEffect, useMemo, useRef } from 'react'
import styles from './EntryGate.module.css'

const STAR_COUNT = 36
const EXIT_MS = 700
const EXIT_MS_REDUCED = 150

const prefersReduced = () =>
    typeof window !== 'undefined' &&
    window.matchMedia('(prefers-reduced-motion: reduce)').matches

// Bintang gerbang dibuat sekali per mount (murni CSS, tanpa canvas)
function makeStars() {
    return Array.from({ length: STAR_COUNT }, () => ({
        x: Math.random() * 100,
        y: Math.random() * 100,
        size: 1 + Math.random() * 2,
        dur: 2.4 + Math.random() * 3.2,
        delay: -Math.random() * 5,
        o: 0.45 + Math.random() * 0.55,
    }))
}

// onEnter(true)  = masuk dengan suara (HARUS dipanggil sinkron di handler ketukan)
// onEnter(false) = masuk tanpa suara
// leaving        = App sudah menerima ketukan, gerbang memudar lalu memanggil onExited
export default function EntryGate({ onEnter, leaving = false, onExited }) {
    const stars = useMemo(makeStars, [])
    const rootRef = useRef(null)
    const boardRef = useRef(null)
    const exitedRef = useRef(onExited)
    exitedRef.current = onExited

    useEffect(() => {
        boardRef.current?.focus({ preventScroll: true })
    }, [])

    useEffect(() => {
        if (!leaving) return
        const t = setTimeout(
            () => exitedRef.current?.(),
            prefersReduced() ? EXIT_MS_REDUCED : EXIT_MS
        )
        return () => clearTimeout(t)
    }, [leaving])

    // Fokus dikunci di dalam gerbang (dua tombol saja)
    const onKeyDown = (e) => {
        if (e.key !== 'Tab') return
        const items = rootRef.current?.querySelectorAll('button')
        if (!items || items.length === 0) return
        const first = items[0]
        const last = items[items.length - 1]
        if (e.shiftKey && document.activeElement === first) {
            e.preventDefault()
            last.focus()
        } else if (!e.shiftKey && document.activeElement === last) {
            e.preventDefault()
            first.focus()
        }
    }

    const choose = (withSound) => {
        if (leaving) return
        onEnter(withSound)
    }

    return (
        <div
            ref={rootRef}
            className={`${styles.gate} ${leaving ? styles.leaving : ''}`}
            role="dialog"
            aria-modal="true"
            aria-labelledby="gate-title"
            aria-describedby="gate-note"
            onKeyDown={onKeyDown}
        >
            <div className={styles.sky} aria-hidden="true">
                {stars.map((s, i) => (
                    <span
                        key={i}
                        className={styles.star}
                        style={{
                            left: `${s.x}%`,
                            top: `${s.y}%`,
                            width: `${s.size}px`,
                            height: `${s.size}px`,
                            '--d': `${s.dur}s`,
                            '--delay': `${s.delay}s`,
                            '--o': s.o,
                        }}
                    />
                ))}
            </div>

            <div className={styles.content}>
                <div className={styles.core} aria-hidden="true">
                    <span className={styles.coreRing} />
                    <span className={styles.coreDot} />
                </div>

                <h2 id="gate-title" className={styles.kicker}>Cosmic Journey</h2>

                <button
                    ref={boardRef}
                    type="button"
                    className={styles.boardBtn}
                    onClick={() => choose(true)}
                >
                    Board the ship
                </button>

                <p id="gate-note" className={styles.note}>Best with sound on</p>

                <button
                    type="button"
                    className={styles.quiet}
                    onClick={() => choose(false)}
                >
                    Continue without sound
                </button>
            </div>
        </div>
    )
}

import { useCallback, useEffect, useLayoutEffect, useRef, useState } from 'react'
import { UNIVERSE_LEVELS as LEVELS } from '../data/index.js'
import { sfx } from '../hooks/useSfx.js'
import { SCENES, RHO, CX, CY, EARTH_ANCHOR, sunAt, diskTransform } from './universeScenes.jsx'
import styles from './UniverseScale.module.css'

const LAST = LEVELS.length - 1
const clamp = (n, a, b) => Math.min(Math.max(n, a), b)
const sstep = (x) => { const t = clamp(x, 0, 1); return t * t * (3 - 2 * t) }
const OMEGA = 0.14 // rad/detik, putaran galaksi (≈45 detik per putaran)

/**
 * Kamera zoom antar scene. Scene i-1 "hidup" di dalam scene i pada titik A[i]
 * (Bumi di orbitnya, Tata Surya di sekitar Matahari, dst) dan tampil dengan
 * skala RHO[i]. Pada transisi t∈[0,1]:
 *  - scene i-1 (anak) mengecil dari 1 → RHO[i] menuju titik A[i]
 *  - scene i (induk) membesar dari 1/RHO[i] → 1, dengan A[i] berpindah dari pusat
 * Keduanya memakai titik b(t) yang sama, jadi selalu konsisten.
 * Hasil: transform layar = s·p + (x, y) dalam koordinat viewBox 400×250.
 */
function layerState(i, v, A) {
  if (i >= 1 && v < i) {                       // masuk sebagai induk
    const t = v - (i - 1)
    if (t <= 0) return null
    const a = A[i], s = Math.pow(RHO[i], t - 1)
    const bx = CX + (a[0] - CX) * t, by = CY + (a[1] - CY) * t
    return { s, x: bx - s * a[0], y: by - s * a[1], op: sstep((t - 0.5) / 0.4) }
  }
  if (i < LAST) {                              // keluar sebagai anak
    const t = v - i
    if (t >= 1) return null
    const a = A[i + 1], s = Math.pow(RHO[i + 1], t)
    const bx = CX + (a[0] - CX) * t, by = CY + (a[1] - CY) * t
    return { s, x: bx - s * CX, y: by - s * CY, op: 1 - sstep((t - 0.65) / 0.3) }
  }
  return { s: 1, x: 0, y: 0, op: 1 }           // scene terakhir
}

// 2 angka signifikan: 704.500 → "700,000"
const sig2 = (n) => {
  const p = Math.pow(10, Math.floor(Math.log10(n)) - 1)
  return (Math.round(n / p) * p).toLocaleString('en-US')
}

export default function UniverseScale() {
  const [v, setV] = useState(0)          // posisi slider (float, 0..5)
  const vRef = useRef(0)
  const target = useRef(0)
  const dragging = useRef(false)
  const raf = useRef(0)
  const trackRef = useRef(null)
  const reduce = useRef(false)
  const firstRun = useRef(true)
  const figRef = useRef(null)
  const layers = useRef([])
  const diskRef = useRef(null)
  const sunRef = useRef(null)
  const theta = useRef(0.6)
  const inView = useRef(true)
  // Titik tujuan tiap scene di dalam scene berikutnya (index = scene induk)
  const anchors = useRef([null, EARTH_ANCHOR, [CX, CY], sunAt(0.6), [CX, CY], [CX, CY]])

  const apply = useCallback((val) => {
    layers.current.forEach((el, i) => {
      if (!el) return
      const st = layerState(i, val, anchors.current)
      if (!st || st.op < 0.01) {
        el.style.visibility = 'hidden'
        el.style.opacity = 0
        el.classList.add(styles.idle)
        return
      }
      el.classList.remove(styles.idle)
      el.style.visibility = 'visible'
      el.style.opacity = st.op
      el.style.transform = `translate(${(st.x / 400) * 100}%, ${(st.y / 250) * 100}%) scale(${st.s})`
    })
  }, [])

  const spinGalaxy = useCallback((th) => {
    diskRef.current?.setAttribute('transform', diskTransform(th))
    const [sx, sy] = sunAt(th)
    sunRef.current?.setAttribute('transform', `translate(${sx} ${sy})`)
    anchors.current[3] = [sx, sy]
  }, [])

  useLayoutEffect(() => { spinGalaxy(theta.current); apply(v) }, [v, apply, spinGalaxy])

  // Putaran galaksi: hanya jalan saat scene Milky Way terlihat dan komponen di layar
  useEffect(() => {
    const io = new IntersectionObserver(([en]) => { inView.current = en.isIntersecting }, { threshold: 0.05 })
    if (figRef.current) io.observe(figRef.current)
    let id = 0, last = performance.now()
    const loop = (now) => {
      id = requestAnimationFrame(loop)
      const dt = Math.min((now - last) / 1000, 0.1)
      last = now
      const val = vRef.current
      if (reduce.current || !inView.current || val <= 2 || val >= 5) return
      theta.current -= dt * OMEGA   // negatif: lengan spiral "tertinggal" (trailing)
      spinGalaxy(theta.current)
      apply(val)
    }
    id = requestAnimationFrame(loop)
    return () => { cancelAnimationFrame(id); io.disconnect() }
  }, [apply, spinGalaxy])

  useEffect(() => {
    reduce.current = window.matchMedia('(prefers-reduced-motion: reduce)').matches
    return () => cancelAnimationFrame(raf.current)
  }, [])

  const level = Math.round(v)

  // Bunyi saat level berganti (termasuk saat menyapu slider)
  useEffect(() => {
    if (firstRun.current) { firstRun.current = false; return }
    sfx.chime(level)
  }, [level])

  const animate = useCallback(() => {
    let last = performance.now()
    const step = (now) => {
      if (dragging.current) { raf.current = 0; return }
      const dt = Math.min((now - last) / 1000, 0.1)
      last = now
      const d = target.current - vRef.current
      if (Math.abs(d) < 0.002) {
        vRef.current = target.current
        setV(target.current)
        raf.current = 0
        return
      }
      vRef.current += d * (1 - Math.exp(-dt * 4.5))
      setV(vRef.current)
      raf.current = requestAnimationFrame(step)
    }
    raf.current = requestAnimationFrame(step)
  }, [])

  const goTo = useCallback((i) => {
    target.current = clamp(i, 0, LAST)
    if (reduce.current) { vRef.current = target.current; setV(target.current); return }
    if (!raf.current) animate()
  }, [animate])

  const fromPointer = (e) => {
    const r = trackRef.current.getBoundingClientRect()
    const f = clamp((e.clientX - r.left) / r.width, 0, 1)
    vRef.current = f * LAST
    setV(vRef.current)
  }
  const onDown = (e) => {
    e.currentTarget.setPointerCapture(e.pointerId)
    dragging.current = true
    cancelAnimationFrame(raf.current); raf.current = 0
    fromPointer(e)
  }
  const onMove = (e) => { if (dragging.current) fromPointer(e) }
  const onUp = () => {
    if (!dragging.current) return
    dragging.current = false
    goTo(Math.round(vRef.current))
  }
  const onKey = (e) => {
    const k = e.key
    if (k === 'ArrowRight' || k === 'ArrowUp') { e.preventDefault(); goTo(level + 1) }
    else if (k === 'ArrowLeft' || k === 'ArrowDown') { e.preventDefault(); goTo(level - 1) }
    else if (k === 'Home') { e.preventDefault(); goTo(0) }
    else if (k === 'End') { e.preventDefault(); goTo(LAST) }
  }

  const cur = LEVELS[level]
  const factor = level > 0 ? sig2(cur.ly / LEVELS[level - 1].ly) : null

  return (
    <figure ref={figRef} className={styles.figure}>
      <figcaption className={styles.caption}>
        <span className="t-label">Scale of the Universe</span>
        <span className={styles.hint}>Drag the slider to zoom out</span>
      </figcaption>

      <div className={styles.stage}>
        {SCENES.map((Scene, i) => (
          <div key={i} ref={(el) => { layers.current[i] = el }} className={styles.scene} aria-hidden="true">
            {i === 3 ? <Scene diskRef={diskRef} sunRef={sunRef} /> : <Scene />}
          </div>
        ))}
        <div className={styles.level}>LEVEL {String(level + 1).padStart(2, '0')} / {String(LEVELS.length).padStart(2, '0')}</div>
        {factor && <div className={styles.factor}>×{factor} wider than the previous view</div>}
      </div>

      <div className={styles.info} aria-live="polite">
        <div key={cur.id} className={styles.infoInner}>
          <h3 className={styles.name}>{cur.name}</h3>
          <p className={styles.tagline}>{cur.tagline}</p>
          <dl className={styles.stats}>
            <div><dt>Diameter</dt><dd>{cur.size}</dd></div>
            <div><dt>{cur.light[0]}</dt><dd>{cur.light[1]}</dd></div>
            <div><dt>Contains</dt><dd>{cur.holds}</dd></div>
          </dl>
        </div>
      </div>

      <div className={styles.controls}>
        <button className={styles.step} onClick={() => goTo(level - 1)} disabled={level === 0} aria-label="Zoom in">−</button>

        <div className={styles.trackPad}>
          <div
            ref={trackRef}
            className={styles.track}
            onPointerDown={onDown}
            onPointerMove={onMove}
            onPointerUp={onUp}
            onPointerCancel={onUp}
          >
            <div className={styles.rail} />
            <div className={styles.fill} style={{ width: `${(v / LAST) * 100}%` }} />
            {LEVELS.map((l, i) => (
              <span key={l.id} className={`${styles.tick} ${i <= v + 0.01 ? styles.tickOn : ''}`} style={{ left: `${(i / LAST) * 100}%` }} />
            ))}
            <div
              className={styles.thumb}
              style={{ left: `${(v / LAST) * 100}%` }}
              role="slider"
              tabIndex={0}
              aria-label="Zoom level"
              aria-valuemin={0}
              aria-valuemax={LAST}
              aria-valuenow={level}
              aria-valuetext={cur.name}
              onKeyDown={onKey}
            />
          </div>
          <div className={styles.labels}>
            {LEVELS.map((l, i) => (
              <button
                key={l.id}
                className={`${styles.lab} ${i === level ? styles.labOn : ''}`}
                style={{ left: `${(i / LAST) * 100}%` }}
                onClick={() => goTo(i)}
              >
                {l.short}
              </button>
            ))}
          </div>
        </div>

        <button className={styles.step} onClick={() => goTo(level + 1)} disabled={level === LAST} aria-label="Zoom out">+</button>
      </div>
      <p className={styles.note}>Schematic illustrations, not to scale.</p>
    </figure>
  )
}

function mulberry32(seed) {
  return () => {
    seed |= 0; seed = (seed + 0x6d2b79f5) | 0
    let t = Math.imul(seed ^ (seed >>> 15), 1 | seed)
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296
  }
}

const R0 = 0.08, PITCH = Math.tan((15 * Math.PI) / 180), TILT = 0.62, GR = 168

const GALAXY = (() => {
  const rnd = mulberry32(42)
  const px = (x, y) => [x * GR, y * GR]
  const warm = [], blue = [], pink = []
  for (let i = 0; i < 260; i++) {
    const r = Math.abs(gauss(rnd)) * 0.075, a = rnd() * Math.PI * 2
    const x = Math.cos(a) * r * 1.7, y = Math.sin(a) * r
    const c = Math.cos(0.45), s = Math.sin(0.45)
    warm.push(px(x * c - y * s, x * s + y * c))
  }
  for (let i = 0; i < 1500; i++) {
    const k = Math.floor(rnd() * 4), rf = R0 + (1 - R0) * Math.pow(rnd(), 1.4)
    const a = armAngle(k, rf) + gauss(rnd) * (0.045 + 0.1 * rf)
      ; (rnd() < 0.08 ? pink : blue).push(px(rf * Math.cos(a), rf * Math.sin(a)))
  }
  return { warm, blue, pink }
})()

const COSMOS = (() => {
  const rnd = mulberry32(5)
  const inCircle = (rad) => { for (; ;) { const x = (rnd() * 2 - 1) * rad, y = (rnd() * 2 - 1) * rad; if (x * x + y * y <= rad * rad) return [x, y] } }
  const nodes = Array.from({ length: 48 }, () => inCircle(108))
  const edges = new Set()
  nodes.forEach((n, i) => {
    nodes.map((m, j) => [j, Math.hypot(n[0] - m[0], n[1] - m[1])])
      .filter(([j, d]) => j !== i && d < 58).sort((a, b) => a[1] - b[1]).slice(0, 2)
      .forEach(([j]) => edges.add(i < j ? `${i}-${j}` : `${j}-${i}`))
  })
  const lines = [], a = [], b = [], c = []
  const put = (x, y) => (rnd() < 0.5 ? a : rnd() < 0.7 ? b : c).push([CX + x, CY + y])
  edges.forEach((e) => {
    const [i, j] = e.split('-').map(Number), p = nodes[i], q = nodes[j]
    lines.push(`M${CX + p[0]} ${CY + p[1]}L${CX + q[0]} ${CY + q[1]}`)
    for (let k = 0; k < 9; k++) { const t = rnd(); put(p[0] + (q[0] - p[0]) * t + gauss(rnd) * 2.4, p[1] + (q[1] - p[1]) * t + gauss(rnd) * 2.4) }
  })
  nodes.forEach((n) => { for (let k = 0; k < 12; k++) put(n[0] + gauss(rnd) * 4, n[1] + gauss(rnd) * 4) })
  for (let k = 0; k < 120; k++) { const [x, y] = inCircle(116); (rnd() < 0.5 ? a : b).push([CX + x, CY + y]) }
  return { lines: lines.join(''), a, b, c }
})()

import { Canvas, useFrame, useLoader } from '@react-three/fiber'
import * as THREE from 'three'
import { sfx } from '../hooks/useSfx.js'
import styles from './Intro.module.css'

const HOLD_MS = 2500      // lama menahan tombol
const RELEASE_MS = 700    // lama progres kembali tenang saat dilepas
const MOON_KM = 384400
const EARTH_PX = 220      // diameter Bumi (px) di wrapper; dipakai untuk hitung skala transisi
const MOON_R = 3.6
const ISS_R = 2.75

function makeMoonTex() {
  const c = document.createElement('canvas')
  c.width = 128; c.height = 64
  const x = c.getContext('2d')
  x.fillStyle = '#9a9a9a'
  x.fillRect(0, 0, 128, 64)
  for (let i = 0; i < 40; i++) {
    x.fillStyle = `rgba(${Math.random() < 0.5 ? '60,60,60' : '210,210,210'},${Math.random() * 0.35})`
    x.beginPath()
    x.arc(Math.random() * 128, Math.random() * 64, Math.random() * 7 + 1, 0, 6.3)
    x.fill()
  }
  return new THREE.CanvasTexture(c)
}

function OrbitLine({ r, color, opacity }) {
  return (
    <mesh rotation={[-Math.PI / 2, 0, 0]}>
      <ringGeometry args={[r - 0.006, r + 0.006, 128]} />
      <meshBasicMaterial color={color} transparent opacity={opacity} side={THREE.DoubleSide} depthWrite={false} />
    </mesh>
  )
}

function Earth3D({ isWarping, warpStartTime, travelMs, holdRef }) {
  const root = useRef()
  const earth = useRef()
  const moon = useRef()
  const iss = useRef()
  const colorMap = useLoader(THREE.TextureLoader, '/images/earth.jpg')
  const moonTex = useMemo(makeMoonTex, [])

  useEffect(() => {
    if (colorMap) {
      colorMap.wrapS = THREE.RepeatWrapping
      colorMap.wrapT = THREE.RepeatWrapping
      colorMap.anisotropy = 16
    }
  }, [colorMap])

  useFrame((state, delta) => {
    const p = holdRef.current
    const t = isWarping ? Math.min((Date.now() - warpStartTime) / travelMs, 1) : 0
    const el = state.clock.elapsedTime

    // Getar halus saat ditahan, mereda begitu perjalanan mulai
    if (root.current) {
      const a = p * p * 0.06 * (1 - Math.min(t * 2, 1))
      root.current.position.set((Math.random() - 0.5) * a, (Math.random() - 0.5) * a, 0)
    }
    if (earth.current) earth.current.rotation.y += (0.09 + p * 1.2 + t * t * 1.5) * delta

    if (moon.current) {
      const a = el * 0.18
      moon.current.position.set(Math.cos(a) * MOON_R, 0, Math.sin(a) * MOON_R)
    }
    if (iss.current) {
      const a = el * 0.9
      iss.current.position.set(Math.cos(a) * ISS_R, 0, Math.sin(a) * ISS_R)
    }
  })

  return (
    <group ref={root}>
      <group ref={earth}>
        <mesh>
          <sphereGeometry args={[2.18, 64, 64]} />
          <meshBasicMaterial color="#020813" />
        </mesh>
        <mesh>
          <sphereGeometry args={[2.2, 64, 64]} />
          <meshStandardMaterial map={colorMap} roughness={0.8} metalness={0.2} alphaTest={0.5} />
        </mesh>
        <mesh>
          <sphereGeometry args={[2.25, 64, 64]} />
          <meshStandardMaterial color="#6ec5ff" transparent opacity={0.15} depthWrite={false} side={THREE.FrontSide} />
        </mesh>
      </group>

      {/* Bulan */}
      <group rotation={[1.1, 0, -0.2]}>
        <OrbitLine r={MOON_R} color="#8fb0ff" opacity={0.15} />
        <mesh ref={moon}>
          <sphereGeometry args={[0.32, 32, 32]} />
          <meshStandardMaterial map={moonTex} roughness={1} />
        </mesh>
      </group>

      {/* ISS (inklinasi ±51,6°, orbit cepat) */}
      <group rotation={[0.9, 0, 0.35]}>
        <OrbitLine r={ISS_R} color="#ffd27a" opacity={0.12} />
        <group ref={iss} scale={1.5}>
          <mesh><boxGeometry args={[0.24, 0.012, 0.012]} /><meshBasicMaterial color="#e6e9f2" /></mesh>
          <mesh><boxGeometry args={[0.05, 0.04, 0.04]} /><meshBasicMaterial color="#cfd4e2" /></mesh>
          <mesh position={[-0.1, 0, 0]}><boxGeometry args={[0.05, 0.004, 0.14]} /><meshBasicMaterial color="#d9a441" /></mesh>
          <mesh position={[0.1, 0, 0]}><boxGeometry args={[0.05, 0.004, 0.14]} /><meshBasicMaterial color="#d9a441" /></mesh>
        </group>
      </group>
    </group>
  )
}

const wib = () =>

function Clock() {
  const [t, setT] = useState(wib)
  useEffect(() => {

}

function Intro({ onNext, isWarping = false, warpStartTime = 0, travelMs = 2800 }) {
  const holdRef = useRef(0) // 0..1, dibaca oleh Three.js tanpa re-render
  const wrapRef = useRef(null)
  const btnRef = useRef(null)
  const distRef = useRef(null)
  const st = useRef({ holding: false, last: 0, raf: 0, stop: null })
  const nextRef = useRef(onNext)
  nextRef.current = onNext

  const tick = useCallback((now) => {
    const s = st.current
    const dt = Math.min(now - s.last, 50)
    s.last = now
    const p = Math.min(1, Math.max(0, holdRef.current + (s.holding ? dt / HOLD_MS : -dt / RELEASE_MS)))
    holdRef.current = p

    btnRef.current?.style.setProperty('--p', p)
    if (distRef.current) {
      distRef.current.textContent = `${((MOON_KM * (1 - p)) / 1e5).toFixed(3)} × 10⁵ km from home`
    }

    if (s.holding && p >= 1) {
      // Penuh: berangkat. Suara dibiarkan selesai sendiri.
      s.holding = false
      s.stop = null
      s.raf = 0
      nextRef.current()
      return
    }
    s.raf = p > 0 || s.holding ? requestAnimationFrame(tick) : 0
  }, [])

  const start = () => {
    const s = st.current
    if (s.holding || isWarping || holdRef.current >= 1) return
    s.holding = true
    s.last = performance.now()
    if (!s.stop) s.stop = sfx.open()
    if (!s.raf) s.raf = requestAnimationFrame(tick)
  }

  const end = () => {
    const s = st.current
    if (!s.holding) return
    s.holding = false
    s.stop?.()
    s.stop = null
  }

  useEffect(() => () => {
    cancelAnimationFrame(st.current.raf)
    st.current.stop?.()
  }, [])

  // Saat berangkat: Bumi bergeser & mengecil ke posisi Bumi di section Solar System
  useLayoutEffect(() => {
    const el = wrapRef.current
    if (!isWarping || !el) return
    const r = el.getBoundingClientRect()
    const vw = window.innerWidth
    const vh = window.innerHeight
    const mobile = vw < 768
    const camZ = mobile ? 34 : 26                                   // sama dgn CameraRig overview
    const unit = vh / (2 * camZ * Math.tan(THREE.MathUtils.degToRad(20))) // px per unit scene Solar
    const posX = mobile ? -2.1 : 2.5                                // posisi Earth (index 2) di grid
    const posY = mobile ? 1.6 : 3
    const tx = vw / 2 + posX * unit - (r.left + r.width / 2)
    const ty = vh / 2 - posY * unit - (r.top + r.height / 2)
    const s = ((68 / 65) * 2 * unit) / EARTH_PX                     // radius Earth = 68/65
    el.style.transitionDuration = `${travelMs}ms`
    el.style.transform = `translate(${tx}px, ${ty}px) scale(${s})`
  }, [isWarping, travelMs])

  return (
    <section id="intro" className={`cosmic-section ${styles.intro}`}>
      <div className="noise-overlay" />
      <div className={styles.nebulaBg} />
      <div
        className={`${styles.sunGlow} ${isWarping ? styles.sunGlowOn : ''}`}
        style={{ transitionDuration: `${travelMs}ms` }}
      />

      <div className={`${styles.hud} ${isWarping ? styles.hudHidden : ''}`}>
        <p className={styles.hudLabel}>WIB · GMT+7 · Indonesia Barat</p>
        <Clock />
        <div className={styles.hudIss}>
          <span className={styles.issDot} />
          <span className={styles.hudLabel}>ISS · ±408 km · orbit 92 min</span>
        </div>
      </div>

      <div ref={wrapRef} className={`${styles.earthWrap} ${isWarping ? styles.earthWrapWarp : ''}`}>
        <div className={styles.earthFloat}>
          <div className={styles.earthGlow} />
          <div className={styles.atmosphereRing} />
          <Canvas className={styles.earthCanvas} dpr={[1, 1.5]} camera={{ position: [0, 0, 15.8], fov: 45 }}>
            <ambientLight intensity={0.05} />
            <directionalLight position={[5, 4, 2]} intensity={2.5} color="#fff4e6" />
            <Suspense fallback={null}>
              <Earth3D isWarping={isWarping} warpStartTime={warpStartTime} travelMs={travelMs} holdRef={holdRef} />
            </Suspense>
          </Canvas>
        </div>
      </div>

      <div className={`${styles.content} ${isWarping ? styles.contentHidden : ''}`}>
        <p ref={distRef} className={`t-label ${styles.coords}`}>3.844 × 10⁵ km from home</p>
        <h1 className={styles.heading}>
          Welcome aboard,<br /><em>astronaut.</em>
        </h1>
        <p className={styles.sub}>
          Fasten your seatbelt. Our journey begins.
        </p>
        <p className={styles.local} lang="id">
          Kencangkan sabuk pengamanmu.
        </p>
        <button
          ref={btnRef}
          className={styles.beginBtn}
          onPointerDown={(e) => { e.currentTarget.setPointerCapture?.(e.pointerId); start() }}
          onPointerUp={end}
          onPointerCancel={end}

  )
}

// ── KOMPONEN EFEK MESIN KETIK (TERMINAL SCI-FI) ────────────────────────
function Typewriter({ text, speed = 25, delay = 0 }) {
  const [displayed, setDisplayed] = useState('')

  useEffect(() => {
    setDisplayed('')
    let timeout
    let interval

    timeout = setTimeout(() => {
      let i = 0
      interval = setInterval(() => {
        setDisplayed(text.substring(0, i + 1))
        i++
        if (i >= text.length) clearInterval(interval)
      }, speed)
    }, delay)

    return () => {
      clearTimeout(timeout)
      clearInterval(interval)
    }
  }, [text, speed, delay])

  return <span>{displayed}</span>
}

// ── KOMPONEN 2.5D PARALLAX CARD ────────────────────────────────────────
function TiltCard({ p, onClick }) {
  const [tilt, setTilt] = useState({ x: 0, y: 0 })
  const cardRef = useRef(null)

  const handleMouseMove = (e) => {
    if (!cardRef.current) return
    const rect = cardRef.current.getBoundingClientRect()
    const x = e.clientX - rect.left
    const y = e.clientY - rect.top
    const centerX = rect.width / 2
    const centerY = rect.height / 2
    const rotateX = ((y - centerY) / centerY) * -15
    const rotateY = ((x - centerX) / centerX) * 15
    setTilt({ x: rotateX, y: rotateY })
  }

  const handleMouseLeave = () => setTilt({ x: 0, y: 0 })

  return (
    <div className={styles.cardWrapper}>
      <div
        ref={cardRef}
        className={styles.cardHUD}
        style={{
          '--glow': p.glowColor,
          '--border': p.borderColor,
          transform: `perspective(1000px) rotateX(${tilt.x}deg) rotateY(${tilt.y}deg)`,
        }}
        onMouseMove={handleMouseMove}
        onMouseEnter={() => sfx.hover()}
        onMouseLeave={handleMouseLeave}
        onClick={() => onClick(p)}
        role="button"
        tabIndex={0}
        aria-label={`Open ${p.name}`}
        onKeyDown={(e) => {
          if (e.key === 'Enter' || e.key === ' ') {
            e.preventDefault()
            onClick(p)
          }
        }}
      >
        <div className={styles.icon3D} style={{ boxShadow: `0 0 30px ${p.glowColor}88` }}>
          {getIcon(p.id)}
        </div>
        <h3 className={styles.cardTitle3D}>{p.name}</h3>
        <p className={styles.cardDesc3D}>{p.description.substring(0, 75)}...</p>

        <div className={styles.cardFooter3D}>
          <span className={styles.discover}>Initialize Scan →</span>
        </div>
      </div>
    </div>
  )
}

// A. Tekstur Cincin Garis-garis (Saturnus/Uranus)
function createRingAlphaMap() {
  const canvas = document.createElement('canvas')
  canvas.width = 1024; canvas.height = 1024
  const ctx = canvas.getContext('2d')

  ctx.fillStyle = '#000000'
  ctx.fillRect(0, 0, 1024, 1024)

  for (let i = 0; i < 70; i++) {
    ctx.beginPath()
    const r = 290 + (i * 3.1)
    ctx.arc(512, 512, r, 0, Math.PI * 2)
    ctx.setLineDash([Math.random() * 200 + 30, Math.random() * 15])
    ctx.strokeStyle = `rgba(255, 255, 255, ${Math.random() * 0.9 + 0.1})`
    ctx.lineWidth = Math.random() * 3 + 0.5
    ctx.stroke()
  }
  return new THREE.CanvasTexture(canvas)
}

// B. Tekstur Glow Atmosfer (Untuk Sprite Belakang Planet)
function createGlowTexture() {
  const canvas = document.createElement('canvas')
  canvas.width = 256; canvas.height = 256
  const ctx = canvas.getContext('2d')
  const center = 128
  const gradient = ctx.createRadialGradient(center, center, 0, center, center, center)

  // Mereplika persis CSS: radial-gradient(circle, transparent 45%, rgba(...) 65%, rgba(...) 80%, transparent 100%)
  gradient.addColorStop(0.0, 'rgba(255, 255, 255, 0)')
  gradient.addColorStop(0.45, 'rgba(255, 255, 255, 0)')
  gradient.addColorStop(0.65, 'rgba(255, 255, 255, 0.25)')
  gradient.addColorStop(0.80, 'rgba(255, 255, 255, 0.5)')
  gradient.addColorStop(1.0, 'rgba(255, 255, 255, 0)')

  ctx.fillStyle = gradient
  ctx.fillRect(0, 0, 256, 256)
  return new THREE.CanvasTexture(canvas)
}

// C. Tekstur Glow Matahari (gradien penuh, hangat)
function createSunTexture() {
  const canvas = document.createElement('canvas')
  canvas.width = 512; canvas.height = 512

  const c = 256

// ── 3. Kamera Sinematik & Interaktivitas 360° (OrbitControls) ──────────
function CameraRig({ focusedPos }) {
  const { size, camera } = useThree()
  const isMobile = size.width < 768
  const overviewZ = isMobile ? 34 : 26

  const controlsRef = useRef()
  const [transitioning, setTransitioning] = useState(false)
  const prevFocusedRef = useRef(null)

  useEffect(() => {
    if (focusedPos !== prevFocusedRef.current) {
      setTransitioning(true)
      const timer = setTimeout(() => {
        setTransitioning(false)
      }, 700)
      prevFocusedRef.current = focusedPos
      return () => clearTimeout(timer)
    }
  }, [focusedPos])

  useFrame(() => {
    // Mode Overview: Lerp kamera kembali ke posisi tengah
    if (!focusedPos) {
      camera.position.lerp(new THREE.Vector3(0, 0, overviewZ), 0.05)
      const currentLookAt = new THREE.Vector3(0, 0, -1)
        .applyQuaternion(camera.quaternion)
        .add(camera.position)
      currentLookAt.lerp(new THREE.Vector3(0, 0, 0), 0.05)
      camera.lookAt(currentLookAt)
      return
    }

    // Mode Transisi Zoom-in ke Planet
    if (transitioning) {
      const targetPos = new THREE.Vector3(focusedPos.x, focusedPos.y, 8)
      camera.position.lerp(targetPos, 0.08)

      const lookTarget = new THREE.Vector3(focusedPos.x, focusedPos.y, 0)
      const currentLookAt = new THREE.Vector3(0, 0, -1)
        .applyQuaternion(camera.quaternion)
        .add(camera.position)
      currentLookAt.lerp(lookTarget, 0.08)
      camera.lookAt(currentLookAt)

      if (controlsRef.current) {
        controlsRef.current.target.set(focusedPos.x, focusedPos.y, 0)
      }
    }
  })

  return focusedPos ? (
    <OrbitControls
      ref={controlsRef}
      enabled={!transitioning}
      target={[focusedPos.x, focusedPos.y, 0]}
      enablePan={false}
      enableZoom={true}
      minDistance={3.5}
      maxDistance={12}
      rotateSpeed={0.8}
      enableDamping={true}
      dampingFactor={0.06}
    />
  ) : null
}

            rotation={[-Math.PI / 2, 0, 0]}
            castShadow
            receiveShadow
            onClick={(e) => {
              e.stopPropagation()
              onClick({ x: posX, y: posY })
            }}
            onPointerOver={() => document.body.style.cursor = 'pointer'}
            onPointerOut={() => document.body.style.cursor = 'auto'}
          >
            <ringGeometry args={[radius * 1.35, radius * 2.3, 64]} />
            <meshStandardMaterial
              color={data.color || '#ead6b8'}
              side={THREE.DoubleSide}
              transparent={true}
              alphaMap={sharedRingTexture}
              opacity={0.85}
              roughness={0.6}
            />
          </mesh>
        )}
      </group>

      <Html position={[0, -radius - 1.2, 0]} center style={{ pointerEvents: 'none' }}>
        <div className={styles.planetLabel} style={{ opacity: isFocused ? 0 : 1 }}>
          {data.name}
        </div>
      </Html>
    </group>
  )
}

// ── 5. Komponen Utama Solar System ─────────────────────────────────────
function SolarSystem({ onFact, onNext }) {
  const [focusedPos, setFocusedPos] = useState(null)
  const [activePlanet, setActivePlanet] = useState(null)

  const handlePlanetClick = (planet, pos) => {
    sfx.chime(PLANETS.indexOf(planet))
    setFocusedPos(pos)
    setActivePlanet(planet.id)
    onFact({
      title: planet.name,
      subtitle: `${planet.type} · ${planet.moons} moon${planet.moons !== 1 ? 's' : ''}`,
      text: planet.fact,
      extra: `Orbital period: ${planet.orbit}`,
    })
  }

  const handleBackgroundClick = () => {
    if (focusedPos) sfx.close()
    setFocusedPos(null)
    setActivePlanet(null)
    onFact(null)
  }

  // Esc → back to overview
  useEffect(() => {
    if (!focusedPos) return
    const onKey = (e) => { if (e.key === 'Escape') handleBackgroundClick() }
    window.addEventListener('keydown', onKey)
    return () => window.removeEventListener('keydown', onKey)
  }, [focusedPos])

  return (
    <section id="solar" className={`cosmic-section ${styles.solar}`}>
      <div className="noise-overlay" />
      <div className={styles.bgGradient} />

      <div className={`${styles.header} ${focusedPos !== null ? styles.headerHidden : ''}`}>
        <p className="t-label">Stage I — 1 AU from Earth</p>
        <h2 className={styles.title}>The Solar System</h2>
        <p className={styles.hint}>Click any planet to discover its secrets</p>
      </div>

      <button
        className={`${styles.backBtn} ${focusedPos !== null ? styles.backBtnVisible : ''}`}
        onClick={handleBackgroundClick}
      >
        ← Overview
      </button>

      <div className={`${styles.orbitHint} ${focusedPos !== null ? styles.orbitHintVisible : ''}`}>
        <span>✦ Drag to rotate 360° · Scroll to zoom</span>
      </div>

      <div className={styles.canvasWrap}>
        <Canvas
          shadows
          dpr={[1, 1.5]}
          camera={{ position: [0, 0, 26], fov: 40 }}
          onPointerMissed={handleBackgroundClick}
        >
          <ambientLight intensity={0.05} color="#ffffff" />

          <directionalLight
            position={[-20, 6, 4]}
            intensity={3.5}
            color="#fff8f0"
            castShadow
            shadow-bias={-0.0005}
          >
            <orthographicCamera attach="shadow-camera" args={[-30, 30, 15, -15]} />
          </directionalLight>

          <Sun />

          <Suspense fallback={null}>
            {PLANETS.map((planet, i) => (
              <Planet3D
                key={planet.id}
                data={planet}
                index={i}
                isFocused={activePlanet === planet.id}
                onClick={(pos) => handlePlanetClick(planet, pos)}
              />
            ))}
          </Suspense>

          <CameraRig focusedPos={focusedPos} />
        </Canvas>
      </div>

      <div className={`${styles.nextNav} ${focusedPos !== null ? styles.nextNavHidden : ''}`}>
        <button className={styles.nextBtn} onClick={onNext}>
          Explore Cosmic Phenomena <span className={styles.arrow}>↓</span>
        </button>
      </div>
    </section>
  )
}

export default memo(SolarSystem)
