import { useState, useRef, useCallback, useEffect } from 'react'

const SECTION_MUSIC = {
  intro: '/audio/intro.mp3',
  solar: '/audio/solar.mp3',
  phenomena: '/audio/phenomena.mp3',
  deepspace: '/audio/deepspace.mp3',
  ending: '/audio/ending.mp3',
  wishing: '/audio/ending.mp3', // lanjut dari Ending tanpa putus (src sama: tidak crossfade)
  gallery: '/audio/intro.mp3', // hangat & ramah, seperti kembali ke titik berangkat
  default: '/audio/intro.mp3',
}

const FADE_DURATION = 1500
const PRELOAD_DELAY = 1200 // beri waktu render + tekstur Bumi dulu, baru musik Intro disiapkan

// iOS: volume elemen audio DIABAIKAN, jadi fade dan volume lewat GainNode.
// Rantai per track: elemen -> MediaElementSource -> trackGain (fade) -> analyser -> master (volume) -> speaker.

const iosAudioSession = () => {
  // Safari 16.4+: jalur 'playback' tetap bunyi walau tombol senyap menyala
  try { if (navigator.audioSession) navigator.audioSession.type = 'playback' } catch (_) { }
}

// Hemat data seluler: 'none' = tidak mengunduh apa pun sebelum diketuk
function preloadMode() {
  const c = navigator.connection || navigator.mozConnection || navigator.webkitConnection
  if (!c) return 'auto' // iOS Safari tidak punya API ini (dan memang mengabaikan preload)
  if (c.saveData) return 'none'
  if (/(^|-)2g$/.test(c.effectiveType || '')) return 'none'
  if (c.effectiveType === '3g' || c.type === 'cellular') return 'metadata'
  return 'auto'
}

// resume() di Safari lama bisa mengembalikan undefined
const softResume = (ctx) => {
  try {
    const r = ctx.resume()
    r?.catch?.(() => { })
  } catch (_) { }
}

export function useAmbientAudio() {
  const [playing, setPlaying] = useState(false)
  const [volume, setVolumeState] = useState(0.5)
  const [audioProgress, setAudioProgress] = useState({ currentTime: 0, duration: 0, progress: 0 })
  // true = pengunjung ingin suara tetapi AudioContext belum running (iOS senyap/interupsi, atau play gagal)
  const [blocked, setBlocked] = useState(false)

  const audioRef = useRef(null)
  const preparedRef = useRef(null) // elemen musik Intro yang sudah disiapkan sebelum gerbang diketuk
  const currentSrc = useRef(null)
  const volumeRef = useRef(0.5)

  const isPlaying = useRef(false)
  const manualPause = useRef(false) // pengguna sengaja pause / memilih tanpa suara
  const wantSound = useRef(false)   // pengguna pernah memilih suara (gerbang atau tombol play)
  const started = useRef(false)     // gerbang sudah diketuk

  const ctxRef = useRef(null)
  const analyserRef = useRef(null)
  const gainNodeRef = useRef(null)

  const syncBlocked = useCallback(() => {
    const ctx = ctxRef.current
    const want = wantSound.current && !manualPause.current
    setBlocked(!!(want && ctx && ctx.state !== 'running'))
  }, [])

  const getCtx = useCallback(() => {
    if (!ctxRef.current) {
      const AC = window.AudioContext || window.webkitAudioContext
      if (!AC) return null
      const ctx = new AC()
      const analyser = ctx.createAnalyser()
      analyser.fftSize = 256
      analyser.smoothingTimeConstant = 0.8
      const master = ctx.createGain()
      master.gain.value = volumeRef.current
      analyser.connect(master)
      master.connect(ctx.destination)
      ctx.onstatechange = syncBlocked
      ctxRef.current = ctx
      analyserRef.current = analyser
      gainNodeRef.current = master
    }
    return ctxRef.current
  }, [syncBlocked])

  // Sambungkan elemen ke grafik audio SEBELUM play(), dengan gain 0 (tidak ada letupan di iOS).
  // Idempotent. Mengembalikan true bila WebAudio dipakai.
  const connectAudio = useCallback((audio) => {
    if (audioRef.current !== audio) return false
    try {
      const ctx = getCtx()
      if (!ctx) {
        try { audio.volume = 0 } catch (_) { }
        return false
      }
      if (ctx.state !== 'running') softResume(ctx)
      if (!audio._node) {
        audio._node = ctx.createMediaElementSource(audio)
        audio._gain = ctx.createGain()
        audio._gain.gain.value = 0
        audio._node.connect(audio._gain)
        audio._gain.connect(analyserRef.current)
      }
      return true
    } catch (_) {
      try { audio.volume = 0 } catch (__) { }
      return false
    }
  }, [getCtx])

  const getAudioData = useCallback((targetArray) => {
    if (analyserRef.current && targetArray) {
      analyserRef.current.getByteFrequencyData(targetArray)
      return true
    }
    return false
  }, [])

  // Fade level track (0..1). Satu timer selesai per elemen: fade baru membatalkan yang lama.
  // Cadangan tanpa WebAudio: volume elemen diubah langsung (tanpa fade halus).
  const rampTo = useCallback((audio, level, ms = FADE_DURATION, onDone) => {
    if (!audio) return
    clearTimeout(audio._fadeT)
    const ctx = ctxRef.current
    if (audio._gain && ctx) {
      const p = audio._gain.gain
      const t = ctx.currentTime
      p.cancelScheduledValues(t)
      p.setValueAtTime(p.value, t)
      p.linearRampToValueAtTime(level, t + ms / 1000)
    } else {
      try { audio.volume = level * volumeRef.current } catch (_) { }
    }
    if (onDone) audio._fadeT = setTimeout(onDone, ms + 40)
  }, [])

  // Pensiunkan elemen audio: fade-out, lalu bebaskan (tanpa src='' agar tidak memicu error)
  const retire = useCallback((audio) => {
    if (!audio || audio._retired) return
    audio._retired = true
    rampTo(audio, 0, FADE_DURATION, () => {
      audio.pause()
      audio.onerror = null
      audio.ontimeupdate = null
      audio.removeAttribute('src')
      try { audio.load() } catch (_) { }
      try { audio._node?.disconnect() } catch (_) { }
      try { audio._gain?.disconnect() } catch (_) { }
    })
  }, [rampTo])

  const createTrack = useCallback((src, preload = 'auto') => {
    const audio = new Audio()
    audio.crossOrigin = 'anonymous'
    audio.loop = true
    audio.preload = preload
    audio._key = src

    audio.onerror = () => {
      if (audioRef.current === audio) {
        console.warn(`[useAudio] Gagal load: ${src}. Pastikan file ada di public/audio/`)
      }
    }

    // Update progres paling sering 1x per detik (UI hanya menampilkan detik)
    let lastSec = -1
    audio.ontimeupdate = () => {
      if (audioRef.current !== audio) return
      const sec = Math.floor(audio.currentTime)
      if (sec === lastSec) return
      lastSec = sec
      setAudioProgress({
        currentTime: audio.currentTime,
        duration: audio.duration || 0,
        progress: audio.duration ? (audio.currentTime / audio.duration) : 0,
      })
    }

    audio.src = src
    return audio
  }, [])

  // Mengembalikan Promise (menolak jika browser memblokir atau play dibatalkan).
  // Untuk iOS, pemanggilan pertama harus sinkron di dalam handler gestur.
  const playSrc = useCallback((src) => {
    const cur = audioRef.current

    // Track yang sama sudah ada: lanjutkan (membatalkan fade-out yang belum selesai)
    if (cur && cur._key === src && !cur._retired) {
      connectAudio(cur)
      const p = cur.paused ? cur.play() : Promise.resolve()
      return p.then(() => {
        if (audioRef.current !== cur) return
        rampTo(cur, 1)
      })
    }

    retire(cur)
    setAudioProgress({ currentTime: 0, duration: 0, progress: 0 })

    let audio
    if (preparedRef.current && preparedRef.current._key === src) {
      audio = preparedRef.current
      preparedRef.current = null
    } else {
      audio = createTrack(src)
    }

    audioRef.current = audio
    currentSrc.current = src

    connectAudio(audio) // gain 0 sebelum play
    return audio.play().then(() => {
      if (audioRef.current !== audio) return // sudah digantikan selagi menunggu
      rampTo(audio, 1)
    })
  }, [connectAudio, createTrack, retire, rampTo])

  // Dipanggil SEKALI dari ketukan gerbang. withSound=false: tanpa musik (tombol play tetap bisa dipakai).
  const startAudio = useCallback((withSound) => {
    if (started.current) return
    started.current = true
    if (!withSound) {
      manualPause.current = true
      return
    }
    wantSound.current = true
    manualPause.current = false
    iosAudioSession()
    playSrc(currentSrc.current || SECTION_MUSIC.intro)
      .then(() => {
        isPlaying.current = true
        setPlaying(true)
        setTimeout(syncBlocked, 500)
      })
      .catch(() => setBlocked(true))
  }, [playSrc, syncBlocked])

  // Elemen yang dijeda sistem (iOS saat interupsi) dinyalakan lagi bila pengunjung masih ingin musik
  const kick = useCallback(() => {
    const a = audioRef.current
    if (a && a.paused && isPlaying.current && !manualPause.current) a.play().catch(() => { })
  }, [])

  // Dipanggil dari petunjuk "Sound is paused" (ketukan pengunjung = gestur sah)
  const resumeAudio = useCallback(() => {
    if (!wantSound.current) return
    manualPause.current = false
    iosAudioSession()
    const ctx = ctxRef.current
    if (ctx) softResume(ctx)
    if (!isPlaying.current) {
      playSrc(currentSrc.current || SECTION_MUSIC.intro)
        .then(() => {
          isPlaying.current = true
          setPlaying(true)
        })
        .catch(() => { })
    } else {
      kick()
    }
    setTimeout(syncBlocked, 500)
  }, [playSrc, kick, syncBlocked])

  const toggleMusic = useCallback(() => {
    if (!isPlaying.current) {
      manualPause.current = false
      wantSound.current = true
      started.current = true
      iosAudioSession()
      playSrc(currentSrc.current || SECTION_MUSIC.intro)
        .then(() => {
          isPlaying.current = true
          setPlaying(true)
          setTimeout(syncBlocked, 500)
        })
        .catch(() => { })
    } else {
      manualPause.current = true
      const audio = audioRef.current
      if (audio) rampTo(audio, 0, FADE_DURATION, () => audio.pause())
      isPlaying.current = false
      setPlaying(false)
      setBlocked(false)
    }
  }, [playSrc, rampTo, syncBlocked])

  const changeSection = useCallback((sectionId) => {
    const src = SECTION_MUSIC[sectionId] ?? SECTION_MUSIC.default
    if (src === currentSrc.current) return

    if (isPlaying.current) {
      playSrc(src).catch(() => { })
    } else {
      currentSrc.current = src
    }
  }, [playSrc])

  const setVolume = useCallback((val) => {
    const v = Math.max(0, Math.min(1, val))
    setVolumeState(v)
    volumeRef.current = v
    const ctx = ctxRef.current
    if (gainNodeRef.current && ctx) {
      gainNodeRef.current.gain.setTargetAtTime(v, ctx.currentTime, 0.02)
    } else {
      const a = audioRef.current
      if (a && !a.paused) { try { a.volume = v } catch (_) { } }
    }
  }, [])

  // Kembali ke tab / aplikasi: coba hidupkan lagi AudioContext (iOS menandainya 'interrupted')
  useEffect(() => {
    const onVis = () => {
      if (document.visibilityState !== 'visible') return
      const ctx = ctxRef.current
      if (ctx && wantSound.current && !manualPause.current) {
        if (ctx.state !== 'running') softResume(ctx)
        kick()
        setTimeout(syncBlocked, 500)
      }
    }
    document.addEventListener('visibilitychange', onVis)
    return () => document.removeEventListener('visibilitychange', onVis)
  }, [kick, syncBlocked])

  // Siapkan musik Intro lebih awal, tanpa memblokir render dan hemat data seluler
  useEffect(() => {
    let cancelled = false
    let idleId = 0
    const run = () => {
      if (cancelled || preparedRef.current || audioRef.current) return
      preparedRef.current = createTrack(SECTION_MUSIC.intro, preloadMode())
    }
    const timer = setTimeout(() => {
      if (window.requestIdleCallback) idleId = window.requestIdleCallback(run, { timeout: 2000 })
      else run()
    }, PRELOAD_DELAY)
    return () => {
      cancelled = true
      clearTimeout(timer)
      if (idleId && window.cancelIdleCallback) window.cancelIdleCallback(idleId)
      const p = preparedRef.current
      preparedRef.current = null
      if (p) {
        p.onerror = null
        p.ontimeupdate = null
        p.removeAttribute('src')
        try { p.load() } catch (_) { }
      }
    }
  }, [createTrack])

  useEffect(() => {
    return () => {
      started.current = false // StrictMode: mount ulang mulai dari keadaan bersih
      const a = audioRef.current
      if (a) {
        clearTimeout(a._fadeT)
        a.pause()
        a.onerror = null
        a.ontimeupdate = null
        a.removeAttribute('src')
        try { a.load() } catch (_) { }
      }
      audioRef.current = null
      currentSrc.current = null
      isPlaying.current = false
      try { if (ctxRef.current) ctxRef.current.onstatechange = null } catch (_) { }
      try { ctxRef.current?.close() } catch (_) { }
      ctxRef.current = null
      analyserRef.current = null
      gainNodeRef.current = null
    }
  }, [])

  return {
    playing,
    toggleMusic,
    volume,
    setVolume,
    changeSection,
    getAudioData,
    audioProgress,
    startAudio,
    blocked,
    resumeAudio,
  }
}

// ============================================================
// COSMIC JOURNEY: Synthesized sound effects (no audio files)
// Usage: import { sfx } from '../hooks/useSfx.js'  ->  sfx.chime(2)
// Must be called from a user gesture (click/keypress) the first time.
// Fase 10: sfx.unlock() dipanggil dari ketukan gerbang masuk.
// ============================================================

let ctx = null
let master = null
let noiseBuf = null
let muted = false
try { muted = localStorage.getItem('cj-sfx-muted') === '1' } catch (_) { }

const SCALE = [523.25, 587.33, 659.25, 783.99, 880.0, 987.77, 1174.66, 1318.51] // C major pentatonic-ish

function getCtx() {
  if (muted) return null
  if (!ctx) {
    const AC = window.AudioContext || window.webkitAudioContext
    if (!AC) return null
    ctx = new AC()
    master = ctx.createGain()
    master.gain.value = 0.35
    master.connect(ctx.destination)
  }
  if (ctx.state === 'suspended') ctx.resume()
  return ctx
}

function getNoise(c) {
  if (!noiseBuf) {
    noiseBuf = c.createBuffer(1, c.sampleRate * 2, c.sampleRate)
    const d = noiseBuf.getChannelData(0)
    for (let i = 0; i < d.length; i++) d[i] = Math.random() * 2 - 1
  }
  return noiseBuf
}

// Mengembalikan gain node agar suara bisa dihentikan lebih awal (lihat sfx.open)
function tone({ freq, to, type = 'sine', dur = 0.4, vol = 0.2, delay = 0, attack = 0.01 }) {
  const c = getCtx()
  if (!c) return null
  const t = c.currentTime + delay
  const o = c.createOscillator()
  const g = c.createGain()
  o.type = type
  o.frequency.setValueAtTime(freq, t)
  if (to) o.frequency.exponentialRampToValueAtTime(to, t + dur)
  g.gain.setValueAtTime(0.0001, t)
  g.gain.exponentialRampToValueAtTime(vol, t + Math.min(attack, dur * 0.5))
  g.gain.exponentialRampToValueAtTime(0.0001, t + dur)
  o.connect(g)
  g.connect(master)
  o.start(t)
  o.stop(t + dur + 0.05)
  return g
}

function noise({ dur = 1, vol = 0.2, from = 300, to = 3000, q = 1, delay = 0, swell = 0.5 }) {
  const c = getCtx()
  if (!c) return null
  const t = c.currentTime + delay
  const src = c.createBufferSource()
  src.buffer = getNoise(c)
  src.loop = true
  const f = c.createBiquadFilter()
  f.type = 'bandpass'
  f.Q.value = q
  f.frequency.setValueAtTime(from, t)
  f.frequency.exponentialRampToValueAtTime(to, t + dur)
  const g = c.createGain()
  g.gain.setValueAtTime(0.0001, t)
  g.gain.exponentialRampToValueAtTime(vol, t + dur * swell)
  g.gain.exponentialRampToValueAtTime(0.0001, t + dur)
  src.connect(f)
  f.connect(g)
  g.connect(master)
  src.start(t)
  src.stop(t + dur + 0.05)
  return g
}

export const sfx = {
  /** Buka AudioContext sfx dari gestur pengguna (iOS): resume + buffer senyap 1 sampel. */
  unlock() {
    try { if (navigator.audioSession) navigator.audioSession.type = 'playback' } catch (_) { }
    const c = getCtx()
    if (!c) return
    try {
      const b = c.createBuffer(1, 1, 22050)
      const s = c.createBufferSource()
      s.buffer = b
      s.connect(c.destination)
      s.start(0)
    } catch (_) { }
  },

  /** Tiny tick for hovering cards */
  hover() { tone({ freq: 1400, to: 1800, dur: 0.06, vol: 0.035 }) },

  /** Generic soft click */
  click() { tone({ freq: 660, to: 440, dur: 0.09, vol: 0.12, type: 'triangle' }) },

  /** Bell-like chime. Pass an index (e.g. planet number) to vary the pitch. */
  chime(i = 0) {
    const f = SCALE[i % SCALE.length]
    tone({ freq: f, dur: 1.5, vol: 0.16 })
    tone({ freq: f * 2, dur: 0.9, vol: 0.05, delay: 0.01 })
    tone({ freq: f * 1.5, dur: 1.1, vol: 0.04, delay: 0.02 })
  },

  /** Deep swelling drone. Mengembalikan fungsi stop() yang memudarkan suara dengan cepat. */
  open() {
    const nodes = [
      tone({ freq: 55, to: 82, dur: 2.4, vol: 0.3, attack: 0.7 }),
      tone({ freq: 110, to: 165, dur: 2.4, vol: 0.1, attack: 0.9, type: 'triangle' }),
      noise({ dur: 2.4, vol: 0.07, from: 80, to: 420, q: 0.7, swell: 0.6 }),
    ].filter(Boolean)
    return () => {
      if (!ctx) return
      const t = ctx.currentTime
      nodes.forEach((g) => {
        try {
          const v = Math.max(g.gain.value, 0.0001)
          g.gain.cancelScheduledValues(t)
          g.gain.setValueAtTime(v, t)
          g.gain.exponentialRampToValueAtTime(0.0001, t + 0.25)
        } catch (_) { }
      })
    }
  },

  /** Descending blip: closing / going back */
  close() { tone({ freq: 520, to: 180, dur: 0.35, vol: 0.14 }) },

  /** Hyperspace whoosh: warp jumps (tidak lagi dipakai intro) */
  warp(dur = 3) {
    noise({ dur, vol: 0.32, from: 200, to: 5200, q: 0.8, swell: 0.85 })
    tone({ freq: 80, to: 900, dur, vol: 0.08, type: 'sawtooth', attack: dur * 0.6 })
  },

  /** Ascending sparkle arpeggio: message revealed */
  reveal() {
    ;[523.25, 659.25, 783.99, 1046.5, 1318.51].forEach((f, i) =>
      tone({ freq: f, dur: 1.6, vol: 0.13, delay: i * 0.12 })
    )
  },

  setMuted(m) {
    muted = !!m
    try { localStorage.setItem('cj-sfx-muted', muted ? '1' : '0') } catch (_) { }
  },
  isMuted() { return muted },
}

import { useEffect, useMemo, useRef } from 'react'
import styles from './EntryGate.module.css'

const STAR_COUNT = 36
const EXIT_MS = 700
const EXIT_MS_REDUCED = 150

const prefersReduced = () =>
    typeof window !== 'undefined' &&
    window.matchMedia('(prefers-reduced-motion: reduce)').matches

// Bintang gerbang dibuat sekali per mount (murni CSS, tanpa canvas)
function makeStars() {
    return Array.from({ length: STAR_COUNT }, () => ({
        x: Math.random() * 100,
        y: Math.random() * 100,
        size: 1 + Math.random() * 2,
        dur: 2.4 + Math.random() * 3.2,
        delay: -Math.random() * 5,
        o: 0.45 + Math.random() * 0.55,
    }))
}

// onEnter(true)  = masuk dengan suara (HARUS dipanggil sinkron di handler ketukan)
// onEnter(false) = masuk tanpa suara
// leaving        = App sudah menerima ketukan, gerbang memudar lalu memanggil onExited
export default function EntryGate({ onEnter, leaving = false, onExited }) {
    const stars = useMemo(makeStars, [])
    const rootRef = useRef(null)
    const boardRef = useRef(null)
    const exitedRef = useRef(onExited)
    exitedRef.current = onExited

    useEffect(() => {
        boardRef.current?.focus({ preventScroll: true })
    }, [])

    useEffect(() => {
        if (!leaving) return
        const t = setTimeout(
            () => exitedRef.current?.(),
            prefersReduced() ? EXIT_MS_REDUCED : EXIT_MS
        )
        return () => clearTimeout(t)
    }, [leaving])

    // Fokus dikunci di dalam gerbang (dua tombol saja)
    const onKeyDown = (e) => {
        if (e.key !== 'Tab') return
        const items = rootRef.current?.querySelectorAll('button')
        if (!items || items.length === 0) return
        const first = items[0]
        const last = items[items.length - 1]
        if (e.shiftKey && document.activeElement === first) {
            e.preventDefault()
            last.focus()
        } else if (!e.shiftKey && document.activeElement === last) {
            e.preventDefault()
            first.focus()
        }
    }

    const choose = (withSound) => {
        if (leaving) return
        onEnter(withSound)
    }

    return (
        <div
            ref={rootRef}
            className={`${styles.gate} ${leaving ? styles.leaving : ''}`}
            role="dialog"
            aria-modal="true"
            aria-labelledby="gate-title"
            aria-describedby="gate-note"
            onKeyDown={onKeyDown}
        >
            <div className={styles.sky} aria-hidden="true">
                {stars.map((s, i) => (
                    <span
                        key={i}
                        className={styles.star}
                        style={{
                            left: `${s.x}%`,
                            top: `${s.y}%`,
                            width: `${s.size}px`,
                            height: `${s.size}px`,
                            '--d': `${s.dur}s`,
                            '--delay': `${s.delay}s`,
                            '--o': s.o,
                        }}
                    />
                ))}
            </div>

            <div className={styles.content}>
                <div className={styles.core} aria-hidden="true">
                    <span className={styles.coreRing} />
                    <span className={styles.coreDot} />
                </div>

                <h2 id="gate-title" className={styles.kicker}>Cosmic Journey</h2>

                <button
                    ref={boardRef}
                    type="button"
                    className={styles.boardBtn}
                    onClick={() => choose(true)}
                >
                    Board the ship
                </button>

                <p id="gate-note" className={styles.note}>Best with sound on</p>

                <button
                    type="button"
                    className={styles.quiet}
                    onClick={() => choose(false)}

function getAudioMetrics(data) {
  if (!data || data.length === 0) return { bass: 0, avg: 0, mid: 0 }
  const bassEnd = Math.floor(data.length * 0.06)
  const midEnd = Math.floor(data.length * 0.35)
  let bass = 0, mid = 0, total = 0
  for (let i = 0; i < data.length; i++) {
    const v = data[i] / 255
    total += v
    if (i < bassEnd) bass += v
    else if (i < midEnd) mid += v
  }
  return { bass: bass / bassEnd, mid: mid / (midEnd - bassEnd), avg: total / data.length }
}

function fmt(sec) {
  if (!sec || isNaN(sec)) return '0:00'
  const m = Math.floor(sec / 60)
  const s = Math.floor(sec % 60).toString().padStart(2, '0')
  return `${m}:${s}`
}

function ArtCanvas({ theme, color, still }) {
  const canvasRef = useRef(null)
  const rafRef = useRef(null)
  const t0 = useRef(Date.now())
  const particles = useRef([])

  useEffect(() => {
    const canvas = canvasRef.current
    if (!canvas) return
    const ctx = canvas.getContext('2d')
    const W = canvas.width, H = canvas.height

    const initParticles = () => {
      particles.current = []
      if (theme === 'stars' || theme === 'nebula' || theme === 'clouds') {
        for (let i = 0; i < 40; i++) {
          particles.current.push({
            x: Math.random() * W, y: Math.random() * H,
            r: Math.random() * 1.5 + 0.3,
            speed: Math.random() * 0.3 + 0.05,
            phase: Math.random() * Math.PI * 2,
            opacity: Math.random() * 0.8 + 0.2,
          })
        }
      }
    }
    initParticles()

    const draw = () => {
      const t = (Date.now() - t0.current) / 1000
      const c1 = color[0], c2 = color[1]

      ctx.clearRect(0, 0, W, H)

      if (theme === 'nebula') {
        for (let i = 0; i < 3; i++) {
          const cx = W * (0.3 + i * 0.2) + Math.sin(t * 0.3 + i) * 8
          const cy = H * (0.4 + Math.sin(t * 0.2 + i * 1.2) * 0.2)
          const r = 22 + Math.sin(t + i) * 4
          const g = ctx.createRadialGradient(cx, cy, 0, cx, cy, r)
          g.addColorStop(0, c1 + (i === 1 ? 'cc' : '88'))
          g.addColorStop(0.5, c2 + '44')
          g.addColorStop(1, 'transparent')
          ctx.fillStyle = g
          ctx.beginPath(); ctx.arc(cx, cy, r, 0, Math.PI * 2); ctx.fill()
        }
        particles.current.forEach(p => {
          p.phase += 0.02
          const a = 0.4 + Math.abs(Math.sin(p.phase)) * 0.6
          ctx.beginPath(); ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2)
          ctx.fillStyle = `rgba(200,220,255,${a})`; ctx.fill()
        })
      }

      if (theme === 'solar') {
        const cx = W * 0.5, cy = H * 0.5
        const pulseR = 14
        for (let ring = 3; ring >= 0; ring--) {
          const g = ctx.createRadialGradient(cx, cy, 0, cx, cy, pulseR + ring * 8)
          g.addColorStop(0, ring === 0 ? '#ffffff' : c1 + 'ee')
          g.addColorStop(0.4, c1 + (ring === 0 ? 'cc' : '66'))
          g.addColorStop(1, 'transparent')
          ctx.fillStyle = g
          ctx.beginPath(); ctx.arc(cx, cy, pulseR + ring * 8, 0, Math.PI * 2); ctx.fill()
        }
        for (let i = 0; i < 6; i++) {
          const angle = t * 0.4 + i * Math.PI / 3
          const len = 10 + Math.sin(t + i) * 4
          const x2 = cx + Math.cos(angle) * (pulseR + len)
          const y2 = cy + Math.sin(angle) * (pulseR + len)
          ctx.strokeStyle = c1 + '88'
          ctx.lineWidth = 1.5
          ctx.beginPath(); ctx.moveTo(cx, cy); ctx.lineTo(x2, y2); ctx.stroke()
        }
      }

      if (theme === 'blackhole') {
        const cx = W * 0.5, cy = H * 0.5
        const coreR = 10
        for (let i = 5; i >= 0; i--) {
          const rx = coreR + 6 + i * 4
          const ry = (coreR + 6 + i * 4) * 0.28
          const alpha = (0.6 - i * 0.08) * 0.7
          ctx.save()
          ctx.translate(cx, cy)
          ctx.rotate(t * 0.5)
          ctx.strokeStyle = i < 2 ? c1 + 'ff' : c2 + Math.floor(alpha * 255).toString(16).padStart(2, '0')
          ctx.lineWidth = i < 2 ? 2 : 1
          ctx.beginPath(); ctx.ellipse(0, 0, rx, ry, 0, 0, Math.PI * 2)
          ctx.stroke(); ctx.restore()
        }
        const cg = ctx.createRadialGradient(cx, cy, 0, cx, cy, coreR)
        cg.addColorStop(0, '#000000'); cg.addColorStop(0.6, '#110008'); cg.addColorStop(1, c1 + '44')
        ctx.fillStyle = cg
        ctx.beginPath(); ctx.arc(cx, cy, coreR, 0, Math.PI * 2); ctx.fill()
      }

      if (theme === 'stars') {
        particles.current.forEach(p => {
          p.x -= p.speed * 0.8
          if (p.x < 0) { p.x = W; p.y = Math.random() * H }
          p.phase += 0.015
          const a = 0.3 + Math.abs(Math.sin(p.phase)) * 0.7
          ctx.beginPath(); ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2)
          ctx.fillStyle = `rgba(180,210,255,${a})`; ctx.fill()
        })
        const g = ctx.createRadialGradient(W * 0.3, H * 0.5, 0, W * 0.3, H * 0.5, W * 0.6)
        g.addColorStop(0, c1 + '22'); g.addColorStop(1, 'transparent')
        ctx.fillStyle = g; ctx.fillRect(0, 0, W, H)
      }

      if (theme === 'clouds') {
        const cloudDefs = [
          { x: 0.2, y: 0.35, r: 18, speed: 0.20 },
          { x: 0.55, y: 0.55, r: 22, speed: 0.15 },
          { x: 0.8, y: 0.3, r: 14, speed: 0.25 },
        ]
        cloudDefs.forEach((cl, i) => {
          const cx = (cl.x + Math.sin(t * cl.speed + i) * 0.15) * W
          const cy = (cl.y + Math.cos(t * cl.speed * 0.15 + i) * 0.16) * H
          const r = cl.r
          for (let b = 3; b >= 0; b--) {
            const g = ctx.createRadialGradient(cx, cy, 0, cx, cy, r + b * 5)
            g.addColorStop(0, `rgba(200,210,255,${0.18 - b * 0.04})`)
            g.addColorStop(1, 'transparent')
            ctx.fillStyle = g
            ctx.beginPath(); ctx.arc(cx, cy, r + b * 5, 0, Math.PI * 2); ctx.fill()
          }
        })
        particles.current.forEach(p => {
          p.phase += 0.008
          const a = 0.2 + Math.abs(Math.sin(p.phase)) * 0.5
          ctx.beginPath(); ctx.arc(p.x, p.y, p.r * 0.8, 0, Math.PI * 2)
          ctx.fillStyle = `rgba(220,225,255,${a})`; ctx.fill()
        })
      }

      if (!still) rafRef.current = requestAnimationFrame(draw)
    }

    draw()
    return () => cancelAnimationFrame(rafRef.current)
  }, [theme, color, still])

  return <canvas ref={canvasRef} width={56} height={56} className={styles.artCanvas} />
}

// ── Bar visualizer ─────────────────────────────────────────
// still: saat idle gambar garis datar sekali (tanpa gelombang bergerak)
function BarVisualizer({ getAudioData, playing, color, onMetrics, still }) {
  const canvasRef = useRef(null)
  const rafRef = useRef(null)
  const t0 = useRef(Date.now())
  const freqRef = useRef(new Uint8Array(128))

  useEffect(() => {
    const canvas = canvasRef.current
    if (!canvas) return
    const ctx = canvas.getContext('2d')
    const W = canvas.width, H = canvas.height
    const BINS = 28
    const freqData = freqRef.current

    const draw = () => {
      ctx.clearRect(0, 0, W, H)
      const t = (Date.now() - t0.current) / 1000

      if (!getAudioData || !playing) {
        ctx.strokeStyle = color[0] + '55'
        ctx.lineWidth = 1.5
        ctx.beginPath()
        for (let x = 0; x < W; x++) {
          const y = still
            ? H / 2
            : H / 2 + Math.sin(x * 0.18 + t * 1.2) * 3.5 + Math.sin(x * 0.08 + t * 0.7) * 2
          x === 0 ? ctx.moveTo(x, y) : ctx.lineTo(x, y)
        }
        ctx.stroke()
        if (onMetrics) onMetrics(null)
        if (!still) rafRef.current = requestAnimationFrame(draw)
        return
      }

      getAudioData(freqData)
      if (onMetrics) onMetrics(freqData)

      const step = Math.floor(freqData.length / BINS)
      const barW = (W / BINS) - 1

      const grad = ctx.createLinearGradient(0, H, 0, 0)
      grad.addColorStop(0, color[0] + 'dd')
      grad.addColorStop(0.6, color[1] + 'aa')
      grad.addColorStop(1, '#ffffff44')

      for (let i = 0; i < BINS; i++) {
        let sum = 0
        for (let j = 0; j < step; j++) sum += (freqData[i * step + j] || 0)
        const v = (sum / step) / 255
        const barH = Math.max(2, v * H * 1.15)
        const x = i * (barW + 1)
        ctx.fillStyle = grad
        ctx.beginPath()
        if (ctx.roundRect) ctx.roundRect(x, H - barH, barW, barH, 2)
        else ctx.rect(x, H - barH, barW, barH)
        ctx.fill()
      }
      rafRef.current = requestAnimationFrame(draw)
    }

    draw()
    return () => cancelAnimationFrame(rafRef.current)
  }, [getAudioData, playing, color, onMetrics, still])

  return <canvas ref={canvasRef} width={196} height={32} className={styles.barCanvas} />
}

          </span>
        </div>
      </div>

      {/* Bar visualizer */}
      <div className={styles.vizWrap} aria-hidden="true">
        <BarVisualizer
          getAudioData={getAudioData}
          playing={playing}
          color={track.color}
          onMetrics={onMetricsCallback}
          still={still}
        />
      </div>

      {/* Progress bar — real duration */}
      <div className={styles.progressArea}>
        <span className={styles.timeLabel}>{fmt(curTime)}</span>
        <div className={styles.progressTrack}>
          <div
            className={styles.progressFill}
            style={{ width: `${prog * 100}%`, background: `linear-gradient(90deg, ${track.color[0]}, ${track.color[1]})` }}
          />
        </div>
        <span className={styles.timeLabel}>{fmt(dur)}</span>
      </div>

      {/* Controls row */}
      <div className={styles.controlsRow}>
        <button className={styles.playBtn} onClick={onToggle}
          style={{ '--btn-col': track.color[0] }}
          aria-label={playing ? 'Pause' : 'Play'}
        >
          {playIcon(18)}
        </button>

        <div className={styles.volRow}>
          {volIcon}
          <input
            type="range" min="0" max="1" step="0.02"
            value={volume}
            onChange={e => onVolumeChange(parseFloat(e.target.value))}

            aria-label="Volume"
          />
          <span className={styles.volNum}>{Math.round(volume * 100)}</span>
        </div>

        <ComfortButton comfort={comfort} onToggle={onToggleComfort} />
      </div>
    </div>
  )
}

      aria-valuemin={0}
      aria-valuemax={100}
    />
  )
}

export default function App() {
  const [fact, setFact] = useState(null)
  const [message, setMessage] = useState(null)

  // Gerbang masuk: 'show' > 'leaving' (memudar) > 'done'. Tidak pernah kembali ke 'show'
  // dalam satu kunjungan, jadi HomeWarp / "Return home" tidak memunculkannya lagi.
  const [gate, setGate] = useState('show')
  const mainRef = useRef(null)

  const [activeSection, setActiveSection] = useState('intro')
  const [renderedSection, setRenderedSection] = useState('intro')
  const [fadeState, setFadeState] = useState('visible')
  const navTimer = useRef(0) // timer pergantian section (dibatalkan jika ada perpindahan baru)

  // Perjalanan intro > solar: 'idle' > 'warp' > 'idle'
  const [warp, setWarp] = useState({ phase: 'idle', startTime: 0 })
  const warpingRef = useRef(false)

  const [fadePhase, setFadePhase] = useState('idle')

  const [visitor, setVisitor] = useState(INITIAL_VISITOR)
  const updateVisitor = useCallback((patch) => setVisitor((v) => ({ ...v, ...patch })), [])
  // true = Ending langsung membuka pesan konstelasi (dari Secret Ending: "Stay with my constellation")
  const [resumeMessage, setResumeMessage] = useState(false)

  const [homeWarping, setHomeWarping] = useState(false)
  const homeTimers = useRef([])
  // true = HomeWarp sudah memberi tahu bahwa Bumi mendekat: bintang kembali terang (depth 0)
  const [homeBright, setHomeBright] = useState(false)
  const brightenStars = useCallback(() => setHomeBright(true), [])

  // Mode Comfort (Fase 12): tombolnya ada di MusicPlayer. State hidup di sini, TANPA localStorage
  // (kembali mati saat halaman dimuat ulang; tetap menyala setelah "Return home").
  // Efek: starfield satu tingkat lebih redup dan lebih tenang (calm), body.comfort-mode
  // (kontras teks naik, animasi CSS dihentikan), FactPopup tampil lebih lama, teks langsung tampil.
  const [comfort, setComfort] = useState(false)
  const toggleComfort = useCallback(() => setComfort((c) => !c), [])

  const {
    playing,
    toggleMusic,
    volume,
    setVolume,
    changeSection,
    getAudioData,
    audioProgress,
    startAudio,
    blocked: audioBlocked,
    resumeAudio,
  } = useAmbientAudio()

  // Ketukan gerbang. SEMUA yang butuh gestur (AudioContext sfx, musik) harus dipanggil
  // sinkron di sini, sebelum await / setTimeout apa pun, supaya iOS mengizinkannya.
  const handleGateEnter = useCallback((withSound) => {
    sfx.setMuted(!withSound)
    if (withSound) sfx.unlock()
    startAudio(withSound)
    setGate((g) => (g === 'show' ? 'leaving' : g))
  }, [startAudio])

  const handleGateExited = useCallback(() => setGate('done'), [])

  const handleNavigate = useCallback((id) => {
    setFact(null)
    setMessage(null)
    if (id === activeSection) return
    if (id !== 'ending') setResumeMessage(false)
    setActiveSection(id)
    setFadeState('hidden')
    changeSection(MUSIC_ALIAS[id] ?? id)

    // Perpindahan cepat beruntun: hanya tujuan terakhir yang dirender (tanpa kilatan section perantara)
    clearTimeout(navTimer.current)
    navTimer.current = setTimeout(() => {
      setRenderedSection(id)
      setFadeState('visible')
    }, 400)
  }, [activeSection, changeSection])

  // NavDots: diabaikan selama transisi / HomeWarp (tombol tetap bisa dijangkau keyboard walau tersembunyi)
  const handleNavDot = useCallback((id) => {
    if (warpingRef.current) return
    handleNavigate(id)
  }, [handleNavigate])

  // Dipanggil Intro saat tombol tahan penuh: Intro menganimasikan Bumi, App mengganti section
  const handleWarpNavigate = useCallback((id) => {
    if (warpingRef.current || id === activeSection) return
    warpingRef.current = true
    setFact(null)
    setMessage(null)
    setWarp({ phase: 'warp', startTime: Date.now() })

    setTimeout(() => {
      handleNavigate(id)
      setTimeout(() => {
        setWarp({ phase: 'idle', startTime: 0 })
        warpingRef.current = false
      }, ARRIVE_MS)
    }, TRAVEL_MS)
  }, [activeSection, handleNavigate])

    if (warpingRef.current || id === activeSection) return
    warpingRef.current = true
    setFact(null)
    setMessage(null)
    setFadePhase('dark')

    setTimeout(() => {
      handleNavigate(id)
      setFadePhase('light')

      setTimeout(() => {
        setFadePhase('idle')
        warpingRef.current = false
      }, FADE_IN)
    }, FADE_OUT)
  }, [activeSection, handleNavigate])

  // Tidak menyentuh state gerbang, jadi gerbang tidak muncul lagi.
  const handleReturnHome = useCallback(async () => {
    if (warpingRef.current) return
    warpingRef.current = true
    setFact(null)
    setMessage(null)
    setHomeBright(false)
    try {
      const { HOME_MS } = await import('./components/HomeWarp.jsx')
      setHomeWarping(true)
      const t1 = setTimeout(() => {
        handleNavigate('intro')
        const t2 = setTimeout(() => {
          setHomeWarping(false)
          setHomeBright(false)        // Intro sudah depth 0, jadi tidak ada lompatan visual
          setVisitor(INITIAL_VISITOR) // perjalanan baru = pengunjung baru
          resetSessionCallsign()      // callsign anonim baru untuk pengunjung berikutnya
          warpingRef.current = false
        }, HOME_END_PAD)
        homeTimers.current.push(t2)
      }, HOME_MS)
      homeTimers.current.push(t1)
    } catch (_) {
      warpingRef.current = false
    }
  }, [handleNavigate])

  // Secret Ending > "Stay with my constellation"
  const handleBackToConstellation = useCallback(() => {
    setResumeMessage(true)
    handleFadeNavigate('ending')
  }, [handleFadeNavigate])

  // Wishing: wish berhasil terkirim dan animasi bintang naik selesai > Secret Ending
  const handleWishSubmitted = useCallback(() => {
    handleFadeNavigate('secret')
  }, [handleFadeNavigate])

  useEffect(() => () => {
    homeTimers.current.forEach(clearTimeout)
    clearTimeout(navTimer.current)
  }, [])

  // Selama gerbang tampil, konten di belakangnya tidak bisa difokus keyboard / pembaca layar
  useEffect(() => {
    const el = mainRef.current
    if (!el) return
    if (gate === 'show') el.setAttribute('inert', '')
    else el.removeAttribute('inert')
  }, [gate])

  const transitioning = warp.phase !== 'idle' || fadePhase !== 'idle' || homeWarping
  const hideUi = transitioning || gate !== 'done'
  useEffect(() => {
    document.body.classList.toggle('hide-global-ui', hideUi)
    return () => document.body.classList.remove('hide-global-ui')
  }, [hideUi])

  // Mode Comfort: kontras teks naik + gerak berkurang lewat body.comfort-mode (index.css).
  // FactPopup, MessageBox, dan Phenomena membaca class ini langsung.
  useEffect(() => {
    document.body.classList.toggle('comfort-mode', comfort)
    return () => document.body.classList.remove('comfort-mode')
  }, [comfort])

  // Preload Solar System supaya perpindahan dari intro tidak tersendat.
  // Tekstur planet diunduh bertahap di latar (tidak memblokir render Intro).
  useEffect(() => {
    let alive = true
    let stop = () => { }
    const t = setTimeout(() => {
      import('./sections/SolarSystem.jsx')
      import('./utils/planetTexture.js')
        .then((m) => { if (alive) stop = m.preloadPlanetTextures() })
        .catch(() => { })
    }, 1500)
    return () => {
      alive = false
      clearTimeout(t)
      stop()
    }
  }, [])

  // Preload section akhir + HomeWarp begitu pengunjung mencapai Deep Space,
  // supaya layar tidak kosong saat pindah ke Ending / Wishing / Secret / Gallery / pulang.
  const preloaded = useRef(false)
  useEffect(() => {
    if (preloaded.current || !PRELOAD_FROM.has(activeSection)) return
    const t = setTimeout(() => {
      preloaded.current = true
      const quiet = () => { }
      import('./sections/Ending.jsx').catch(quiet)
      import('./sections/Wishing.jsx').catch(quiet)
      import('./sections/SecretEnding.jsx').catch(quiet)
      import('./sections/Gallery.jsx').catch(quiet)
      import('./components/HomeWarp.jsx').catch(quiet)
    }, 1200)
    return () => clearTimeout(t)
  }, [activeSection])

  const showFact = useCallback((d) => setFact(d), [])
  const closeFact = useCallback(() => setFact(null), [])

  const showMessage = useCallback(() => {
    setMessage(UNIVERSE_MESSAGES[Math.floor(Math.random() * UNIVERSE_MESSAGES.length)])
  }, [])
  const closeMessage = useCallback(() => setMessage(null), [])

  // Wishing dan Secret Ending tidak punya titik sendiri di NavDots/ProgressBar, jadi ikut 'ending'
  const navSection = activeSection === 'wishing' || activeSection === 'secret' ? 'ending' : activeSection
  const currentIdx = SECTION_IDS.indexOf(navSection)
  const progress = Math.max(0, currentIdx / (SECTION_IDS.length - 1))

  // Depth starfield mengikuti activeSection (berubah saat navigasi dimulai; StarField
  // meng-easing selama ~2 dtk sehingga tumpang tindih dengan fade section, tanpa lompatan).
  const baseDepth = SECTION_DEPTH[activeSection] ?? 0
  const starDepth = homeBright
    ? 0
    : Math.min(1, baseDepth + (comfort ? COMFORT_DEPTH_STEP : 0))

  return (
    <>
      <StarField count={500} mouseParallax depth={starDepth} calm={comfort} />
      <ProgressBar progress={progress} />
      <NavDots active={navSection} onNavigate={handleNavDot} />

      <MusicPlayer
        playing={playing}
        onToggle={toggleMusic}
        volume={volume}
        onVolumeChange={setVolume}
        activeSection={MUSIC_ALIAS[activeSection] ?? activeSection}
        getAudioData={getAudioData}
        audioProgress={audioProgress}
        comfort={comfort}
        onToggleComfort={toggleComfort}
      />

      <div
        className={`fade-overlay ${fadePhase === 'dark' ? 'fade-dark' : ''} ${fadePhase === 'light' ? 'fade-light' : ''}`}
        style={{ '--fade-out': `${FADE_OUT}ms`, '--fade-in': `${FADE_IN}ms` }}
        aria-hidden="true"
      />

      <main ref={mainRef} className={`section-wrapper ${fadeState}`}>
        <Suspense fallback={null}>
          {renderedSection === 'intro' && (
            <Intro
              onNext={() => handleWarpNavigate('solar')}
              isWarping={warp.phase !== 'idle'}
              warpStartTime={warp.startTime}
              travelMs={TRAVEL_MS}
            />
          )}
          {renderedSection === 'solar' && (
            <SolarSystem onFact={showFact} onNext={() => handleFadeNavigate('phenomena')} />
          )}
          {renderedSection === 'phenomena' && (
            <Phenomena onFact={showFact} onNext={() => handleFadeNavigate('deepspace')} />
          )}
          {renderedSection === 'deepspace' && (
            <DeepSpace onFact={showFact} onNext={() => handleFadeNavigate('ending')} />
          )}
          {renderedSection === 'ending' && (
            <Ending
              visitor={visitor}
              onReveal={updateVisitor}
              onNext={() => handleFadeNavigate('wishing')}
              onBackToChoices={() => handleFadeNavigate('secret')}
              resumeMessage={resumeMessage}
            />
          )}
          {renderedSection === 'wishing' && (
            <Wishing
              visitor={visitor}
              onVisitorChange={updateVisitor}
              onGallery={() => handleFadeNavigate('gallery')}
              onSubmitted={handleWishSubmitted}
              onBackToConstellation={handleBackToConstellation}
              onReturnHome={handleReturnHome}
            />
          )}
          {renderedSection === 'secret' && (
            <SecretEnding
              visitor={visitor}
              onVisitorChange={updateVisitor}
              onStay={handleBackToConstellation}
              onGallery={() => handleFadeNavigate('gallery')}
              onReturnHome={handleReturnHome}
            />
          )}
          {renderedSection === 'gallery' && (
            <Gallery
              visitor={visitor}
              onBackToConstellation={handleBackToConstellation}
              onReturnHome={handleReturnHome}
            />
          )}
        </Suspense>
      </main>

      {homeWarping && (
        <Suspense fallback={null}>
          <HomeWarp
            active
            name={visitor.name}
            points={visitor.constellationPoints}
            onBrighten={brightenStars}
          />
        </Suspense>
      )}

      <FactPopup fact={fact} onClose={closeFact} />
      <MessageBox message={message} onClose={closeMessage} />

      {audioBlocked && gate === 'done' && (
        <div style={HINT_WRAP} role="status">
          <button type="button" style={HINT_BTN} onClick={resumeAudio}>
            Sound is paused. Tap to turn it on.
            <br />
            On iPhone, also check the silent switch.
          </button>
        </div>
      )}

      {gate !== 'done' && (
        <Suspense fallback={<div style={GATE_FALLBACK} aria-hidden="true" />}>
          <EntryGate
            onEnter={handleGateEnter}
            leaving={gate === 'leaving'}
            onExited={handleGateExited}
          />
        </Suspense>
      )}
    </>
  )
}

import { useEffect, useRef, memo } from 'react'

// Waktu-konstan easing depth (detik). 3 × TAU ≈ 1,6 dtk → ~95% tercapai, ~2 dtk hampir tuntas.
const DEPTH_TAU = 0.55
const clamp01 = (v) => Math.min(1, Math.max(0, v))

// Mode tenang (Comfort), nilai pada calm penuh (di-easing 0..1, tanpa lompatan):
const CALM_SLOW = 0.3     // kedip dan drift tinggal 30%
const CALM_DIM = 0.45     // kecerahan canvas dikurangi 45% lagi (di atas redup dari depth)
const CALM_KEEP = 0.2     // ambang rank turun: bintang yang digambar makin sedikit
const CALM_HALO = 0.5     // halo bintang terang mengecil

// depth 0..1: makin tinggi → canvas lebih redup, bintang yang digambar berkurang
// (sisa kebanyakan yang redup), halo mengecil, kedip melambat. Dibaca lewat ref di loop gambar.
// calm (boolean): mode Comfort, juga dibaca lewat ref (tidak memicu inisialisasi ulang).
function StarField({ count = 500, mouseParallax = true, depth = 0, calm = false }) {
  const canvasRef = useRef(null)
  const starsRef = useRef([])
  const shootersRef = useRef([])
  const mouseRef = useRef({ x: 0, y: 0 })
  const animRef = useRef(null)

  // Depth dan calm: target (dari prop) dan nilai berjalan (di-easing di loop, tanpa state per frame)
  const depthTargetRef = useRef(clamp01(depth))
  const depthCurRef = useRef(clamp01(depth))
  const calmTargetRef = useRef(calm ? 1 : 0)
  const calmCurRef = useRef(calm ? 1 : 0)
  const kickRef = useRef(null) // membangunkan loop (penting saat reduced motion: loop berhenti bila sudah stabil)

  useEffect(() => {
    const canvas = canvasRef.current
    if (!canvas) return
    const ctx = canvas.getContext('2d')
    const reduce = window.matchMedia('(prefers-reduced-motion: reduce)').matches

    let lastW = 0
    let lastH = 0
    let lastT = 0

    // Layar sempit (HP) cukup dengan ±55% bintang: tampilan sama, beban jauh lebih ringan
    const targetCount = () => (window.innerWidth < 768 ? Math.round(count * 0.55) : count)

    // ── Init stars ────────────────────────────────────────
    const initStars = () => {
      starsRef.current = Array.from({ length: targetCount() }, () => {
        const tier = Math.random()
        // Radius disesuaikan agar bintang terlihat jelas
        const r = tier > 0.92 ? Math.random() * 1.8 + 1.4   // Bintang terang
          : tier > 0.75 ? Math.random() * 1.0 + 0.8   // Bintang sedang
            : Math.random() * 0.6 + 0.4                 // Bintang redup
        return {
          x: Math.random() * canvas.width,
          y: Math.random() * canvas.height,
          r,
          // rank 0..1 tetap per bintang: bintang besar cenderung berrank tinggi (dibuang duluan saat depth naik)
          rank: 0.5 * ((r - 0.4) / 2.8) + 0.5 * Math.random(),
          alpha: Math.random(),
          speed: reduce ? 0 : Math.random() * 0.005 + 0.001,
          phase: Math.random() * Math.PI * 2,
          drift: reduce ? 0 : (Math.random() - 0.5) * 0.02,
          layer: Math.floor(Math.random() * 3),
          color: (() => {
            const c = Math.random()
            if (c > 0.90) return { r: 255, g: 220, b: 120 }
            if (c > 0.80) return { r: 200, g: 180, b: 255 }
            if (c > 0.70) return { r: 140, g: 200, b: 255 }
            if (c > 0.60) return { r: 255, g: 180, b: 180 }
            return { r: 210, g: 225, b: 255 }
          })(),
        }
      })
    }

    // ── Resize ────────────────────────────────────────────
    // Address bar HP naik-turun hanya mengubah TINGGI: bintang tidak dibuat ulang,
    // posisi Y cukup diskalakan. Bintang baru dibuat hanya jika LEBAR berubah.
    const resize = () => {
      const w = window.innerWidth
      const h = window.innerHeight
      const widthChanged = w !== lastW
      const oldH = lastH
      lastW = w
      lastH = h
      canvas.width = w
      canvas.height = h

      if (!starsRef.current.length || widthChanged) {
        initStars()
      } else if (oldH && h !== oldH) {
        const k = h / oldH
        starsRef.current.forEach((s) => { s.y *= k })
      }
      if (reduce) draw() // loop tidak berjalan terus di mode reduced: gambar ulang manual
    }

    // ── Shooting stars ────────────────────────────────────
    const spawnShooter = () => ({
      x: Math.random() * canvas.width * 0.7,
      y: Math.random() * canvas.height * 0.4,
      len: Math.random() * 180 + 80,
      speed: Math.random() * 14 + 8,
      angle: Math.PI / 4 + (Math.random() - 0.5) * 0.3,
      opacity: 1,
      width: Math.random() * 1.5 + 0.5,
      trail: [],
    })

    // Spawn shooting star setiap 3–7 detik (dimatikan saat reduced motion; dilewati saat Comfort)
    const shooterInterval = reduce ? 0 : setInterval(() => {
      if (!document.hidden && calmTargetRef.current === 0 && shootersRef.current.length < 3) {
        shootersRef.current.push(spawnShooter())
      }
    }, 3000 + Math.random() * 4000)

    // ── Draw ──────────────────────────────────────────────
    function draw() {
      ctx.clearRect(0, 0, canvas.width, canvas.height)

      // c: 0 = normal, 1 = Comfort penuh (di-easing, jadi transisinya mulus)
      const c = calmCurRef.current
      const calmK = 1 - (1 - CALM_SLOW) * c
      const mx = mouseRef.current.x * (1 - c)
      const my = mouseRef.current.y * (1 - c)

      // Parameter turunan depth + calm (dihitung sekali per frame)
      const d = depthCurRef.current
      const dim = (1 - 0.5 * d) * (1 - CALM_DIM * c)   // kecerahan canvas: 1 → 0,5 (depth), lalu × 0,55 (Comfort)
      const keep = 1.1 - 0.65 * d - CALM_KEEP * c      // ambang rank: makin kecil, makin sedikit bintang
      const slow = 1 - 0.65 * d                        // kecepatan kedip: 100% → 35%
      const haloK = (2.2 - 0.9 * d) * (1 - CALM_HALO * c * 0.5) // radius halo
      const haloA = 0.22 * (1 - 0.5 * d) * (1 - CALM_HALO * c)

      ctx.globalAlpha = dim

      // Draw stars
      starsRef.current.forEach((s) => {
        s.phase += s.speed * slow * calmK
        s.x += s.drift * calmK
        if (s.x < -2) s.x = canvas.width + 2
        if (s.x > canvas.width + 2) s.x = -2

        // Memudar halus di sekitar ambang (tidak "pop" saat bintang hilang)
        const vis = clamp01((keep - s.rank) / 0.1)
        if (vis <= 0.01) return

        const parallaxStr = [0.003, 0.009, 0.020][s.layer]
        const px = s.x + mx * parallaxStr
        const py = s.y + my * parallaxStr

        // Twinkle: bintang besar lebih dramatis (reduced: kecerahan tetap)
        const twinkle = reduce ? 0.8 : 0.3 + Math.abs(Math.sin(s.phase)) * 0.7
        const { r, g, b } = s.color
        const alpha = twinkle * (s.r > 1.2 ? 1 : 0.85) * vis

        ctx.beginPath()
        ctx.arc(px, py, s.r, 0, Math.PI * 2)
        ctx.fillStyle = `rgba(${r},${g},${b},${alpha})`
        ctx.fill()

        // Luminous halo untuk bintang terang (jauh lebih efisien dari shadowBlur CPU filter)
        if (s.r > 1.0) {
          ctx.beginPath()
          ctx.arc(px, py, s.r * haloK, 0, Math.PI * 2)
          ctx.fillStyle = `rgba(${r},${g},${b},${alpha * haloA})`
          ctx.fill()
        }
      })

      // Draw shooting stars (ikut dim lewat globalAlpha). Yang sudah terlanjur muncul dibiarkan selesai.
      shootersRef.current = shootersRef.current.filter((s) => {
        s.x += Math.cos(s.angle) * s.speed
        s.y += Math.sin(s.angle) * s.speed
        s.opacity -= 0.012

        if (s.opacity <= 0 || s.x > canvas.width + 50 || s.y > canvas.height + 50) return false

        const tailX = s.x - Math.cos(s.angle) * s.len
        const tailY = s.y - Math.sin(s.angle) * s.len

        const grad = ctx.createLinearGradient(tailX, tailY, s.x, s.y)
        grad.addColorStop(0, `rgba(255,255,255,0)`)
        grad.addColorStop(0.7, `rgba(200,220,255,${s.opacity * 0.4})`)
        grad.addColorStop(1, `rgba(255,255,255,${s.opacity})`)

        ctx.beginPath()
        ctx.moveTo(tailX, tailY)
        ctx.lineTo(s.x, s.y)
        ctx.strokeStyle = grad
        ctx.lineWidth = s.width
        ctx.shadowBlur = 8
        ctx.shadowColor = `rgba(180,210,255,${s.opacity})`
        ctx.stroke()
        ctx.shadowBlur = 0

        return true
      })

      ctx.globalAlpha = 1
    }

    // ── Loop ──────────────────────────────────────────────
    // Normal: berjalan terus (dijeda saat tab tersembunyi).
    // Reduced motion: hanya berjalan selama depth / calm masih berpindah, lalu berhenti sendiri.
    const ease = (cur, target, dt) => {
      const diff = target - cur
      return Math.abs(diff) < 0.0005 ? target : cur + diff * (1 - Math.exp(-dt / DEPTH_TAU))
    }

    const loop = (now) => {
      const dt = lastT ? Math.min((now - lastT) / 1000, 0.1) : 0.016
      lastT = now

      depthCurRef.current = ease(depthCurRef.current, depthTargetRef.current, dt)
      calmCurRef.current = ease(calmCurRef.current, calmTargetRef.current, dt)

      draw()

      if (
        reduce &&
        depthCurRef.current === depthTargetRef.current &&
        calmCurRef.current === calmTargetRef.current
      ) {
        animRef.current = null
        lastT = 0
        return
      }
      animRef.current = requestAnimationFrame(loop)
    }
    const startLoop = () => {
      if (animRef.current || document.hidden) return
      lastT = 0
      animRef.current = requestAnimationFrame(loop)
    }
    const stopLoop = () => {
      cancelAnimationFrame(animRef.current)
      animRef.current = null
    }
    const onVisibility = () => (document.hidden ? stopLoop() : startLoop())

    const onMouseMove = (e) => {
      mouseRef.current = {
        x: e.clientX - window.innerWidth / 2,
        y: e.clientY - window.innerHeight / 2,
      }
    }

    kickRef.current = startLoop

    resize()
    window.addEventListener('resize', resize)
    document.addEventListener('visibilitychange', onVisibility)
    // Parallax mouse dimatikan saat reduced motion
    if (mouseParallax && !reduce) window.addEventListener('mousemove', onMouseMove, { passive: true })
    startLoop()

    return () => {
      kickRef.current = null
      window.removeEventListener('resize', resize)
      window.removeEventListener('mousemove', onMouseMove)
      document.removeEventListener('visibilitychange', onVisibility)
      stopLoop()
      if (shooterInterval) clearInterval(shooterInterval)
      shootersRef.current = []
    }
  }, [count, mouseParallax])

  // Perubahan depth hanya memperbarui target (ref) dan membangunkan loop; efek utama tidak dijalankan ulang
  useEffect(() => {
    depthTargetRef.current = clamp01(depth)
    kickRef.current?.()
  }, [depth])

  // Perubahan calm: target di-easing di loop (mulus), loop dibangunkan bila sedang berhenti
  useEffect(() => {
    calmTargetRef.current = calm ? 1 : 0
    kickRef.current?.()
  }, [calm])

  return (
    <canvas
      ref={canvasRef}
      aria-hidden="true"
      style={{
        position: 'fixed',
        top: 0, left: 0,
        width: '100%', height: '100%',
        pointerEvents: 'none',
        zIndex: 0,
      }}
    />
  )
}
