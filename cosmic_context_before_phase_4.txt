  sender: '',              // pengirim (hanya mode "For Someone Else")
  constellationPoints: [],
  finalMessage: '',
  coords: '',
  wishSubmitted: false,
  mySubmission: null,      // data kiriman wish pengunjung ini (untuk highlight "Your wish")
  skipped: false,          // true = Wishing di-skip langsung ke Gallery (tidak membuka Secret Ending)

const HOME_END_PAD = 800 // jeda setelah pindah ke intro sebelum overlay HomeWarp dilepas

// Perjalanan Bumi (intro → solar): Bumi menjauh & mengecil ke posisinya di tata surya
const TRAVEL_MS = 2800 // dari tombol penuh sampai section berganti
const ARRIVE_MS = 1000 // jeda sampai UI global muncul lagi

// Transisi gelap-terang untuk perpindahan section selain intro → solar
const FADE_OUT = 700
const FADE_IN = 800

export default function App() {
  const [fact, setFact] = useState(null)
  const [message, setMessage] = useState(null)

  const [activeSection, setActiveSection] = useState('intro')
  const [renderedSection, setRenderedSection] = useState('intro')
  const [fadeState, setFadeState] = useState('visible')

  // Perjalanan intro → solar: 'idle' → 'warp' → 'idle'
  const [warp, setWarp] = useState({ phase: 'idle', startTime: 0 })
  const warpingRef = useRef(false)

  const [fadePhase, setFadePhase] = useState('idle')

  const [visitor, setVisitor] = useState(INITIAL_VISITOR)
  const updateVisitor = useCallback((patch) => setVisitor((v) => ({ ...v, ...patch })), [])
  // true = Ending langsung membuka pesan konstelasi (dari Secret Ending: "Stay with my constellation")
  const [resumeMessage, setResumeMessage] = useState(false)

  const [homeWarping, setHomeWarping] = useState(false)
  const homeTimers = useRef([])

  const {
    playing,
    toggleMusic,
    volume,
    setVolume,
    changeSection,
    getAudioData,
    audioProgress,
  } = useAmbientAudio()

  const handleNavigate = useCallback((id) => {
    setFact(null)
    setMessage(null)
    if (id === activeSection) return
    if (id !== 'ending') setResumeMessage(false)
    setActiveSection(id)
    setFadeState('hidden')
    changeSection(MUSIC_ALIAS[id] ?? id)

    setTimeout(() => {
      setRenderedSection(id)
      setFadeState('visible')
    }, 400)
  }, [activeSection, changeSection])

  // Dipanggil Intro saat tombol tahan penuh: Intro menganimasikan Bumi, App mengganti section
  const handleWarpNavigate = useCallback((id) => {
    if (warpingRef.current || id === activeSection) return
    warpingRef.current = true
    setFact(null)
    setMessage(null)
    setWarp({ phase: 'warp', startTime: Date.now() })

    setTimeout(() => {
      handleNavigate(id)
      setTimeout(() => {
        setWarp({ phase: 'idle', startTime: 0 })
        warpingRef.current = false
      }, ARRIVE_MS)
    }, TRAVEL_MS)
  }, [activeSection, handleNavigate])

  const handleFadeNavigate = useCallback((id) => {
    if (warpingRef.current || id === activeSection) return
    warpingRef.current = true
    setFact(null)
    setMessage(null)
    setFadePhase('dark')

    setTimeout(() => {
      handleNavigate(id)
      setFadePhase('light')

      setTimeout(() => {
        setFadePhase('idle')
        warpingRef.current = false
      }, FADE_IN)
    }, FADE_OUT)
  }, [activeSection, handleNavigate])

  // Pulang ke Bumi (HomeWarp) — bisa dipanggil dari section mana pun (mis. Secret Ending)
  const handleReturnHome = useCallback(async () => {
    if (warpingRef.current) return
    warpingRef.current = true
    setFact(null)
    setMessage(null)
    try {
      const { HOME_MS } = await import('./components/HomeWarp.jsx')
      setHomeWarping(true)
      const t1 = setTimeout(() => {
        handleNavigate('intro')
        const t2 = setTimeout(() => {
          setHomeWarping(false)
          setVisitor(INITIAL_VISITOR) // perjalanan baru = pengunjung baru
          resetSessionCallsign()      // callsign anonim baru untuk pengunjung berikutnya
          warpingRef.current = false
        }, HOME_END_PAD)
        homeTimers.current.push(t2)
      }, HOME_MS)
      homeTimers.current.push(t1)
    } catch (_) {
      warpingRef.current = false
    }
  }, [handleNavigate])

  // Secret Ending → "Stay with my constellation"
  const handleBackToConstellation = useCallback(() => {
    setResumeMessage(true)
    handleFadeNavigate('ending')
  }, [handleFadeNavigate])

  // Wishing: wish berhasil terkirim dan animasi bintang naik selesai → Secret Ending
  const handleWishSubmitted = useCallback(() => {
    handleFadeNavigate('secret')
  }, [handleFadeNavigate])

  useEffect(() => () => homeTimers.current.forEach(clearTimeout), [])

  const transitioning = warp.phase !== 'idle' || fadePhase !== 'idle' || homeWarping
  useEffect(() => {
    document.body.classList.toggle('hide-global-ui', transitioning)
    return () => document.body.classList.remove('hide-global-ui')
  }, [transitioning])

  // Preload Solar System supaya perpindahan dari intro tidak tersendat
  useEffect(() => {
    const t = setTimeout(() => import('./sections/SolarSystem.jsx'), 1500)
    return () => clearTimeout(t)
  }, [])

  const showFact = useCallback((d) => setFact(d), [])
  const closeFact = useCallback(() => setFact(null), [])

  const showMessage = useCallback(() => {
    setMessage(UNIVERSE_MESSAGES[Math.floor(Math.random() * UNIVERSE_MESSAGES.length)])
  }, [])
  const closeMessage = useCallback(() => setMessage(null), [])

  // Secret Ending tidak punya titik sendiri di NavDots/ProgressBar → ikut 'wishing'
  const navSection = activeSection === 'secret' ? 'wishing' : activeSection
  const currentIdx = SECTION_IDS.indexOf(navSection)
  const progress = Math.max(0, currentIdx / (SECTION_IDS.length - 1))

  return (
    <>
      <StarField count={500} mouseParallax />
      <ProgressBar progress={progress} />
      <NavDots active={navSection} onNavigate={handleNavigate} />

      <MusicPlayer
        playing={playing}
        onToggle={toggleMusic}
        volume={volume}
        onVolumeChange={setVolume}
        activeSection={activeSection}
        getAudioData={getAudioData}
        audioProgress={audioProgress}
      />

      <div
        className={`fade-overlay ${fadePhase === 'dark' ? 'fade-dark' : ''} ${fadePhase === 'light' ? 'fade-light' : ''}`}
        style={{ '--fade-out': `${FADE_OUT}ms`, '--fade-in': `${FADE_IN}ms` }}
        aria-hidden="true"
      />

      <main className={`section-wrapper ${fadeState}`}>
        <Suspense fallback={null}>
          {renderedSection === 'intro' && (
            <Intro
              onNext={() => handleWarpNavigate('solar')}
              isWarping={warp.phase !== 'idle'}
              warpStartTime={warp.startTime}
              travelMs={TRAVEL_MS}
            />
          )}
          {renderedSection === 'solar' && (
            <SolarSystem onFact={showFact} onNext={() => handleFadeNavigate('phenomena')} />
          )}
          {renderedSection === 'phenomena' && (
            <Phenomena onFact={showFact} onNext={() => handleFadeNavigate('deepspace')} />
          )}
          {renderedSection === 'deepspace' && (
            <DeepSpace onFact={showFact} onNext={() => handleFadeNavigate('ending')} />
          )}
          {renderedSection === 'ending' && (
            <Ending
              visitor={visitor}
              onReveal={updateVisitor}
              onNext={() => handleFadeNavigate('wishing')}
              onBackToChoices={() => handleFadeNavigate('secret')}
              resumeMessage={resumeMessage}
            />
          )}
          {renderedSection === 'wishing' && (
            <Wishing
              visitor={visitor}
              onVisitorChange={updateVisitor}
              onGallery={() => handleFadeNavigate('gallery')}
              onSubmitted={handleWishSubmitted}
              onBackToConstellation={handleBackToConstellation}
              onReturnHome={handleReturnHome}
            />
          )}
          {renderedSection === 'secret' && (

              onVisitorChange={updateVisitor}

              onGallery={() => handleFadeNavigate('gallery')}

            <Gallery
              visitor={visitor}

          )}

      {homeWarping && (

          <HomeWarp active name={visitor.name} points={visitor.constellationPoints} />

// COSMIC JOURNEY — Wishes service
// Fase 2: dummy (Promise + delay). Fase 6: isi file ini diganti fetch ke
// /api/submit-wish tanpa mengubah kontrak fungsi yang dipakai UI.

export const MESSAGE_MIN = 5
export const MESSAGE_MAX = 280
export const NAME_MAX = 30

// id = nilai yang dikirim ke backend (whitelist server-side di Fase 6)
export const CATEGORIES = [
    { id: 'feedback', label: 'Feedback' },
    { id: 'suggestion', label: 'Suggestion' },
    { id: 'developer', label: 'About the developer' },
    { id: 'wish', label: 'Just a wish' },
]
const CATEGORY_IDS = new Set(CATEGORIES.map((c) => c.id))

// ── Callsign anonim: dibuat sekali per sesi (mis. "Astronaut-7F2") ──
const CALLSIGN_CHARS = '23456789ABCDEFGHJKLMNPQRSTUVWXYZ' // tanpa 0/O/1/I agar tidak membingungkan
let sessionCallsign = null

export function getSessionCallsign() {
    if (sessionCallsign) return sessionCallsign
    const buf = new Uint32Array(3)
    if (typeof window !== 'undefined' && window.crypto?.getRandomValues) {
        window.crypto.getRandomValues(buf)
    } else {
        for (let i = 0; i < buf.length; i++) buf[i] = Math.floor(Math.random() * 4294967296)
    }
    const code = Array.from(buf, (n) => CALLSIGN_CHARS[n % CALLSIGN_CHARS.length]).join('')
    sessionCallsign = `Astronaut-${code}`
    return sessionCallsign
}

// Dipanggil saat pengunjung "pulang ke Bumi" (perjalanan baru = pengunjung baru)
export function resetSessionCallsign() {
    sessionCallsign = null
}

// ── Pembersihan teks (server nanti melakukan hal yang sama) ──
const cleanName = (s) =>
    String(s ?? '').replace(/[\u0000-\u001F\u007F-\u009F]/g, ' ').replace(/\s+/g, ' ').trim()

const cleanMessage = (s) =>
    String(s ?? '')
        .replace(/\r\n?/g, '\n')
        .replace(/[\u0000-\u0008\u000B\u000C\u000E-\u001F\u007F-\u009F]/g, '')
        .replace(/\n{3,}/g, '\n\n')
        .trim()

const wait = (ms) => new Promise((r) => setTimeout(r, ms))

function fail(code, message) {
    const err = new Error(message)
    err.code = code
    return err
}

/**
 * Kontrak (tetap sama di Fase 6):
 *   submitWish({ displayName, isAnonymous, category, message, rating, isPrivate, consent, website })
 *   → Promise<record>  atau throw Error dengan .code: 'validation' | 'rate_limit' | 'network' | 'server'
 *   record = { id, displayName, isAnonymous, category, message, rating, isPrivate, createdAt, status }
 * `website` = field honeypot (harus kosong; server menolak jika terisi).
 */
export async function submitWish(payload) {
    await wait(700 + Math.random() * 500)

    const message = cleanMessage(payload?.message)
    const rating = Number(payload?.rating)
    const isPrivate = Boolean(payload?.isPrivate)

    if (message.length < MESSAGE_MIN || message.length > MESSAGE_MAX) {
        throw fail('validation', `Your message needs ${MESSAGE_MIN}–${MESSAGE_MAX} characters.`)
    }
    if (!Number.isInteger(rating) || rating < 1 || rating > 5) {
        throw fail('validation', 'Please choose a rating from 1 to 5 stars.')
    }
    if (!CATEGORY_IDS.has(payload?.category)) {
        throw fail('validation', 'Please choose a category.')
    }
    if (!isPrivate && !payload?.consent) {
        throw fail('validation', 'Please agree to show your message in the Gallery, or send it privately.')
    }

    const isAnonymous = Boolean(payload?.isAnonymous)
    const displayName = cleanName(payload?.displayName).slice(0, NAME_MAX) || getSessionCallsign()

    // Honeypot terisi (bot): di fase backend ditolak/di-skip. Di dummy: pura-pura sukses.
    const record = {
        id: `local-${Date.now().toString(36)}`,
        displayName,
        isAnonymous,
        category: payload.category,
        message,
        rating,
        isPrivate,
        createdAt: new Date().toISOString(),
        status: 'pending',
    }
    return record
}

// ============================================================
// COSMIC JOURNEY — Synthesized sound effects (no audio files)
// Usage: import { sfx } from '../hooks/useSfx.js'  →  sfx.chime(2)
// Must be called from a user gesture (click/keypress) the first time.
// ============================================================

let ctx = null
let master = null
let noiseBuf = null
let muted = false
try { muted = localStorage.getItem('cj-sfx-muted') === '1' } catch (_) { }

const SCALE = [523.25, 587.33, 659.25, 783.99, 880.0, 987.77, 1174.66, 1318.51] // C major pentatonic-ish

function getCtx() {
  if (muted) return null
  if (!ctx) {
    const AC = window.AudioContext || window.webkitAudioContext
    if (!AC) return null
    ctx = new AC()
    master = ctx.createGain()
    master.gain.value = 0.35
    master.connect(ctx.destination)
  }
  if (ctx.state === 'suspended') ctx.resume()
  return ctx
}

function getNoise(c) {
  if (!noiseBuf) {
    noiseBuf = c.createBuffer(1, c.sampleRate * 2, c.sampleRate)
    const d = noiseBuf.getChannelData(0)
    for (let i = 0; i < d.length; i++) d[i] = Math.random() * 2 - 1
  }
  return noiseBuf
}

// Mengembalikan gain node agar suara bisa dihentikan lebih awal (lihat sfx.open)
function tone({ freq, to, type = 'sine', dur = 0.4, vol = 0.2, delay = 0, attack = 0.01 }) {
  const c = getCtx()
  if (!c) return null
  const t = c.currentTime + delay
  const o = c.createOscillator()
  const g = c.createGain()
  o.type = type
  o.frequency.setValueAtTime(freq, t)
  if (to) o.frequency.exponentialRampToValueAtTime(to, t + dur)
  g.gain.setValueAtTime(0.0001, t)
  g.gain.exponentialRampToValueAtTime(vol, t + Math.min(attack, dur * 0.5))
  g.gain.exponentialRampToValueAtTime(0.0001, t + dur)
  o.connect(g)
  g.connect(master)
  o.start(t)
  o.stop(t + dur + 0.05)
  return g
}

function noise({ dur = 1, vol = 0.2, from = 300, to = 3000, q = 1, delay = 0, swell = 0.5 }) {
  const c = getCtx()
  if (!c) return null
  const t = c.currentTime + delay
  const src = c.createBufferSource()
  src.buffer = getNoise(c)
  src.loop = true
  const f = c.createBiquadFilter()
  f.type = 'bandpass'
  f.Q.value = q
  f.frequency.setValueAtTime(from, t)
  f.frequency.exponentialRampToValueAtTime(to, t + dur)
  const g = c.createGain()
  g.gain.setValueAtTime(0.0001, t)
  g.gain.exponentialRampToValueAtTime(vol, t + dur * swell)
  g.gain.exponentialRampToValueAtTime(0.0001, t + dur)
  src.connect(f)
  f.connect(g)
  g.connect(master)
  src.start(t)
  src.stop(t + dur + 0.05)
  return g
}

export const sfx = {
  /** Tiny tick for hovering cards */
  hover() { tone({ freq: 1400, to: 1800, dur: 0.06, vol: 0.035 }) },

  /** Generic soft click */
  click() { tone({ freq: 660, to: 440, dur: 0.09, vol: 0.12, type: 'triangle' }) },

  /** Bell-like chime. Pass an index (e.g. planet number) to vary the pitch. */
  chime(i = 0) {
    const f = SCALE[i % SCALE.length]
    tone({ freq: f, dur: 1.5, vol: 0.16 })
    tone({ freq: f * 2, dur: 0.9, vol: 0.05, delay: 0.01 })
    tone({ freq: f * 1.5, dur: 1.1, vol: 0.04, delay: 0.02 })
  },

  /** Deep swelling drone. Mengembalikan fungsi stop() yang memudarkan suara dengan cepat. */
  open() {
    const nodes = [
      tone({ freq: 55, to: 82, dur: 2.4, vol: 0.3, attack: 0.7 }),
      tone({ freq: 110, to: 165, dur: 2.4, vol: 0.1, attack: 0.9, type: 'triangle' }),
      noise({ dur: 2.4, vol: 0.07, from: 80, to: 420, q: 0.7, swell: 0.6 }),
    ].filter(Boolean)
    return () => {
      if (!ctx) return
      const t = ctx.currentTime
      nodes.forEach((g) => {
        try {
          const v = Math.max(g.gain.value, 0.0001)
          g.gain.cancelScheduledValues(t)
          g.gain.setValueAtTime(v, t)
          g.gain.exponentialRampToValueAtTime(0.0001, t + 0.25)
        } catch (_) { }
      })
    }
  },

import { useState, useRef, useCallback, useEffect } from 'react'

const SECTION_MUSIC = {
  intro: '/audio/intro.mp3',
  solar: '/audio/solar.mp3',
  phenomena: '/audio/phenomena.mp3',
  deepspace: '/audio/deepspace.mp3',
  ending: '/audio/ending.mp3',
  wishing: '/audio/ending.mp3', // lanjut dari Ending tanpa putus (src sama → tidak crossfade)
  gallery: '/audio/intro.mp3', // hangat & ramah, seperti kembali ke titik berangkat
  default: '/audio/intro.mp3',
}

const FADE_DURATION = 1500
const FADE_STEPS = 30

export function useAmbientAudio() {
  const [playing, setPlaying] = useState(false)
  const [volume, setVolumeState] = useState(0.5)
  const [audioProgress, setAudioProgress] = useState({ currentTime: 0, duration: 0, progress: 0 })

  const audioRef = useRef(null)
  const currentSrc = useRef(null)
  const volumeRef = useRef(0.5)

  const isPlaying = useRef(false)
  const manualPause = useRef(false) // Mencegah autoplay menyala jika user sudah sengaja pause
  const initAutoplay = useRef(false)

  const ctxRef = useRef(null)
  const analyserRef = useRef(null)
  const sourceRef = useRef(null)
  const gainNodeRef = useRef(null)

  const getCtx = useCallback(() => {
    if (!ctxRef.current) {
      const ctx = new (window.AudioContext || window.webkitAudioContext)()
      const analyser = ctx.createAnalyser()
      analyser.fftSize = 256
      analyser.smoothingTimeConstant = 0.8
      const gain = ctx.createGain()
      gain.gain.value = volumeRef.current
      analyser.connect(gain)
      gain.connect(ctx.destination)
      ctxRef.current = ctx
      analyserRef.current = analyser
      gainNodeRef.current = gain
    }
    return ctxRef.current
  }, [])

  const connectAudio = useCallback((audioEl) => {
    const ctx = getCtx()
    if (ctx.state === 'suspended') ctx.resume()
    try { sourceRef.current?.disconnect() } catch (_) { }
    const src = ctx.createMediaElementSource(audioEl)
    src.connect(analyserRef.current)
    sourceRef.current = src
  }, [getCtx])

  const getAudioData = useCallback((targetArray) => {
    if (analyserRef.current && targetArray) {
      analyserRef.current.getByteFrequencyData(targetArray)
      return true
    }
    return false
  }, [])

  const fadeTo = useCallback((audioEl, from, to, onDone) => {
    if (!audioEl) return
    let current = from
    const step = (to - from) / FADE_STEPS
    const ms = FADE_DURATION / FADE_STEPS
    const timer = setInterval(() => {
      current += step
      const clamped = Math.max(0, Math.min(1, current))
      try { audioEl.volume = clamped * volumeRef.current } catch (_) { }
      const done = step > 0 ? current >= to : current <= to
      if (done) {
        clearInterval(timer); onDone?.()
      }
    }, ms)
    return timer
  }, [])

  const playSrc = useCallback((src) => {
    if (currentSrc.current === src && audioRef.current && !audioRef.current.paused) return null

    const oldAudio = audioRef.current
    if (oldAudio) {
      const oldVol = volumeRef.current > 0 ? oldAudio.volume / volumeRef.current : 0
      fadeTo(oldAudio, oldVol, 0, () => {
        oldAudio.pause()
        oldAudio.src = ''
      })
    }

    setAudioProgress({ currentTime: 0, duration: 0, progress: 0 })

    const audio = new Audio(src)
    audio.loop = true
    audio.volume = 0
    audio.crossOrigin = 'anonymous'
    audio.preload = 'auto'

    audio.addEventListener('error', () =>
      console.warn(`[useAudio] Gagal load: ${src} — pastikan file ada di public/audio/`)
    )

    audio.addEventListener('timeupdate', () => {
      setAudioProgress({
        currentTime: audio.currentTime,
        duration: audio.duration || 0,
        progress: audio.duration ? (audio.currentTime / audio.duration) : 0,
      })
    })

    audioRef.current = audio
    currentSrc.current = src

    const playPromise = audio.play()
    playPromise.then(() => {
      connectAudio(audio)
      fadeTo(audio, 0, 1, null)
    }).catch(() => { })

    return playPromise
  }, [fadeTo, connectAudio])

  // Logika Autoplay di percobaan pertama web load
  useEffect(() => {
    if (initAutoplay.current) return
    initAutoplay.current = true

    const attemptPlay = () => {
      const src = SECTION_MUSIC.default
      const promise = playSrc(src)

      if (promise) {
        promise.then(() => {
          isPlaying.current = true
          setPlaying(true)
        }).catch(() => {
          // Autoplay diblokir, pasang listener interaksi sekali pakai
          const onInteract = async () => {
            if (manualPause.current || isPlaying.current) return

            const ctx = getCtx()
            if (ctx.state === 'suspended') await ctx.resume()

            const current = currentSrc.current || SECTION_MUSIC.default
            playSrc(current)?.then(() => {
              isPlaying.current = true
              setPlaying(true)
            }).catch(e => console.warn(e))
          }

          window.addEventListener('pointerdown', onInteract, { once: true })
          window.addEventListener('keydown', onInteract, { once: true })
          window.addEventListener('touchstart', onInteract, { once: true })
        })
      }
    }

    attemptPlay()
  }, [playSrc, getCtx])

  const toggleMusic = useCallback(() => {
    if (!isPlaying.current) {
      manualPause.current = false // User membatalkan pause
      const src = currentSrc.current || SECTION_MUSIC.default
      playSrc(src)?.then(() => {
        isPlaying.current = true
        setPlaying(true)
      })
    } else {
      manualPause.current = true // User menyengaja pause
      const audio = audioRef.current
      if (audio) {
        const vol = volumeRef.current > 0 ? audio.volume / volumeRef.current : 0
        fadeTo(audio, vol, 0, () => audio.pause())
      }
      isPlaying.current = false
      setPlaying(false)
    }
  }, [playSrc, fadeTo])

  const changeSection = useCallback((sectionId) => {
    const src = SECTION_MUSIC[sectionId] ?? SECTION_MUSIC.default
    if (src === currentSrc.current) return

    if (isPlaying.current) {
      playSrc(src)
    } else {
      currentSrc.current = src
    }
  }, [playSrc])

  const setVolume = useCallback((val) => {
    const v = Math.max(0, Math.min(1, val))
    setVolumeState(v)
    volumeRef.current = v
    if (audioRef.current && !audioRef.current.paused) {
      audioRef.current.volume = v
    }
    if (gainNodeRef.current) gainNodeRef.current.gain.value = v
  }, [])

  useEffect(() => {
    return () => {
      if (audioRef.current) {
        audioRef.current.pause(); audioRef.current.src = ''
      }
      try { ctxRef.current?.close() } catch (_) { }
    }
  }, [])

  return { playing, toggleMusic, volume, setVolume, changeSection, getAudioData, audioProgress }
}
