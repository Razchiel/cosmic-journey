import React, { useState, useCallback, useEffect, useRef, Suspense } from 'react'
import StarField from './components/StarField.jsx'
import NavDots from './components/NavDots.jsx'
import MusicPlayer from './components/MusicPlayer.jsx'
import ProgressBar from './components/ProgressBar.jsx'
import FactPopup from './components/FactPopup.jsx'
import MessageBox from './components/MessageBox.jsx'

// Lazy loading sections (kode three.js tidak akan diload di awal)
const Intro = React.lazy(() => import('./sections/Intro.jsx'))
const SolarSystem = React.lazy(() => import('./sections/SolarSystem.jsx'))
const Phenomena = React.lazy(() => import('./sections/Phenomena.jsx'))
const DeepSpace = React.lazy(() => import('./sections/DeepSpace.jsx'))
const Ending = React.lazy(() => import('./sections/Ending.jsx'))
const Wishing = React.lazy(() => import('./sections/Wishing.jsx'))
const Gallery = React.lazy(() => import('./sections/Gallery.jsx'))
// HomeWarp memuat three.js → lazy; dipanggil dari section mana pun lewat handleReturnHome
const HomeWarp = React.lazy(() => import('./components/HomeWarp.jsx'))

import { useAmbientAudio } from './hooks/useAudio.js'
import { UNIVERSE_MESSAGES } from './data/index.js'
import { resetSessionCallsign } from './services/wishes.js'

const SECTION_IDS = ['intro', 'solar', 'phenomena', 'deepspace', 'ending', 'wishing', 'gallery']

// State pengunjung (in-memory saja — sengaja TIDAK disimpan di localStorage/server)
// Diisi Ending saat reveal, dibaca Wishing & Gallery. Direset setelah pulang ke Bumi.
const INITIAL_VISITOR = {
  name: '',                // nama penerima konstelasi
  sender: '',              // pengirim (hanya mode "For Someone Else")
  constellationPoints: [],
  finalMessage: '',
  coords: '',
  wishSubmitted: false,
  mySubmission: null,      // data kiriman wish pengunjung ini (untuk highlight "Your wish")
  skipped: false,          // true = Wishing di-skip langsung ke Gallery (tidak membuka Secret Ending)
}

const HOME_END_PAD = 800 // jeda setelah pindah ke intro sebelum overlay HomeWarp dilepas

// Perjalanan Bumi (intro → solar): Bumi menjauh & mengecil ke posisinya di tata surya
const TRAVEL_MS = 2800 // dari tombol penuh sampai section berganti
const ARRIVE_MS = 1000 // jeda sampai UI global muncul lagi

// Transisi gelap-terang untuk perpindahan section selain intro → solar
const FADE_OUT = 700
const FADE_IN = 800

export default function App() {
  const [fact, setFact] = useState(null)
  const [message, setMessage] = useState(null)

  const [activeSection, setActiveSection] = useState('intro')
  const [renderedSection, setRenderedSection] = useState('intro')
  const [fadeState, setFadeState] = useState('visible')

  // Perjalanan intro → solar: 'idle' → 'warp' → 'idle'
  const [warp, setWarp] = useState({ phase: 'idle', startTime: 0 })
  const warpingRef = useRef(false)

  const [fadePhase, setFadePhase] = useState('idle')

  const [visitor, setVisitor] = useState(INITIAL_VISITOR)
  const updateVisitor = useCallback((patch) => setVisitor((v) => ({ ...v, ...patch })), [])
  // true = Ending langsung membuka pesan konstelasi (dari Secret Ending: "Stay with my constellation")
  const [resumeMessage, setResumeMessage] = useState(false)

  const [homeWarping, setHomeWarping] = useState(false)
  const homeTimers = useRef([])

  const {
    playing,
    toggleMusic,
    volume,
    setVolume,
    changeSection,
    getAudioData,
    audioProgress,
  } = useAmbientAudio()

  const handleNavigate = useCallback((id) => {
    setFact(null)
    setMessage(null)
    if (id === activeSection) return
    if (id !== 'ending') setResumeMessage(false)
    setActiveSection(id)
    setFadeState('hidden')
    changeSection(id)

    setTimeout(() => {
      setRenderedSection(id)
      setFadeState('visible')
    }, 400)
  }, [activeSection, changeSection])

  // Dipanggil Intro saat tombol tahan penuh: Intro menganimasikan Bumi, App mengganti section
  const handleWarpNavigate = useCallback((id) => {
    if (warpingRef.current || id === activeSection) return
    warpingRef.current = true
    setFact(null)
    setMessage(null)
    setWarp({ phase: 'warp', startTime: Date.now() })

    setTimeout(() => {
      handleNavigate(id)
      setTimeout(() => {
        setWarp({ phase: 'idle', startTime: 0 })
        warpingRef.current = false
      }, ARRIVE_MS)
    }, TRAVEL_MS)
  }, [activeSection, handleNavigate])

  const handleFadeNavigate = useCallback((id) => {
    if (warpingRef.current || id === activeSection) return
    warpingRef.current = true
    setFact(null)
    setMessage(null)
    setFadePhase('dark')

    setTimeout(() => {
      handleNavigate(id)
      setFadePhase('light')

      setTimeout(() => {
        setFadePhase('idle')
        warpingRef.current = false
      }, FADE_IN)
    }, FADE_OUT)
  }, [activeSection, handleNavigate])

  // Pulang ke Bumi (HomeWarp) — bisa dipanggil dari section mana pun (mis. Secret Ending)
  const handleReturnHome = useCallback(async () => {
    if (warpingRef.current) return
    warpingRef.current = true
    setFact(null)
    setMessage(null)
    try {
      const { HOME_MS } = await import('./components/HomeWarp.jsx')
      setHomeWarping(true)
      const t1 = setTimeout(() => {
        handleNavigate('intro')
        const t2 = setTimeout(() => {
          setHomeWarping(false)
          setVisitor(INITIAL_VISITOR) // perjalanan baru = pengunjung baru
          resetSessionCallsign()      // callsign anonim baru untuk pengunjung berikutnya
          warpingRef.current = false
        }, HOME_END_PAD)
        homeTimers.current.push(t2)
      }, HOME_MS)
      homeTimers.current.push(t1)
    } catch (_) {
      warpingRef.current = false
    }
  }, [handleNavigate])

  // Secret Ending → "Stay with my constellation"
  const handleBackToConstellation = useCallback(() => {
    setResumeMessage(true)
    handleFadeNavigate('ending')
  }, [handleFadeNavigate])

  // Wishing: wish berhasil terkirim dan animasi bintang naik selesai.
  // SEMENTARA (Fase 2): belum ada Secret Ending, jadi langsung ke Gallery.
  // Fase 3: ganti tujuan ini ke Secret Ending.
  const handleWishSubmitted = useCallback(() => {
    handleFadeNavigate('gallery')
  }, [handleFadeNavigate])

  useEffect(() => () => homeTimers.current.forEach(clearTimeout), [])

  const transitioning = warp.phase !== 'idle' || fadePhase !== 'idle' || homeWarping
  useEffect(() => {
    document.body.classList.toggle('hide-global-ui', transitioning)
    return () => document.body.classList.remove('hide-global-ui')
  }, [transitioning])

  // Preload Solar System supaya perpindahan dari intro tidak tersendat
  useEffect(() => {
    const t = setTimeout(() => import('./sections/SolarSystem.jsx'), 1500)
    return () => clearTimeout(t)
  }, [])

  const showFact = useCallback((d) => setFact(d), [])
  const closeFact = useCallback(() => setFact(null), [])

  const showMessage = useCallback(() => {
    setMessage(UNIVERSE_MESSAGES[Math.floor(Math.random() * UNIVERSE_MESSAGES.length)])
  }, [])
  const closeMessage = useCallback(() => setMessage(null), [])

  const currentIdx = SECTION_IDS.indexOf(activeSection)
  const progress = Math.max(0, currentIdx / (SECTION_IDS.length - 1))

  return (
    <>
      <StarField count={500} mouseParallax />
      <ProgressBar progress={progress} />
      <NavDots active={activeSection} onNavigate={handleNavigate} />

      <MusicPlayer
        playing={playing}
        onToggle={toggleMusic}
        volume={volume}
        onVolumeChange={setVolume}
        activeSection={activeSection}
        getAudioData={getAudioData}
        audioProgress={audioProgress}
      />

      <div
        className={`fade-overlay ${fadePhase === 'dark' ? 'fade-dark' : ''} ${fadePhase === 'light' ? 'fade-light' : ''}`}
        style={{ '--fade-out': `${FADE_OUT}ms`, '--fade-in': `${FADE_IN}ms` }}
        aria-hidden="true"
      />

      <main className={`section-wrapper ${fadeState}`}>
        <Suspense fallback={null}>
          {renderedSection === 'intro' && (
            <Intro
              onNext={() => handleWarpNavigate('solar')}
              isWarping={warp.phase !== 'idle'}
              warpStartTime={warp.startTime}
              travelMs={TRAVEL_MS}
            />
          )}
          {renderedSection === 'solar' && (
            <SolarSystem onFact={showFact} onNext={() => handleFadeNavigate('phenomena')} />
          )}
          {renderedSection === 'phenomena' && (
            <Phenomena onFact={showFact} onNext={() => handleFadeNavigate('deepspace')} />
          )}
          {renderedSection === 'deepspace' && (
            <DeepSpace onFact={showFact} onNext={() => handleFadeNavigate('ending')} />
          )}
          {renderedSection === 'ending' && (
            <Ending
              visitor={visitor}
              onReveal={updateVisitor}
              onNext={() => handleFadeNavigate('wishing')}
              resumeMessage={resumeMessage}
            />
          )}
          {renderedSection === 'wishing' && (
            <Wishing
              visitor={visitor}
              onVisitorChange={updateVisitor}
              onGallery={() => handleFadeNavigate('gallery')}
              onSubmitted={handleWishSubmitted}
              onBackToConstellation={handleBackToConstellation}
              onReturnHome={handleReturnHome}
            />
          )}
          {renderedSection === 'gallery' && (
            <Gallery
              visitor={visitor}
              onBackToConstellation={handleBackToConstellation}
              onReturnHome={handleReturnHome}
            />
          )}
        </Suspense>
      </main>

      {homeWarping && (
        <Suspense fallback={null}>
          <HomeWarp active name={visitor.name} points={visitor.constellationPoints} />
        </Suspense>
      )}

      <FactPopup fact={fact} onClose={closeFact} />
      <MessageBox message={message} onClose={closeMessage} />
    </>
  )
}

import { memo, useEffect, useRef, useState } from 'react'
import { sfx } from '../hooks/useSfx.js'
import WishForm from '../components/WishForm.jsx'
import styles from './Wishing.module.css'

const rand = (a, b) => a + Math.random() * (b - a)
const randInt = (a, b) => Math.floor(rand(a, b + 1))

const HINT_AFTER = 20      // detik tanpa tangkapan sebelum glow bintang terang naik
const FADE_MS = 650        // bintang biasa memudar setelah di-tap
const BURST_MS = 1400      // ledakan cahaya bintang terang
const BURST_MS_REDUCED = 900

function Wishing({ onVisitorChange, onGallery, onSubmitted }) {
    const wrapRef = useRef(null)
    const canvasRef = useRef(null)

    const [round, setRound] = useState(0)          // naik = main ulang (lepas bintang)
    const [caught, setCaught] = useState(false)    // bintang terang sudah tertangkap
    const [formOpen, setFormOpen] = useState(false)
    const [draft, setDraft] = useState(null)       // isi form yang tersimpan saat form ditutup

    // Dipanggil dari loop canvas (lewat ref agar effect tidak perlu dibuat ulang)
    const handleCaught = () => {
        setCaught(true)
        setFormOpen(true)
    }
    const caughtCbRef = useRef(handleCaught)
    caughtCbRef.current = handleCaught

    useEffect(() => {
        const canvas = canvasRef.current
        const wrap = wrapRef.current
        if (!canvas || !wrap) return
        const ctx = canvas.getContext('2d')
        const reduce = window.matchMedia('(prefers-reduced-motion: reduce)').matches

        // ── Ukuran canvas (DPR dibatasi 1.5 agar hemat di HP) ───────────
        let w = 0, h = 0
        const resize = () => {
            const r = wrap.getBoundingClientRect()
            w = r.width
            h = r.height
            const dpr = Math.min(window.devicePixelRatio || 1, 1.5)
            canvas.width = Math.max(1, Math.round(w * dpr))
            canvas.height = Math.max(1, Math.round(h * dpr))
            ctx.setTransform(dpr, 0, 0, dpr, 0, 0)
        }
        resize()
        const ro = new ResizeObserver(resize)
        ro.observe(wrap)

        // ── State permainan (semua di ref/closure, tanpa re-render) ─────
        // Antrean: false = bintang biasa, true = bintang terang. Terang tidak pernah jadi yang pertama.
        const total = reduce ? randInt(5, 6) : randInt(5, 10)
        const queue = Array.from({ length: total - 1 }, () => false)
        queue.splice(randInt(1, queue.length), 0, true)

        const stars = []
        let time = 0
        let nextSpawn = 0.9
        let boost = 0                 // 0..1, glow petunjuk bintang terang
        let phase = 'play'            // 'play' | 'burst'
        let burst = null
        let done = false
        const burstMs = reduce ? BURST_MS_REDUCED : BURST_MS

        const spawn = (bright) => {
            const x = rand(0.06 * w, 0.94 * w)
            let dir = x < w / 2 ? 1 : -1            // arah ke sisi layar yang lebih lebar
            if (!bright && Math.random() < 0.2) dir = -dir
            const room = dir > 0 ? w * 0.97 - x : x - w * 0.03
            let ang = bright ? rand(0.3, 0.7) : rand(0.35, 1.0)
            if (!bright || dir === (x < w / 2 ? 1 : -1)) {
                ang = Math.min(ang, Math.atan(Math.max(room, 40) / (h * 0.75)))
            }
            ang = Math.max(ang, 0.18)
            const base = Math.min(Math.max(h, 360), 900)
            const speed = (bright ? rand(0.15, 0.19) : rand(0.16, 0.26)) * base * (reduce ? 0.6 : 1)
            stars.push({
                x,
                y: rand(-30, -10),
                vx: dir * Math.sin(ang) * speed,
                vy: Math.cos(ang) * speed,
                bright,
                r: bright ? 4.6 : rand(1.6, 2.6),
                tail: bright ? rand(190, 230) : rand(60, 120),
                state: 'fall',          // fall | fade | caught
                t: 0,
                rx: 0, ry: 0,           // posisi riak saat di-tap
                phase: rand(0, Math.PI * 2),
            })
        }

        const requeueBright = () => {
            if (phase !== 'play') return
            queue.splice(randInt(0, Math.min(1, queue.length)), 0, true)
            if (Math.random() < 0.6) queue.unshift(false)   // selingan bintang biasa
            nextSpawn = Math.min(nextSpawn, 1.0)
        }

        // ── Interaksi: hitbox ≥ 48px (radius 26 / 34), mencakup sedikit ekor ──
        const onPointerDown = (e) => {
            if (phase !== 'play') return
            const rect = canvas.getBoundingClientRect()
            const px = e.clientX - rect.left
            const py = e.clientY - rect.top

            let hit = null
            let best = Infinity
            for (const s of stars) {
                if (s.state !== 'fall') continue
                const R = s.bright ? 34 : 26
                const len = Math.hypot(s.vx, s.vy) || 1
                const ax = s.x, ay = s.y
                const bx = s.x - (s.vx / len) * 30
                const by = s.y - (s.vy / len) * 30
                const abx = bx - ax, aby = by - ay
                let t = ((px - ax) * abx + (py - ay) * aby) / (abx * abx + aby * aby)
                t = Math.max(0, Math.min(1, t))
                const d = Math.hypot(px - (ax + abx * t), py - (ay + aby * t))
                if (d <= R) {
                    const score = s.bright ? d - 100 : d   // bintang terang diprioritaskan
                    if (score < best) { best = score; hit = s }
                }
            }
            if (!hit) return
            e.preventDefault()

            if (hit.bright) {
                hit.state = 'caught'
                hit.t = 0
                phase = 'burst'
                burst = { x: hit.x, y: hit.y, t: 0 }
                stars.forEach((s) => {
                    if (s !== hit && s.state === 'fall') {
                        s.state = 'fade'; s.t = 0; s.rx = s.x; s.ry = s.y
                        s.vx *= 0.3; s.vy *= 0.3
                    }
                })
                sfx.reveal()
                sfx.chime(4)
            } else {
                hit.state = 'fade'
                hit.t = 0
                hit.rx = hit.x
                hit.ry = hit.y
                hit.vx *= 0.3
                hit.vy *= 0.3
                sfx.click()
            }
        }
        canvas.addEventListener('pointerdown', onPointerDown, { passive: false })

        // ── Update ──────────────────────────────────────────────────────
        const update = (dt) => {
            time += dt

            if (phase === 'play') {
                if (time > HINT_AFTER) boost = Math.min(1, (time - HINT_AFTER) / 4)

                nextSpawn -= dt
                if (nextSpawn <= 0 && queue.length) {
                    spawn(queue.shift())
                    nextSpawn = rand(1.2, 2.8) * (reduce ? 1.4 : 1)
                }
            } else if (burst) {
                burst.t = Math.min(burst.t + dt * 1000, burstMs)
                if (burst.t >= burstMs && !done) {
                    done = true
                    caughtCbRef.current()
                }
            }

            for (let i = stars.length - 1; i >= 0; i--) {
                const s = stars[i]
                s.t += dt * 1000
                if (s.state === 'caught') continue
                s.x += s.vx * dt
                s.y += s.vy * dt
                if (s.state === 'fade') {
                    if (s.t >= FADE_MS) stars.splice(i, 1)
                    continue
                }
                const len = Math.hypot(s.vx, s.vy) || 1
                const tx = s.x - (s.vx / len) * s.tail
                const ty = s.y - (s.vy / len) * s.tail
                const gone = ty > h + 10 || (s.vx > 0 && tx > w + 10) || (s.vx < 0 && tx < -10)
                if (gone) {
                    stars.splice(i, 1)
                    if (s.bright) requeueBright()   // bintang terang lolos → muncul lagi
                }
            }
        }

        // ── Gambar ──────────────────────────────────────────────────────
        const drawStar = (s) => {
            if (s.state === 'caught') return
            const len = Math.hypot(s.vx, s.vy) || 1
            const ux = s.vx / len, uy = s.vy / len
            const alpha = s.state === 'fade' ? Math.max(0, 1 - s.t / FADE_MS) : 1

            ctx.globalAlpha = alpha

            // ekor gradasi
            const tx = s.x - ux * s.tail
            const ty = s.y - uy * s.tail
            const g = ctx.createLinearGradient(tx, ty, s.x, s.y)
            g.addColorStop(0, 'rgba(170,200,255,0)')
            g.addColorStop(1, `rgba(235,243,255,${s.bright ? 0.95 : 0.8})`)
            ctx.strokeStyle = g
            ctx.lineWidth = s.bright ? 2.6 : 1.3
            ctx.lineCap = 'round'
            ctx.beginPath()
            ctx.moveTo(tx, ty)
            ctx.lineTo(s.x, s.y)
            ctx.stroke()

            // halo (bintang terang berdenyut halus; naik saat petunjuk)
            let R, A
            if (s.bright) {
                const pulse = reduce ? 0.5 : 0.5 + 0.5 * Math.sin(time * 2.4 + s.phase)
                R = 24 + pulse * 8 + boost * 10
                A = 0.22 + pulse * 0.12 + boost * 0.35
            } else {
                R = s.r * 5
                A = 0.4
            }
            const hg = ctx.createRadialGradient(s.x, s.y, 0, s.x, s.y, R)
            hg.addColorStop(0, `rgba(225,236,255,${Math.min(1, A * 2)})`)
            hg.addColorStop(0.35, `rgba(160,190,255,${A})`)
            hg.addColorStop(1, 'rgba(120,150,255,0)')
            ctx.fillStyle = hg
            ctx.beginPath()
            ctx.arc(s.x, s.y, R, 0, Math.PI * 2)
            ctx.fill()

            ctx.fillStyle = '#fff'
            ctx.beginPath()
            ctx.arc(s.x, s.y, s.r, 0, Math.PI * 2)
            ctx.fill()

            ctx.globalAlpha = 1

            // riak kecil saat bintang biasa di-tap
            if (s.state === 'fade') {
                const p = Math.min(1, s.t / FADE_MS)
                ctx.strokeStyle = `rgba(170,200,255,${0.5 * (1 - p)})`
                ctx.lineWidth = 1.2
                ctx.beginPath()
                ctx.arc(s.rx, s.ry, 6 + 34 * p, 0, Math.PI * 2)
                ctx.stroke()
            }
        }

        const drawBurst = () => {
            const p = burst.t / burstMs
            const e = 1 - Math.pow(1 - p, 3)               // ease-out
            const diag = Math.hypot(w, h)
            const A = p < 0.5 ? p * 2 : 1 - (p - 0.5) * 0.9 // naik cepat, mengendap ke ~0.55
            const R = 20 + e * diag * 0.6

            const g = ctx.createRadialGradient(burst.x, burst.y, 0, burst.x, burst.y, R)
            g.addColorStop(0, `rgba(255,255,255,${A})`)
            g.addColorStop(0.25, `rgba(205,222,255,${A * 0.8})`)
            g.addColorStop(1, 'rgba(120,150,255,0)')
            ctx.fillStyle = g
            ctx.fillRect(0, 0, w, h)

            // inti bintang membesar
            ctx.fillStyle = `rgba(255,255,255,${Math.min(1, 0.4 + p)})`
            ctx.beginPath()
            ctx.arc(burst.x, burst.y, 5 + e * 38, 0, Math.PI * 2)
            ctx.fill()

            // gelombang kejut (dilewati saat reduced motion)
            if (!reduce) {
                ctx.strokeStyle = `rgba(190,210,255,${Math.max(0, 0.7 * (1 - p))})`
                ctx.lineWidth = 2
                ctx.beginPath()
                ctx.arc(burst.x, burst.y, e * Math.min(w, h) * 0.55, 0, Math.PI * 2)
                ctx.stroke()
            }
        }

        const draw = () => {
            ctx.clearRect(0, 0, w, h)
            ctx.save()
            if (burst && !reduce && burst.t < 350) {      // getaran singkat, tanpa saat reduced motion
                const k = 3 * (1 - burst.t / 350)
                ctx.translate((Math.random() - 0.5) * k, (Math.random() - 0.5) * k)
            }
            stars.forEach(drawStar)
            if (burst) drawBurst()
            ctx.restore()
        }

        let raf = 0
        let last = performance.now()
        const frame = (now) => {
            raf = requestAnimationFrame(frame)
            const dt = Math.min((now - last) / 1000, 0.05)
            last = now
            update(dt)
            draw()
        }
        raf = requestAnimationFrame(frame)

        return () => {
            cancelAnimationFrame(raf)
            canvas.removeEventListener('pointerdown', onPointerDown)
            ro.disconnect()
        }

            </button>

            {caught && !formOpen && (

// ── SHARE CARD: render konstelasi + pesan ke PNG 1080×1350 ─────────────
function wrapLines(g, text, maxWidth) {
  const words = text.split(' ')
  const lines = []
  let line = ''
  for (const w of words) {
    const test = line ? `${line} ${w}` : w
    if (g.measureText(test).width > maxWidth && line) {
      lines.push(line)
      line = w
    } else {
      line = test
    }
  }
  if (line) lines.push(line)
  return lines
}

async function drawShareCard({ name, from, message, points, coords }) {
  try {
    await Promise.all([
      document.fonts.load('italic 300 60px "Cormorant Garamond"'),
      document.fonts.load('300 24px "Outfit"'),
    ])
  } catch (_) { /* font fallback */ }

  const W = 1080, H = 1350
  const canvas = document.createElement('canvas')
  canvas.width = W
  canvas.height = H
  const g = canvas.getContext('2d')
  const serif = '"Cormorant Garamond", Georgia, serif'
  const sans = '"Outfit", system-ui, sans-serif'
  const setSpacing = (px) => { if ('letterSpacing' in g) g.letterSpacing = `${px}px` }

  // background
  const bg = g.createLinearGradient(0, 0, 0, H)
  bg.addColorStop(0, '#02021a')
  bg.addColorStop(0.55, '#0b0720')
  bg.addColorStop(1, '#16091f')
  g.fillStyle = bg
  g.fillRect(0, 0, W, H)

  // soft nebula washes
  const wash = (x, y, r, color) => {
    const rg = g.createRadialGradient(x, y, 0, x, y, r)
    rg.addColorStop(0, color)
    rg.addColorStop(1, 'rgba(0,0,0,0)')
    g.fillStyle = rg
    g.fillRect(0, 0, W, H)
  }
  wash(220, 260, 520, 'rgba(80,110,255,0.16)')
  wash(900, 1050, 560, 'rgba(160,80,255,0.14)')

  // star dust
  for (let i = 0; i < 320; i++) {
    const r = Math.random() * 1.6 + 0.3
    g.fillStyle = `rgba(210,225,255,${Math.random() * 0.7 + 0.15})`
    g.beginPath()
    g.arc(Math.random() * W, Math.random() * H, r, 0, Math.PI * 2)
    g.fill()
  }

  // frame
  g.strokeStyle = 'rgba(136,170,255,0.25)'
  g.lineWidth = 2
  g.beginPath()
  g.roundRect ? g.roundRect(40, 40, W - 80, H - 80, 28) : g.rect(40, 40, W - 80, H - 80)
  g.stroke()

  // constellation (fit into 520×340 area centred at y=330)
  const xs = points.map((p) => p.x), ys = points.map((p) => p.y)
  const minX = Math.min(...xs), maxX = Math.max(...xs)
  const minY = Math.min(...ys), maxY = Math.max(...ys)
  const scale = Math.min(520 / ((maxX - minX) || 1), 340 / ((maxY - minY) || 1))
  const pts = points.map((p) => ({
    x: W / 2 + (p.x - (minX + maxX) / 2) * scale,
    y: 330 + (p.y - (minY + maxY) / 2) * scale,
  }))

  g.strokeStyle = 'rgba(136,170,255,0.55)'
  g.lineWidth = 2.5
  g.setLineDash([10, 10])
  g.beginPath()
  pts.forEach((p, i) => (i ? g.lineTo(p.x, p.y) : g.moveTo(p.x, p.y)))
  g.stroke()
  g.setLineDash([])

  pts.forEach((p) => {
    const halo = g.createRadialGradient(p.x, p.y, 0, p.x, p.y, 30)
    halo.addColorStop(0, 'rgba(160,190,255,0.85)')
    halo.addColorStop(1, 'rgba(160,190,255,0)')
    g.fillStyle = halo
    g.fillRect(p.x - 30, p.y - 30, 60, 60)
    g.fillStyle = '#fff'
    g.beginPath()
    g.arc(p.x, p.y, 7, 0, Math.PI * 2)
    g.fill()
  })

  g.textAlign = 'center'

  // label under constellation
  g.fillStyle = '#88aacc'
  g.font = `300 22px ${sans}`
  setSpacing(6)
  g.fillText(`CONSTELLATION ${name.toUpperCase()}`, W / 2, 560)

  // divider
  g.strokeStyle = 'rgba(255,255,255,0.12)'
  g.lineWidth = 1
  g.beginPath()
  g.moveTo(160, 600)
  g.lineTo(W - 160, 600)
  g.stroke()

  // header
  g.fillStyle = '#88aacc'
  g.font = `300 20px ${sans}`
  setSpacing(5)
  g.fillText('DECRYPTED TRANSMISSION', W / 2, 668)
  setSpacing(0)

  // "Dear Name,"  — shrink to fit
  let size = 70
  g.fillStyle = '#aaccff'
  do {
    g.font = `italic 300 ${size}px ${serif}`
    size -= 4
  } while (g.measureText(`Dear ${name},`).width > 880 && size > 30)
  g.fillText(`Dear ${name},`, W / 2, 770)

  // message
  // shrink the font until the message fits above the signature/footer
  g.fillStyle = '#e0e5ff'
  let msgSize = 54
  let lines = []
  while (true) {
    g.font = `italic 300 ${msgSize}px ${serif}`
    lines = wrapLines(g, `“${message}”`, 840)
    if (lines.length * msgSize * 1.45 <= 270 || msgSize <= 32) break
    msgSize -= 2
  }
  const lh = msgSize * 1.45
  lines.forEach((ln, i) => g.fillText(ln, W / 2, 870 + i * lh))

  // signature (for "someone else")
  if (from) {
    const sigY = Math.min(870 + (lines.length - 1) * lh + 105, 1185)
    g.fillStyle = '#8596aa'
    g.font = `300 20px ${sans}`
    setSpacing(3)
    g.fillText('WITH COSMIC LOVE,', W / 2, sigY - 44)
    setSpacing(0)
    let sigSize = 56
    g.fillStyle = '#ff99cc'
    do {
      g.font = `italic 300 ${sigSize}px ${serif}`
      sigSize -= 4
    } while (g.measureText(from).width > 700 && sigSize > 28)
    g.fillText(from, W / 2, sigY + 8)
  }

  // footer
  g.fillStyle = '#8596aa'
  g.font = `300 20px ${sans}`
  setSpacing(4)
  g.fillText('COSMIC JOURNEY', W / 2, H - 110)
  g.font = `300 17px ${sans}`
  setSpacing(2)
  g.fillText(`COSMIC COORDS: ${coords}`, W / 2, H - 76)
  setSpacing(0)

  return new Promise((resolve) => canvas.toBlob(resolve, 'image/png'))
}

import { useRef, useState, useEffect, useMemo, memo } from 'react'
import { Canvas, useFrame } from '@react-three/fiber'
import * as THREE from 'three'
import { sfx } from '../hooks/useSfx.js'
import styles from './HomeWarp.module.css'

// Timeline (ms)
const T_LAUNCH = 1800    // roket mulai naik, warp mulai
const T_COUNT = 10000    // hitung mundur "3" (lalu 2, 1 tiap 900 ms)
const T_FLASH = 12700    // kilatan menuju Bumi
export const HOME_MS = 13200 // dipakai Ending sebelum pindah ke Intro

const COUNT = 360
const clamp01 = (v) => Math.min(1, Math.max(0, v))

// Hanya starfield (garis bintang saat warp)
function Scene({ t0 }) {
    const stars = useRef()
    const dummy = useMemo(() => new THREE.Object3D(), [])
    const ps = useMemo(() => Array.from({ length: COUNT }, () => ({
        x: (Math.random() - 0.5) * 40,
        y: (Math.random() - 0.5) * 60,
        z: Math.random() * 13 - 14,
    })), [])

    useFrame((state, delta) => {
        if (!t0.current || !stars.current) return
        const t = performance.now() - t0.current
        const k = clamp01((t - T_LAUNCH) / 6000)   // sebelumnya 3500
        const s = k * k * (3 - 2 * k)

        const amp = s * s * 0.14
        state.camera.position.x = (Math.random() - 0.5) * amp
        state.camera.position.y = (Math.random() - 0.5) * amp

        const speed = 0.2 + s * 60
        const stretch = 1 + s * 110
        ps.forEach((p, i) => {
            p.y -= speed * delta
            if (p.y < -30) { p.y = 30; p.x = (Math.random() - 0.5) * 40 }
            dummy.position.set(p.x, p.y, p.z)
            dummy.scale.set(1, stretch, 1)
            dummy.updateMatrix()
            stars.current.setMatrixAt(i, dummy.matrix)
        })
        stars.current.instanceMatrix.needsUpdate = true
    })

    return (
        <instancedMesh ref={stars} args={[null, null, COUNT]} frustumCulled={false}>
            <boxGeometry args={[0.025, 0.025, 0.025]} />
            <meshBasicMaterial color="#e0f0ff" transparent opacity={0.8} />
        </instancedMesh>
    )
}

// Roket 2D: mengapung & bergoyang → api menyala → berdiri tegak & naik perlahan
function Rocket2D({ t0 }) {
    const body = useRef()
    const flame = useRef()

    useEffect(() => {
        let raf = 0
        const loop = () => {
            const t = performance.now() - t0.current
            const lk = clamp01((t - T_LAUNCH) / (T_FLASH - T_LAUNCH - 800))   // sebelumnya 4700
            const rise = lk * lk * (3 - 2 * lk)                       // naik 0..1
            const up = clamp01((t - T_LAUNCH) / 1400)
            const upright = up * up * (3 - 2 * up)                    // tegak 0..1
            const ig = clamp01((t - T_LAUNCH) / 900)                  // api menyala 0..1
            const sway = Math.sin(t * 0.0016) * 10 * (1 - upright)    // goyang kiri-kanan
            const bob = Math.sin(t * 0.0021) * 10 * (1 - upright)
            const shake = ig * (Math.random() - 0.5) * 1.6
            const y = bob - rise * window.innerHeight * 0.3
            if (body.current) {
                body.current.style.transform =
                    `translate(${shake}px, ${y}px) rotate(${sway + (1 - upright) * 6}deg)`
            }
            if (flame.current) {
                const len = ig * (0.75 + rise * 0.6) + (Math.random() - 0.5) * 0.16 * ig
                flame.current.style.transform = `scale(${1 + (Math.random() - 0.5) * 0.1 * ig}, ${Math.max(len, 0.001)})`
            }
            raf = requestAnimationFrame(loop)
        }
        raf = requestAnimationFrame(loop)
        return () => cancelAnimationFrame(raf)
    }, [t0])

    return (
        <div className={styles.rocketWrap}>
            <div ref={body} className={styles.rocket}>
                <svg viewBox="0 0 80 150" className={styles.rocketSvg}>
                    <g ref={flame} className={styles.flame} style={{ transformOrigin: '40px 108px' }}>
                        <path d="M32 108 C32 128 38 142 40 152 C42 142 48 128 48 108Z" fill="#ff8a30" opacity="0.9" />

function NameConstellation({ name, points }) {
    if (!points?.length) return null

let sessionCallsign = null

export function getSessionCallsign() {
    if (sessionCallsign) return sessionCallsign
    const buf = new Uint32Array(3)
    if (typeof window !== 'undefined' && window.crypto?.getRandomValues) {
        window.crypto.getRandomValues(buf)
    } else {
        for (let i = 0; i < buf.length; i++) buf[i] = Math.floor(Math.random() * 4294967296)
    }
    const code = Array.from(buf, (n) => CALLSIGN_CHARS[n % CALLSIGN_CHARS.length]).join('')
    sessionCallsign = `Astronaut-${code}`
    return sessionCallsign
}

export async function submitWish(payload) {

let ctx = null
let master = null
let noiseBuf = null
let muted = false
try { muted = localStorage.getItem('cj-sfx-muted') === '1' } catch (_) { }

const SCALE = [523.25, 587.33, 659.25, 783.99, 880.0, 987.77, 1174.66, 1318.51] // C major pentatonic-ish

function getCtx() {
  if (muted) return null
  if (!ctx) {
    const AC = window.AudioContext || window.webkitAudioContext
    if (!AC) return null
    ctx = new AC()
    master = ctx.createGain()
    master.gain.value = 0.35
    master.connect(ctx.destination)
  }
  if (ctx.state === 'suspended') ctx.resume()
  return ctx
}

function getNoise(c) {
  if (!noiseBuf) {
    noiseBuf = c.createBuffer(1, c.sampleRate * 2, c.sampleRate)
    const d = noiseBuf.getChannelData(0)
    for (let i = 0; i < d.length; i++) d[i] = Math.random() * 2 - 1
  }
  return noiseBuf
}

// Mengembalikan gain node agar suara bisa dihentikan lebih awal (lihat sfx.open)
function tone({ freq, to, type = 'sine', dur = 0.4, vol = 0.2, delay = 0, attack = 0.01 }) {
  const c = getCtx()
  if (!c) return null
  const t = c.currentTime + delay
  const o = c.createOscillator()
  const g = c.createGain()
  o.type = type
  o.frequency.setValueAtTime(freq, t)
  if (to) o.frequency.exponentialRampToValueAtTime(to, t + dur)
  g.gain.setValueAtTime(0.0001, t)
  g.gain.exponentialRampToValueAtTime(vol, t + Math.min(attack, dur * 0.5))
  g.gain.exponentialRampToValueAtTime(0.0001, t + dur)
  o.connect(g)
  g.connect(master)
  o.start(t)
  o.stop(t + dur + 0.05)
  return g
}

function noise({ dur = 1, vol = 0.2, from = 300, to = 3000, q = 1, delay = 0, swell = 0.5 }) {
  const c = getCtx()
  if (!c) return null
  const t = c.currentTime + delay
  const src = c.createBufferSource()
  src.buffer = getNoise(c)
  src.loop = true
  const f = c.createBiquadFilter()
  f.type = 'bandpass'
  f.Q.value = q
  f.frequency.setValueAtTime(from, t)
  f.frequency.exponentialRampToValueAtTime(to, t + dur)
  const g = c.createGain()
  g.gain.setValueAtTime(0.0001, t)
  g.gain.exponentialRampToValueAtTime(vol, t + dur * swell)
  g.gain.exponentialRampToValueAtTime(0.0001, t + dur)
  src.connect(f)
  f.connect(g)
  g.connect(master)
  src.start(t)
  src.stop(t + dur + 0.05)
  return g
}

import { useState, useRef, useCallback, useEffect } from 'react'

const SECTION_MUSIC = {
  intro: '/audio/intro.mp3',
  solar: '/audio/solar.mp3',
  phenomena: '/audio/phenomena.mp3',
  deepspace: '/audio/deepspace.mp3',
  ending: '/audio/ending.mp3',
  wishing: '/audio/ending.mp3', // lanjut dari Ending tanpa putus (src sama → tidak crossfade)
  gallery: '/audio/intro.mp3', // hangat & ramah, seperti kembali ke titik berangkat
  default: '/audio/intro.mp3',
}

const FADE_DURATION = 1500
const FADE_STEPS = 30

export function useAmbientAudio() {
  const [playing, setPlaying] = useState(false)
  const [volume, setVolumeState] = useState(0.5)
  const [audioProgress, setAudioProgress] = useState({ currentTime: 0, duration: 0, progress: 0 })

  const audioRef = useRef(null)
  const currentSrc = useRef(null)
  const volumeRef = useRef(0.5)

  const isPlaying = useRef(false)
  const manualPause = useRef(false) // Mencegah autoplay menyala jika user sudah sengaja pause
  const initAutoplay = useRef(false)

  const ctxRef = useRef(null)
  const analyserRef = useRef(null)
  const sourceRef = useRef(null)
  const gainNodeRef = useRef(null)

  const getCtx = useCallback(() => {
    if (!ctxRef.current) {
      const ctx = new (window.AudioContext || window.webkitAudioContext)()
      const analyser = ctx.createAnalyser()
      analyser.fftSize = 256
      analyser.smoothingTimeConstant = 0.8
      const gain = ctx.createGain()
      gain.gain.value = volumeRef.current
      analyser.connect(gain)
      gain.connect(ctx.destination)
      ctxRef.current = ctx
      analyserRef.current = analyser
      gainNodeRef.current = gain
    }
    return ctxRef.current
  }, [])

  const connectAudio = useCallback((audioEl) => {
    const ctx = getCtx()
    if (ctx.state === 'suspended') ctx.resume()
    try { sourceRef.current?.disconnect() } catch (_) { }
    const src = ctx.createMediaElementSource(audioEl)
    src.connect(analyserRef.current)
    sourceRef.current = src
  }, [getCtx])

  const getAudioData = useCallback((targetArray) => {
    if (analyserRef.current && targetArray) {
      analyserRef.current.getByteFrequencyData(targetArray)
      return true
    }
    return false
  }, [])

  const fadeTo = useCallback((audioEl, from, to, onDone) => {
    if (!audioEl) return
    let current = from
    const step = (to - from) / FADE_STEPS
    const ms = FADE_DURATION / FADE_STEPS
    const timer = setInterval(() => {
      current += step
      const clamped = Math.max(0, Math.min(1, current))
      try { audioEl.volume = clamped * volumeRef.current } catch (_) { }
      const done = step > 0 ? current >= to : current <= to
      if (done) {
        clearInterval(timer); onDone?.()
      }
    }, ms)
    return timer
  }, [])

  const playSrc = useCallback((src) => {
    if (currentSrc.current === src && audioRef.current && !audioRef.current.paused) return null

    const oldAudio = audioRef.current
    if (oldAudio) {
      const oldVol = volumeRef.current > 0 ? oldAudio.volume / volumeRef.current : 0
      fadeTo(oldAudio, oldVol, 0, () => {
        oldAudio.pause()
        oldAudio.src = ''
      })
    }

    setAudioProgress({ currentTime: 0, duration: 0, progress: 0 })

    const audio = new Audio(src)
    audio.loop = true
    audio.volume = 0
    audio.crossOrigin = 'anonymous'
    audio.preload = 'auto'

    audio.addEventListener('error', () =>
      console.warn(`[useAudio] Gagal load: ${src} — pastikan file ada di public/audio/`)
    )

    audio.addEventListener('timeupdate', () => {
      setAudioProgress({
        currentTime: audio.currentTime,
        duration: audio.duration || 0,
        progress: audio.duration ? (audio.currentTime / audio.duration) : 0,
      })
    })

    audioRef.current = audio
    currentSrc.current = src

    const playPromise = audio.play()
    playPromise.then(() => {
      connectAudio(audio)
      fadeTo(audio, 0, 1, null)
    }).catch(() => { })

    return playPromise
  }, [fadeTo, connectAudio])

  // Logika Autoplay di percobaan pertama web load
  useEffect(() => {
    if (initAutoplay.current) return
    initAutoplay.current = true

    const attemptPlay = () => {
      const src = SECTION_MUSIC.default
      const promise = playSrc(src)

      if (promise) {
        promise.then(() => {
          isPlaying.current = true
          setPlaying(true)
        }).catch(() => {
          // Autoplay diblokir, pasang listener interaksi sekali pakai
          const onInteract = async () => {
            if (manualPause.current || isPlaying.current) return

            const ctx = getCtx()
            if (ctx.state === 'suspended') await ctx.resume()

            const current = currentSrc.current || SECTION_MUSIC.default
            playSrc(current)?.then(() => {
              isPlaying.current = true
              setPlaying(true)
            }).catch(e => console.warn(e))
          }

          window.addEventListener('pointerdown', onInteract, { once: true })
          window.addEventListener('keydown', onInteract, { once: true })
          window.addEventListener('touchstart', onInteract, { once: true })
        })
      }
    }

    attemptPlay()
  }, [playSrc, getCtx])

  const toggleMusic = useCallback(() => {
    if (!isPlaying.current) {
      manualPause.current = false // User membatalkan pause
      const src = currentSrc.current || SECTION_MUSIC.default
      playSrc(src)?.then(() => {
        isPlaying.current = true
        setPlaying(true)
      })
    } else {
      manualPause.current = true // User menyengaja pause
      const audio = audioRef.current
      if (audio) {
        const vol = volumeRef.current > 0 ? audio.volume / volumeRef.current : 0
        fadeTo(audio, vol, 0, () => audio.pause())
      }
      isPlaying.current = false
      setPlaying(false)
    }
  }, [playSrc, fadeTo])

  const changeSection = useCallback((sectionId) => {
    const src = SECTION_MUSIC[sectionId] ?? SECTION_MUSIC.default
    if (src === currentSrc.current) return

    if (isPlaying.current) {
      playSrc(src)
    } else {
      currentSrc.current = src
    }
  }, [playSrc])

  const setVolume = useCallback((val) => {
    const v = Math.max(0, Math.min(1, val))
    setVolumeState(v)
    volumeRef.current = v
    if (audioRef.current && !audioRef.current.paused) {
      audioRef.current.volume = v
    }
    if (gainNodeRef.current) gainNodeRef.current.gain.value = v
  }, [])

  useEffect(() => {
    return () => {
      if (audioRef.current) {
        audioRef.current.pause(); audioRef.current.src = ''
      }
      try { ctxRef.current?.close() } catch (_) { }
    }
  }, [])

  return { playing, toggleMusic, volume, setVolume, changeSection, getAudioData, audioProgress }
}
