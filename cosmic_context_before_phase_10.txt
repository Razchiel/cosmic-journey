  solar: 0.2,
  phenomena: 0.4,
  wishing: 0.45,
  ending: 0.55,
  deepspace: 0.7,
  gallery: 0.8,
  secret: 0.8,
}

export default function App() {
  const [fact, setFact] = useState(null)
  const [message, setMessage] = useState(null)

  const [activeSection, setActiveSection] = useState('intro')
  const [renderedSection, setRenderedSection] = useState('intro')
  const [fadeState, setFadeState] = useState('visible')
  const navTimer = useRef(0) // timer pergantian section (dibatalkan jika ada perpindahan baru)

  // Perjalanan intro → solar: 'idle' → 'warp' → 'idle'
  const [warp, setWarp] = useState({ phase: 'idle', startTime: 0 })
  const warpingRef = useRef(false)

  const [fadePhase, setFadePhase] = useState('idle')

  const [visitor, setVisitor] = useState(INITIAL_VISITOR)
  const updateVisitor = useCallback((patch) => setVisitor((v) => ({ ...v, ...patch })), [])
  // true = Ending langsung membuka pesan konstelasi (dari Secret Ending: "Stay with my constellation")
  const [resumeMessage, setResumeMessage] = useState(false)

  const [homeWarping, setHomeWarping] = useState(false)
  const homeTimers = useRef([])
  // true = HomeWarp sudah memberi tahu bahwa Bumi mendekat: bintang kembali terang (depth 0)
  const [homeBright, setHomeBright] = useState(false)
  const brightenStars = useCallback(() => setHomeBright(true), [])

  // Mode Comfort: hook saja, belum ada UI (Fase 12 akan memanggil setComfort).
  // Efek: starfield satu tingkat lebih redup + class body.comfort-mode (kontras teks naik).
  const [comfort] = useState(false)

  const {
    playing,
    toggleMusic,
    volume,
    setVolume,
    changeSection,
    getAudioData,
    audioProgress,
  } = useAmbientAudio()

  const handleNavigate = useCallback((id) => {
    setFact(null)
    setMessage(null)
    if (id === activeSection) return
    if (id !== 'ending') setResumeMessage(false)
    setActiveSection(id)
    setFadeState('hidden')
    changeSection(MUSIC_ALIAS[id] ?? id)

    // Perpindahan cepat beruntun: hanya tujuan terakhir yang dirender (tanpa kilatan section perantara)
    clearTimeout(navTimer.current)
    navTimer.current = setTimeout(() => {
      setRenderedSection(id)
      setFadeState('visible')
    }, 400)
  }, [activeSection, changeSection])

  // NavDots: diabaikan selama transisi / HomeWarp (tombol tetap bisa dijangkau keyboard walau tersembunyi)
  const handleNavDot = useCallback((id) => {
    if (warpingRef.current) return
    handleNavigate(id)
  }, [handleNavigate])

  // Dipanggil Intro saat tombol tahan penuh: Intro menganimasikan Bumi, App mengganti section
  const handleWarpNavigate = useCallback((id) => {
    if (warpingRef.current || id === activeSection) return
    warpingRef.current = true
    setFact(null)
    setMessage(null)
    setWarp({ phase: 'warp', startTime: Date.now() })

    setTimeout(() => {
      handleNavigate(id)
      setTimeout(() => {
        setWarp({ phase: 'idle', startTime: 0 })
        warpingRef.current = false
      }, ARRIVE_MS)
    }, TRAVEL_MS)
  }, [activeSection, handleNavigate])

  const handleFadeNavigate = useCallback((id) => {
    if (warpingRef.current || id === activeSection) return
    warpingRef.current = true
    setFact(null)
    setMessage(null)
    setFadePhase('dark')

    setTimeout(() => {
      handleNavigate(id)
      setFadePhase('light')

      setTimeout(() => {
        setFadePhase('idle')
        warpingRef.current = false
      }, FADE_IN)
    }, FADE_OUT)
  }, [activeSection, handleNavigate])

  // Pulang ke Bumi (HomeWarp) — bisa dipanggil dari section mana pun (mis. Secret Ending)
  const handleReturnHome = useCallback(async () => {
    if (warpingRef.current) return
    warpingRef.current = true
    setFact(null)
    setMessage(null)
    setHomeBright(false)
    try {
      const { HOME_MS } = await import('./components/HomeWarp.jsx')
      setHomeWarping(true)
      const t1 = setTimeout(() => {
        handleNavigate('intro')
        const t2 = setTimeout(() => {
          setHomeWarping(false)
          setHomeBright(false)        // Intro sudah depth 0, jadi tidak ada lompatan visual
          setVisitor(INITIAL_VISITOR) // perjalanan baru = pengunjung baru
          resetSessionCallsign()      // callsign anonim baru untuk pengunjung berikutnya
          warpingRef.current = false
        }, HOME_END_PAD)
        homeTimers.current.push(t2)
      }, HOME_MS)
      homeTimers.current.push(t1)
    } catch (_) {
      warpingRef.current = false
    }
  }, [handleNavigate])

  // Secret Ending → "Stay with my constellation"
  const handleBackToConstellation = useCallback(() => {
    setResumeMessage(true)
    handleFadeNavigate('ending')
  }, [handleFadeNavigate])

  // Wishing: wish berhasil terkirim dan animasi bintang naik selesai → Secret Ending
  const handleWishSubmitted = useCallback(() => {
    handleFadeNavigate('secret')
  }, [handleFadeNavigate])

  useEffect(() => () => {
    homeTimers.current.forEach(clearTimeout)
    clearTimeout(navTimer.current)
  }, [])

  const transitioning = warp.phase !== 'idle' || fadePhase !== 'idle' || homeWarping
  useEffect(() => {
    document.body.classList.toggle('hide-global-ui', transitioning)
    return () => document.body.classList.remove('hide-global-ui')
  }, [transitioning])

  // Hook Comfort: kontras teks naik lewat token di index.css
  useEffect(() => {
    document.body.classList.toggle('comfort-mode', comfort)
    return () => document.body.classList.remove('comfort-mode')
  }, [comfort])

  // Preload Solar System supaya perpindahan dari intro tidak tersendat
  useEffect(() => {
    const t = setTimeout(() => import('./sections/SolarSystem.jsx'), 1500)
    return () => clearTimeout(t)
  }, [])

  // Preload section akhir + HomeWarp begitu pengunjung mencapai Deep Space,
  // supaya layar tidak kosong saat pindah ke Ending / Wishing / Secret / Gallery / pulang.
  const preloaded = useRef(false)
  useEffect(() => {
    if (preloaded.current || !PRELOAD_FROM.has(activeSection)) return
    const t = setTimeout(() => {
      preloaded.current = true
      const quiet = () => { }
      import('./sections/Ending.jsx').catch(quiet)
      import('./sections/Wishing.jsx').catch(quiet)
      import('./sections/SecretEnding.jsx').catch(quiet)
      import('./sections/Gallery.jsx').catch(quiet)
      import('./components/HomeWarp.jsx').catch(quiet)
    }, 1200)
    return () => clearTimeout(t)
  }, [activeSection])

  const showFact = useCallback((d) => setFact(d), [])
  const closeFact = useCallback(() => setFact(null), [])

  const showMessage = useCallback(() => {
    setMessage(UNIVERSE_MESSAGES[Math.floor(Math.random() * UNIVERSE_MESSAGES.length)])
  }, [])
  const closeMessage = useCallback(() => setMessage(null), [])

  // Secret Ending tidak punya titik sendiri di NavDots/ProgressBar → ikut 'wishing'
  const navSection = activeSection === 'secret' ? 'wishing' : activeSection
  const currentIdx = SECTION_IDS.indexOf(navSection)
  const progress = Math.max(0, currentIdx / (SECTION_IDS.length - 1))

  // Depth starfield mengikuti activeSection (berubah saat navigasi dimulai; StarField
  // meng-easing selama ~2 dtk sehingga tumpang tindih dengan fade section, tanpa lompatan).
  const baseDepth = SECTION_DEPTH[activeSection] ?? 0
  const starDepth = homeBright
    ? 0
    : Math.min(1, baseDepth + (comfort ? COMFORT_DEPTH_STEP : 0))

  return (
    <>
      <StarField count={500} mouseParallax depth={starDepth} />
      <ProgressBar progress={progress} />
      <NavDots active={navSection} onNavigate={handleNavDot} />

      <MusicPlayer
        playing={playing}
        onToggle={toggleMusic}
        volume={volume}
        onVolumeChange={setVolume}
        activeSection={MUSIC_ALIAS[activeSection] ?? activeSection}
        getAudioData={getAudioData}
        audioProgress={audioProgress}
      />

      <div
        className={`fade-overlay ${fadePhase === 'dark' ? 'fade-dark' : ''} ${fadePhase === 'light' ? 'fade-light' : ''}`}
        style={{ '--fade-out': `${FADE_OUT}ms`, '--fade-in': `${FADE_IN}ms` }}
        aria-hidden="true"
      />

      <main className={`section-wrapper ${fadeState}`}>
        <Suspense fallback={null}>
          {renderedSection === 'intro' && (
            <Intro
              onNext={() => handleWarpNavigate('solar')}
              isWarping={warp.phase !== 'idle'}
              warpStartTime={warp.startTime}
              travelMs={TRAVEL_MS}
            />
          )}
          {renderedSection === 'solar' && (
            <SolarSystem onFact={showFact} onNext={() => handleFadeNavigate('phenomena')} />
          )}
          {renderedSection === 'phenomena' && (
            <Phenomena onFact={showFact} onNext={() => handleFadeNavigate('deepspace')} />
          )}
          {renderedSection === 'deepspace' && (
            <DeepSpace onFact={showFact} onNext={() => handleFadeNavigate('ending')} />
          )}
          {renderedSection === 'ending' && (

              visitor={visitor}
              onReveal={updateVisitor}
              onNext={() => handleFadeNavigate('wishing')}
              onBackToChoices={() => handleFadeNavigate('secret')}
              resumeMessage={resumeMessage}

          )}
          {renderedSection === 'wishing' && (

            <SecretEnding

              onVisitorChange={updateVisitor}
              onStay={handleBackToConstellation}
              onGallery={() => handleFadeNavigate('gallery')}

              visitor={visitor}

          />

  )
}

const prefersReduced = () =>
  typeof window !== 'undefined' &&

function getAudioMetrics(data) {
  if (!data || data.length === 0) return { bass: 0, avg: 0, mid: 0 }
  const bassEnd = Math.floor(data.length * 0.06)
  const midEnd = Math.floor(data.length * 0.35)
  let bass = 0, mid = 0, total = 0
  for (let i = 0; i < data.length; i++) {
    const v = data[i] / 255
    total += v
    if (i < bassEnd) bass += v
    else if (i < midEnd) mid += v
  }
  return { bass: bass / bassEnd, mid: mid / (midEnd - bassEnd), avg: total / data.length }
}

function fmt(sec) {
  if (!sec || isNaN(sec)) return '0:00'
  const m = Math.floor(sec / 60)
  const s = Math.floor(sec % 60).toString().padStart(2, '0')
  return `${m}:${s}`
}

// ── Animated artwork canvas per theme (IDLE ANIMATION) ──────────────
function ArtCanvas({ theme, color }) {
  const canvasRef = useRef(null)
  const rafRef = useRef(null)
  const t0 = useRef(Date.now())
  const particles = useRef([])

  useEffect(() => {
    const canvas = canvasRef.current
    if (!canvas) return
    const ctx = canvas.getContext('2d')
    const W = canvas.width, H = canvas.height

    const initParticles = () => {
      particles.current = []
      if (theme === 'stars' || theme === 'nebula' || theme === 'clouds') {
        for (let i = 0; i < 40; i++) {
          particles.current.push({
            x: Math.random() * W, y: Math.random() * H,
            r: Math.random() * 1.5 + 0.3,
            speed: Math.random() * 0.3 + 0.05,
            phase: Math.random() * Math.PI * 2,
            opacity: Math.random() * 0.8 + 0.2,
          })
        }
      }
    }
    initParticles()

    const draw = () => {
      const t = (Date.now() - t0.current) / 1000
      const c1 = color[0], c2 = color[1]

      ctx.clearRect(0, 0, W, H)

      if (theme === 'nebula') {
        for (let i = 0; i < 3; i++) {
          const cx = W * (0.3 + i * 0.2) + Math.sin(t * 0.3 + i) * 8
          const cy = H * (0.4 + Math.sin(t * 0.2 + i * 1.2) * 0.2)
          const r = 22 + Math.sin(t + i) * 4
          const g = ctx.createRadialGradient(cx, cy, 0, cx, cy, r)
          g.addColorStop(0, c1 + (i === 1 ? 'cc' : '88'))
          g.addColorStop(0.5, c2 + '44')
          g.addColorStop(1, 'transparent')
          ctx.fillStyle = g
          ctx.beginPath(); ctx.arc(cx, cy, r, 0, Math.PI * 2); ctx.fill()
        }
        particles.current.forEach(p => {
          p.phase += 0.02
          const a = 0.4 + Math.abs(Math.sin(p.phase)) * 0.6
          ctx.beginPath(); ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2)
          ctx.fillStyle = `rgba(200,220,255,${a})`; ctx.fill()
        })
      }

      if (theme === 'solar') {
        const cx = W * 0.5, cy = H * 0.5
        const pulseR = 14
        for (let ring = 3; ring >= 0; ring--) {
          const g = ctx.createRadialGradient(cx, cy, 0, cx, cy, pulseR + ring * 8)
          g.addColorStop(0, ring === 0 ? '#ffffff' : c1 + 'ee')
          g.addColorStop(0.4, c1 + (ring === 0 ? 'cc' : '66'))
          g.addColorStop(1, 'transparent')
          ctx.fillStyle = g
          ctx.beginPath(); ctx.arc(cx, cy, pulseR + ring * 8, 0, Math.PI * 2); ctx.fill()
        }
        for (let i = 0; i < 6; i++) {
          const angle = t * 0.4 + i * Math.PI / 3
          const len = 10 + Math.sin(t + i) * 4
          const x2 = cx + Math.cos(angle) * (pulseR + len)
          const y2 = cy + Math.sin(angle) * (pulseR + len)
          ctx.strokeStyle = c1 + '88'
          ctx.lineWidth = 1.5
          ctx.beginPath(); ctx.moveTo(cx, cy); ctx.lineTo(x2, y2); ctx.stroke()
        }
      }

      if (theme === 'blackhole') {
        const cx = W * 0.5, cy = H * 0.5
        const coreR = 10
        for (let i = 5; i >= 0; i--) {
          const rx = coreR + 6 + i * 4
          const ry = (coreR + 6 + i * 4) * 0.28
          const alpha = (0.6 - i * 0.08) * 0.7
          ctx.save()
          ctx.translate(cx, cy)
          ctx.rotate(t * 0.5)
          ctx.strokeStyle = i < 2 ? c1 + 'ff' : c2 + Math.floor(alpha * 255).toString(16).padStart(2, '0')
          ctx.lineWidth = i < 2 ? 2 : 1
          ctx.beginPath(); ctx.ellipse(0, 0, rx, ry, 0, 0, Math.PI * 2)
          ctx.stroke(); ctx.restore()
        }
        const cg = ctx.createRadialGradient(cx, cy, 0, cx, cy, coreR)
        cg.addColorStop(0, '#000000'); cg.addColorStop(0.6, '#110008'); cg.addColorStop(1, c1 + '44')
        ctx.fillStyle = cg
        ctx.beginPath(); ctx.arc(cx, cy, coreR, 0, Math.PI * 2); ctx.fill()
      }

      if (theme === 'stars') {
        particles.current.forEach(p => {
          p.x -= p.speed * 0.8
          if (p.x < 0) { p.x = W; p.y = Math.random() * H }
          p.phase += 0.015
          const a = 0.3 + Math.abs(Math.sin(p.phase)) * 0.7
          ctx.beginPath(); ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2)
          ctx.fillStyle = `rgba(180,210,255,${a})`; ctx.fill()
        })
        const g = ctx.createRadialGradient(W * 0.3, H * 0.5, 0, W * 0.3, H * 0.5, W * 0.6)
        g.addColorStop(0, c1 + '22'); g.addColorStop(1, 'transparent')
        ctx.fillStyle = g; ctx.fillRect(0, 0, W, H)
      }

      if (theme === 'clouds') {
        const cloudDefs = [
          { x: 0.2, y: 0.35, r: 18, speed: 0.20 },
          { x: 0.55, y: 0.55, r: 22, speed: 0.15 },
          { x: 0.8, y: 0.3, r: 14, speed: 0.25 },
        ]
        cloudDefs.forEach((cl, i) => {
          const cx = (cl.x + Math.sin(t * cl.speed + i) * 0.15) * W
          const cy = (cl.y + Math.cos(t * cl.speed * 0.15 + i) * 0.16) * H
          const r = cl.r
          for (let b = 3; b >= 0; b--) {
            const g = ctx.createRadialGradient(cx, cy, 0, cx, cy, r + b * 5)
            g.addColorStop(0, `rgba(200,210,255,${0.18 - b * 0.04})`)
            g.addColorStop(1, 'transparent')
            ctx.fillStyle = g
            ctx.beginPath(); ctx.arc(cx, cy, r + b * 5, 0, Math.PI * 2); ctx.fill()
          }
        })
        particles.current.forEach(p => {
          p.phase += 0.008
          const a = 0.2 + Math.abs(Math.sin(p.phase)) * 0.5
          ctx.beginPath(); ctx.arc(p.x, p.y, p.r * 0.8, 0, Math.PI * 2)
          ctx.fillStyle = `rgba(220,225,255,${a})`; ctx.fill()
        })
      }

      rafRef.current = requestAnimationFrame(draw)
    }

    rafRef.current = requestAnimationFrame(draw)
    return () => cancelAnimationFrame(rafRef.current)
  }, [theme, color])

  return <canvas ref={canvasRef} width={56} height={56} className={styles.artCanvas} />
}

function BarVisualizer({ getAudioData, playing, color, onMetrics }) {
  const canvasRef = useRef(null)
  const rafRef = useRef(null)
  const t0 = useRef(Date.now())
  const freqRef = useRef(new Uint8Array(128))

  useEffect(() => {
    const canvas = canvasRef.current
    if (!canvas) return
    const ctx = canvas.getContext('2d')
    const W = canvas.width, H = canvas.height
    const BINS = 28
    const freqData = freqRef.current

    const draw = () => {
      ctx.clearRect(0, 0, W, H)
      const t = (Date.now() - t0.current) / 1000

      if (!getAudioData || !playing) {
        // Idle sine wave
        ctx.strokeStyle = color[0] + '55'
        ctx.lineWidth = 1.5
        ctx.beginPath()
        for (let x = 0; x < W; x++) {
          const y = H / 2 + Math.sin(x * 0.18 + t * 1.2) * 3.5
            + Math.sin(x * 0.08 + t * 0.7) * 2
          x === 0 ? ctx.moveTo(x, y) : ctx.lineTo(x, y)
        }
        ctx.stroke()
        if (onMetrics) onMetrics(null)
        rafRef.current = requestAnimationFrame(draw)
        return
      }

      getAudioData(freqData)
      if (onMetrics) onMetrics(freqData)

      const step = Math.floor(freqData.length / BINS)
      const barW = (W / BINS) - 1

      const grad = ctx.createLinearGradient(0, H, 0, 0)
      grad.addColorStop(0, color[0] + 'dd')
      grad.addColorStop(0.6, color[1] + 'aa')
      grad.addColorStop(1, '#ffffff44')

      for (let i = 0; i < BINS; i++) {
        let sum = 0
        for (let j = 0; j < step; j++) sum += (freqData[i * step + j] || 0)
        const v = (sum / step) / 255
        const barH = Math.max(2, v * H * 1.15)
        const x = i * (barW + 1)
        ctx.fillStyle = grad
        ctx.beginPath()
        if (ctx.roundRect) ctx.roundRect(x, H - barH, barW, barH, 2)
        else ctx.rect(x, H - barH, barW, barH)
        ctx.fill()
      }
      rafRef.current = requestAnimationFrame(draw)
    }

    rafRef.current = requestAnimationFrame(draw)
    return () => cancelAnimationFrame(rafRef.current)
  }, [getAudioData, playing, color, onMetrics])

  return <canvas ref={canvasRef} width={196} height={32} className={styles.barCanvas} />
}

import { useState, useRef, useCallback, useEffect } from 'react'

const SECTION_MUSIC = {
  intro: '/audio/intro.mp3',
  solar: '/audio/solar.mp3',
  phenomena: '/audio/phenomena.mp3',
  deepspace: '/audio/deepspace.mp3',
  ending: '/audio/ending.mp3',
  wishing: '/audio/ending.mp3', // lanjut dari Ending tanpa putus (src sama → tidak crossfade)
  gallery: '/audio/intro.mp3', // hangat & ramah, seperti kembali ke titik berangkat
  default: '/audio/intro.mp3',
}

const FADE_DURATION = 1500
const FADE_STEPS = 30

// Di HP, pointerdown/touchstart bukan gestur yang diakui untuk memulai audio;
// pointerup/touchend/click yang diakui. Semua dipasang sampai audio berhasil main.
const INTERACT_EVENTS = ['pointerdown', 'pointerup', 'touchend', 'keydown', 'click']

// Level fade saat ini (0..1), dibaca dari volume elemen audio
const fadeLevel = (audio, vol) => (vol > 0 ? Math.max(0, Math.min(1, audio.volume / vol)) : 0)

export function useAmbientAudio() {
  const [playing, setPlaying] = useState(false)
  const [volume, setVolumeState] = useState(0.5)
  const [audioProgress, setAudioProgress] = useState({ currentTime: 0, duration: 0, progress: 0 })

  const audioRef = useRef(null)
  const currentSrc = useRef(null)
  const volumeRef = useRef(0.5)

  const isPlaying = useRef(false)
  const manualPause = useRef(false) // Mencegah autoplay menyala jika user sudah sengaja pause
  const initAutoplay = useRef(false)
  const removeInteract = useRef(null)

  const ctxRef = useRef(null)
  const analyserRef = useRef(null)
  const sourceRef = useRef(null)
  const gainNodeRef = useRef(null)

  const getCtx = useCallback(() => {
    if (!ctxRef.current) {
      const ctx = new (window.AudioContext || window.webkitAudioContext)()
      const analyser = ctx.createAnalyser()
      analyser.fftSize = 256
      analyser.smoothingTimeConstant = 0.8
      const gain = ctx.createGain()
      gain.gain.value = volumeRef.current
      analyser.connect(gain)
      gain.connect(ctx.destination)
      ctxRef.current = ctx
      analyserRef.current = analyser
      gainNodeRef.current = gain
    }
    return ctxRef.current
  }, [])

  // Hubungkan elemen audio ke analyser (sekali per elemen). Audio usang diabaikan.
  // Source lama TIDAK diputus di sini: ia tetap terdengar selama fade-out, lalu diputus oleh retire().
  const connectAudio = useCallback((audio) => {
    if (audioRef.current !== audio) return
    try {
      const ctx = getCtx()
      if (ctx.state === 'suspended') ctx.resume()
      if (!audio._node) {
        audio._node = ctx.createMediaElementSource(audio)
        audio._node.connect(analyserRef.current)
      }
      sourceRef.current = audio._node
    } catch (_) { }
  }, [getCtx])

  const getAudioData = useCallback((targetArray) => {
    if (analyserRef.current && targetArray) {
      analyserRef.current.getByteFrequencyData(targetArray)
      return true
    }
    return false
  }, [])

  // Satu timer fade per elemen: fade baru membatalkan fade lama pada elemen yang sama
  const fadeTo = useCallback((audioEl, from, to, onDone) => {
    if (!audioEl) return
    clearInterval(audioEl._fade)
    let current = from
    const step = (to - from) / FADE_STEPS
    const ms = FADE_DURATION / FADE_STEPS
    audioEl._fade = setInterval(() => {
      current += step
      const clamped = Math.max(0, Math.min(1, current))
      try { audioEl.volume = clamped * volumeRef.current } catch (_) { }
      const done = step > 0 ? current >= to : current <= to
      if (done) {
        clearInterval(audioEl._fade)
        audioEl._fade = 0
        onDone?.()
      }
    }, ms)
  }, [])

  // Pensiunkan elemen audio: fade-out, lalu bebaskan (tanpa src='' agar tidak memicu error)
  const retire = useCallback((audio) => {
    if (!audio || audio._retired) return
    audio._retired = true
    fadeTo(audio, fadeLevel(audio, volumeRef.current), 0, () => {
      audio.pause()
      audio.onerror = null
      audio.ontimeupdate = null
      audio.removeAttribute('src')
      try { audio.load() } catch (_) { }
      try { audio._node?.disconnect() } catch (_) { }
    })
  }, [fadeTo])

  // Mengembalikan Promise (menolak jika browser memblokir atau play dibatalkan)
  const playSrc = useCallback((src) => {
    const cur = audioRef.current

    // Track yang sama sudah ada: lanjutkan (membatalkan fade-out yang belum selesai)
    if (cur && cur._key === src && !cur._retired) {
      const p = cur.paused ? cur.play() : Promise.resolve()
      return p.then(() => {
        if (audioRef.current !== cur) return
        connectAudio(cur)
        fadeTo(cur, fadeLevel(cur, volumeRef.current), 1)
      })
    }

    retire(cur)
    setAudioProgress({ currentTime: 0, duration: 0, progress: 0 })

    const audio = new Audio(src)
    audio.loop = true
    audio.volume = 0
    audio.crossOrigin = 'anonymous'
    audio.preload = 'auto'
    audio._key = src

    audio.onerror = () => {
      if (audioRef.current === audio) {
        console.warn(`[useAudio] Gagal load: ${src} — pastikan file ada di public/audio/`)
      }
    }

    // Update progres paling sering 1x per detik (UI hanya menampilkan detik)
    let lastSec = -1
    audio.ontimeupdate = () => {
      if (audioRef.current !== audio) return
      const sec = Math.floor(audio.currentTime)
      if (sec === lastSec) return
      lastSec = sec
      setAudioProgress({
        currentTime: audio.currentTime,
        duration: audio.duration || 0,
        progress: audio.duration ? (audio.currentTime / audio.duration) : 0,
      })
    }

    audioRef.current = audio
    currentSrc.current = src

    return audio.play().then(() => {
      if (audioRef.current !== audio) return // sudah digantikan selagi menunggu
      connectAudio(audio)
      fadeTo(audio, 0, 1)
    })
  }, [fadeTo, connectAudio, retire])

  // Logika Autoplay di percobaan pertama web load
  useEffect(() => {
    if (initAutoplay.current) return
    initAutoplay.current = true

    let trying = false
    const remove = () => {
      INTERACT_EVENTS.forEach((ev) => window.removeEventListener(ev, onInteract))
      removeInteract.current = null
    }
    const onInteract = (e) => {
      if (manualPause.current || isPlaying.current || trying) return
      // Ketukan pada MusicPlayer ditangani tombolnya sendiri (hindari mulai lalu langsung pause)
      if (e?.target?.closest?.('[class*="player"],[class*="showBtn"]')) return
      trying = true
      playSrc(currentSrc.current || SECTION_MUSIC.default)
        .then(() => {
          isPlaying.current = true
          setPlaying(true)
          remove()
        })
        .catch(() => { /* masih diblokir: tunggu gestur berikutnya */ })
        .finally(() => { trying = false })
    }

    const first = playSrc(SECTION_MUSIC.default)
    const a = audioRef.current
    first
      .then(() => {
        if (audioRef.current !== a) return
        isPlaying.current = true
        setPlaying(true)
      })
      .catch(() => {
        if (audioRef.current !== a) return // percobaan usang (mis. StrictMode)
        INTERACT_EVENTS.forEach((ev) => window.addEventListener(ev, onInteract))
        removeInteract.current = remove
      })
  }, [playSrc])

  const toggleMusic = useCallback(() => {
    if (!isPlaying.current) {
      manualPause.current = false // User membatalkan pause
      playSrc(currentSrc.current || SECTION_MUSIC.default)
        .then(() => {
          isPlaying.current = true
          setPlaying(true)
        })
        .catch(() => { })
    } else {
      manualPause.current = true // User menyengaja pause
      const audio = audioRef.current
      if (audio) fadeTo(audio, fadeLevel(audio, volumeRef.current), 0, () => audio.pause())
      isPlaying.current = false
      setPlaying(false)
    }
  }, [playSrc, fadeTo])

  const changeSection = useCallback((sectionId) => {
    const src = SECTION_MUSIC[sectionId] ?? SECTION_MUSIC.default
    if (src === currentSrc.current) return

    if (isPlaying.current) {
      playSrc(src).catch(() => { })
    } else {
      currentSrc.current = src
    }
  }, [playSrc])

  const setVolume = useCallback((val) => {
    const v = Math.max(0, Math.min(1, val))
    const prev = volumeRef.current
    setVolumeState(v)
    volumeRef.current = v
    const a = audioRef.current
    if (a && !a.paused) {
      // pertahankan level fade yang sedang berjalan
      const level = prev > 0 ? fadeLevel(a, prev) : 1
      a.volume = level * v
    }
    if (gainNodeRef.current) gainNodeRef.current.gain.value = v
  }, [])

  useEffect(() => {
    return () => {
      removeInteract.current?.()
      initAutoplay.current = false // StrictMode: jalankan ulang autoplay setelah mount ulang
      const a = audioRef.current
      if (a) {
        clearInterval(a._fade)
        a.pause()
        a.onerror = null
        a.ontimeupdate = null
        a.removeAttribute('src')
        try { a.load() } catch (_) { }
      }
      audioRef.current = null
      currentSrc.current = null
      try { ctxRef.current?.close() } catch (_) { }
      ctxRef.current = null
      analyserRef.current = null
      gainNodeRef.current = null
      sourceRef.current = null
    }
  }, [])

  return { playing, toggleMusic, volume, setVolume, changeSection, getAudioData, audioProgress }
}

// ============================================================
// COSMIC JOURNEY — Synthesized sound effects (no audio files)
// Usage: import { sfx } from '../hooks/useSfx.js'  →  sfx.chime(2)
// Must be called from a user gesture (click/keypress) the first time.
// ============================================================

let ctx = null
let master = null
let noiseBuf = null
let muted = false
try { muted = localStorage.getItem('cj-sfx-muted') === '1' } catch (_) { }

const SCALE = [523.25, 587.33, 659.25, 783.99, 880.0, 987.77, 1174.66, 1318.51] // C major pentatonic-ish

function getCtx() {
  if (muted) return null
  if (!ctx) {
    const AC = window.AudioContext || window.webkitAudioContext
    if (!AC) return null
    ctx = new AC()
    master = ctx.createGain()
    master.gain.value = 0.35
    master.connect(ctx.destination)
  }
  if (ctx.state === 'suspended') ctx.resume()
  return ctx
}

function getNoise(c) {
  if (!noiseBuf) {
    noiseBuf = c.createBuffer(1, c.sampleRate * 2, c.sampleRate)
    const d = noiseBuf.getChannelData(0)
    for (let i = 0; i < d.length; i++) d[i] = Math.random() * 2 - 1
  }
  return noiseBuf
}

// Mengembalikan gain node agar suara bisa dihentikan lebih awal (lihat sfx.open)
function tone({ freq, to, type = 'sine', dur = 0.4, vol = 0.2, delay = 0, attack = 0.01 }) {
  const c = getCtx()
  if (!c) return null
  const t = c.currentTime + delay
  const o = c.createOscillator()
  const g = c.createGain()
  o.type = type
  o.frequency.setValueAtTime(freq, t)
  if (to) o.frequency.exponentialRampToValueAtTime(to, t + dur)
  g.gain.setValueAtTime(0.0001, t)
  g.gain.exponentialRampToValueAtTime(vol, t + Math.min(attack, dur * 0.5))
  g.gain.exponentialRampToValueAtTime(0.0001, t + dur)
  o.connect(g)
  g.connect(master)
  o.start(t)
  o.stop(t + dur + 0.05)
  return g
}

function noise({ dur = 1, vol = 0.2, from = 300, to = 3000, q = 1, delay = 0, swell = 0.5 }) {
  const c = getCtx()
  if (!c) return null
  const t = c.currentTime + delay
  const src = c.createBufferSource()
  src.buffer = getNoise(c)
  src.loop = true
  const f = c.createBiquadFilter()
  f.type = 'bandpass'
  f.Q.value = q
  f.frequency.setValueAtTime(from, t)
  f.frequency.exponentialRampToValueAtTime(to, t + dur)
  const g = c.createGain()
  g.gain.setValueAtTime(0.0001, t)
  g.gain.exponentialRampToValueAtTime(vol, t + dur * swell)
  g.gain.exponentialRampToValueAtTime(0.0001, t + dur)
  src.connect(f)
  f.connect(g)
  g.connect(master)
  src.start(t)
  src.stop(t + dur + 0.05)
  return g
}

function makeMoonTex() {
  const c = document.createElement('canvas')
  c.width = 128; c.height = 64
  const x = c.getContext('2d')
  x.fillStyle = '#9a9a9a'
  x.fillRect(0, 0, 128, 64)
  for (let i = 0; i < 40; i++) {
    x.fillStyle = `rgba(${Math.random() < 0.5 ? '60,60,60' : '210,210,210'},${Math.random() * 0.35})`
    x.beginPath()
    x.arc(Math.random() * 128, Math.random() * 64, Math.random() * 7 + 1, 0, 6.3)
    x.fill()
  }
  return new THREE.CanvasTexture(c)
}

function OrbitLine({ r, color, opacity }) {

}

function Earth3D({ isWarping, warpStartTime, travelMs, holdRef }) {
  const root = useRef()
  const earth = useRef()
  const moon = useRef()
  const iss = useRef()
  const colorMap = useLoader(THREE.TextureLoader, '/images/earth.jpg')
  const moonTex = useMemo(makeMoonTex, [])

  useEffect(() => {
    if (colorMap) {
      colorMap.wrapS = THREE.RepeatWrapping
      colorMap.wrapT = THREE.RepeatWrapping
      colorMap.anisotropy = 16
    }
  }, [colorMap])

  useFrame((state, delta) => {
    const p = holdRef.current
    const t = isWarping ? Math.min((Date.now() - warpStartTime) / travelMs, 1) : 0
    const el = state.clock.elapsedTime

    // Getar halus saat ditahan, mereda begitu perjalanan mulai
    if (root.current) {
      const a = p * p * 0.06 * (1 - Math.min(t * 2, 1))
      root.current.position.set((Math.random() - 0.5) * a, (Math.random() - 0.5) * a, 0)
    }
    if (earth.current) earth.current.rotation.y += (0.09 + p * 1.2 + t * t * 1.5) * delta

    if (moon.current) {
      const a = el * 0.18
      moon.current.position.set(Math.cos(a) * MOON_R, 0, Math.sin(a) * MOON_R)
    }
    if (iss.current) {
      const a = el * 0.9
      iss.current.position.set(Math.cos(a) * ISS_R, 0, Math.sin(a) * ISS_R)
    }
  })

  return (
    <group ref={root}>
      <group ref={earth}>
        <mesh>
          <sphereGeometry args={[2.18, 64, 64]} />
          <meshBasicMaterial color="#020813" />
        </mesh>
        <mesh>
          <sphereGeometry args={[2.2, 64, 64]} />
          <meshStandardMaterial map={colorMap} roughness={0.8} metalness={0.2} alphaTest={0.5} />
        </mesh>
        <mesh>
          <sphereGeometry args={[2.25, 64, 64]} />
          <meshStandardMaterial color="#6ec5ff" transparent opacity={0.15} depthWrite={false} side={THREE.FrontSide} />
        </mesh>
      </group>

      {/* Bulan */}
      <group rotation={[1.1, 0, -0.2]}>
        <OrbitLine r={MOON_R} color="#8fb0ff" opacity={0.15} />
        <mesh ref={moon}>
          <sphereGeometry args={[0.32, 32, 32]} />
          <meshStandardMaterial map={moonTex} roughness={1} />
        </mesh>
      </group>

      {/* ISS (inklinasi ±51,6°, orbit cepat) */}

const wib = () =>

function Clock() {
  const [t, setT] = useState(wib)
  useEffect(() => {
    const id = setInterval(() => setT(wib()), 1000)
    return () => clearInterval(id)
  }, [])

}

function Intro({ onNext, isWarping = false, warpStartTime = 0, travelMs = 2800 }) {
  const holdRef = useRef(0) // 0..1, dibaca oleh Three.js tanpa re-render
  const wrapRef = useRef(null)
  const btnRef = useRef(null)
  const distRef = useRef(null)
  const st = useRef({ holding: false, last: 0, raf: 0, stop: null })
  const nextRef = useRef(onNext)
  nextRef.current = onNext

  const tick = useCallback((now) => {
    const s = st.current
    const dt = Math.min(now - s.last, 50)
    s.last = now
    const p = Math.min(1, Math.max(0, holdRef.current + (s.holding ? dt / HOLD_MS : -dt / RELEASE_MS)))
    holdRef.current = p

    btnRef.current?.style.setProperty('--p', p)
    if (distRef.current) {
      distRef.current.textContent = `${((MOON_KM * (1 - p)) / 1e5).toFixed(3)} × 10⁵ km from home`
    }

    if (s.holding && p >= 1) {
      // Penuh → berangkat. Suara dibiarkan selesai sendiri.
      s.holding = false
      s.stop = null
      s.raf = 0
      nextRef.current()
      return
    }
    s.raf = p > 0 || s.holding ? requestAnimationFrame(tick) : 0
  }, [])

  const start = () => {
    const s = st.current
    if (s.holding || isWarping || holdRef.current >= 1) return
    s.holding = true
    s.last = performance.now()
    if (!s.stop) s.stop = sfx.open()
    if (!s.raf) s.raf = requestAnimationFrame(tick)
  }

  const end = () => {
    const s = st.current
    if (!s.holding) return
    s.holding = false
    s.stop?.()
    s.stop = null
  }

  useEffect(() => () => {
    cancelAnimationFrame(st.current.raf)
    st.current.stop?.()
  }, [])

  // Saat berangkat: Bumi bergeser & mengecil ke posisi Bumi di section Solar System
  useLayoutEffect(() => {
    const el = wrapRef.current
    if (!isWarping || !el) return
    const r = el.getBoundingClientRect()
    const vw = window.innerWidth
    const vh = window.innerHeight
    const mobile = vw < 768
    const camZ = mobile ? 34 : 26                                   // sama dgn CameraRig overview
    const unit = vh / (2 * camZ * Math.tan(THREE.MathUtils.degToRad(20))) // px per unit scene Solar
    const posX = mobile ? -2.1 : 2.5                                // posisi Earth (index 2) di grid
    const posY = mobile ? 1.6 : 3
    const tx = vw / 2 + posX * unit - (r.left + r.width / 2)
    const ty = vh / 2 - posY * unit - (r.top + r.height / 2)
    const s = ((68 / 65) * 2 * unit) / EARTH_PX                     // radius Earth = 68/65
    el.style.transitionDuration = `${travelMs}ms`
    el.style.transform = `translate(${tx}px, ${ty}px) scale(${s})`
  }, [isWarping, travelMs])

  return (
    <section id="intro" className={`cosmic-section ${styles.intro}`}>
      <div className="noise-overlay" />
      <div className={styles.nebulaBg} />
      <div
        className={`${styles.sunGlow} ${isWarping ? styles.sunGlowOn : ''}`}
        style={{ transitionDuration: `${travelMs}ms` }}
      />

      <div className={`${styles.hud} ${isWarping ? styles.hudHidden : ''}`}>
        <p className={styles.hudLabel}>WIB · GMT+7 · Indonesia Barat</p>
        <Clock />
        <div className={styles.hudIss}>
          <span className={styles.issDot} />
          <span className={styles.hudLabel}>ISS · ±408 km · orbit 92 min</span>
        </div>
      </div>

      <div ref={wrapRef} className={`${styles.earthWrap} ${isWarping ? styles.earthWrapWarp : ''}`}>
        <div className={styles.earthFloat}>
          <div className={styles.earthGlow} />
          <div className={styles.atmosphereRing} />
          <Canvas className={styles.earthCanvas} dpr={[1, 1.5]} camera={{ position: [0, 0, 15.8], fov: 45 }}>
            <ambientLight intensity={0.05} />
            <directionalLight position={[5, 4, 2]} intensity={2.5} color="#fff4e6" />
            <Suspense fallback={null}>
              <Earth3D isWarping={isWarping} warpStartTime={warpStartTime} travelMs={travelMs} holdRef={holdRef} />
            </Suspense>
          </Canvas>
        </div>
      </div>

      <div className={`${styles.content} ${isWarping ? styles.contentHidden : ''}`}>
        <p ref={distRef} className={`t-label ${styles.coords}`}>3.844 × 10⁵ km from home</p>
        <h1 className={styles.heading}>
          You are leaving<br /><em>home...</em>
        </h1>
        <p className={styles.sub}>
          A journey through 13.8 billion years of cosmic history awaits.
        </p>
        <button
          ref={btnRef}
          className={styles.beginBtn}
          onPointerDown={(e) => { e.currentTarget.setPointerCapture?.(e.pointerId); start() }}
          onPointerUp={end}
          onPointerCancel={end}
          onBlur={end}
          onKeyDown={(e) => { if ((e.key === ' ' || e.key === 'Enter') && !e.repeat) { e.preventDefault(); start() } }}
          onKeyUp={(e) => { if (e.key === ' ' || e.key === 'Enter') end() }}
          onContextMenu={(e) => e.preventDefault()}
        >
          Begin the Journey <span className={styles.arrow}>↓</span>
        </button>
        <p className={styles.holdHint}>Press &amp; hold to depart</p>
      </div>
    </section>
  )
}
