import React, { useState, useCallback, useEffect, useRef, Suspense } from 'react'

// Lazy loading sections (kode three.js tidak akan diload di awal)
const Intro = React.lazy(() => import('./sections/Intro.jsx'))
const SolarSystem = React.lazy(() => import('./sections/SolarSystem.jsx'))
const Phenomena = React.lazy(() => import('./sections/Phenomena.jsx'))
const DeepSpace = React.lazy(() => import('./sections/DeepSpace.jsx'))
const Ending = React.lazy(() => import('./sections/Ending.jsx'))
const Wishing = React.lazy(() => import('./sections/Wishing.jsx'))
const SecretEnding = React.lazy(() => import('./sections/SecretEnding.jsx'))
const Gallery = React.lazy(() => import('./sections/Gallery.jsx'))
// HomeWarp memuat three.js, jadi lazy; dipanggil dari section mana pun lewat handleReturnHome
const HomeWarp = React.lazy(() => import('./components/HomeWarp.jsx'))

import { useAmbientAudio } from './hooks/useAudio.js'
import { sfx } from './hooks/useSfx.js'
import { UNIVERSE_MESSAGES } from './data/index.js'
import { resetSessionCallsign } from './services/wishes.js'

// Cadangan bila chunk gerbang gagal diunduh (mis. offline): lanjut tanpa gerbang dan tanpa suara,
// supaya pengunjung tidak terjebak di layar kosong.
function GateUnavailable({ onEnter, onExited }) {
  useEffect(() => {
    onEnter(false)
    onExited()
  }, [onEnter, onExited])
  return null
}

// Gerbang masuk dimuat lazy; hanya tampil sekali per kunjungan
const EntryGate = React.lazy(() =>
  import('./components/EntryGate.jsx').catch(() => ({ default: GateUnavailable }))
)

// Penutup polos selama chunk gerbang diunduh (mencegah Intro berkedip di bawahnya)
const GATE_FALLBACK = { position: 'fixed', inset: 0, zIndex: 200, background: 'var(--void)' }

// Petunjuk kecil bila suara tertahan (AudioContext suspended / interupsi / mode senyap iOS)
const HINT_WRAP = {
  position: 'fixed',
  top: 'max(0.75rem, env(safe-area-inset-top))',
  left: 0,
  right: 0,
  zIndex: 150,
  display: 'flex',
  justifyContent: 'center',
  padding: '0 1rem',
  pointerEvents: 'none',
}
const HINT_BTN = {
  pointerEvents: 'auto',
  minHeight: '2.75rem',
  padding: '0.6rem 1.1rem',
  borderRadius: '1.25rem',
  border: '1px solid rgba(120, 160, 255, 0.5)',
  background: 'rgba(8, 8, 36, 0.92)',
  color: 'var(--text-primary)',
  fontSize: 'var(--fs-label)',
  fontWeight: 400,
  letterSpacing: '0.02em',
  lineHeight: 1.45,
  textAlign: 'center',
  boxShadow: '0 0 24px rgba(68, 102, 255, 0.25)',
  touchAction: 'manipulation',
}

// 'secret' (Secret Ending) SENGAJA tidak masuk daftar ini: bukan bagian navigasi bebas,
// hanya terbuka lewat submit wish yang sukses. NavDots/ProgressBar memakai 'wishing' saat di 'secret'.
const SECTION_IDS = ['intro', 'solar', 'phenomena', 'deepspace', 'ending', 'wishing', 'gallery']

// Kedalaman starfield per section (0 = terang penuh, 1 = paling redup).
// Semua id section, termasuk 'secret'. Id yang tidak ada di peta jatuh ke 0.
const SECTION_DEPTH = {
  intro: 0,
  solar: 0.2,
  phenomena: 0.4,
  wishing: 0.45,
  ending: 0.55,
  deepspace: 0.7,
  gallery: 0.8,
  secret: 0.8,
}

// Mode Comfort (UI-nya Fase 12): geser satu tingkat lebih redup
const COMFORT_DEPTH_STEP = 0.15

// Section tersembunyi yang memakai musik section lain (tanpa crossfade putus).
// Juga dipakai untuk label MusicPlayer, supaya 'secret' tidak jatuh ke track default.
const MUSIC_ALIAS = { secret: 'wishing' }

// Dari section ini ke depan, chunk section akhir + HomeWarp diunduh di latar belakang
const PRELOAD_FROM = new Set(['deepspace', 'ending', 'wishing', 'secret', 'gallery'])

// State pengunjung (in-memory saja, sengaja TIDAK disimpan di localStorage/server)
// Diisi Ending saat reveal, dibaca Wishing, Secret Ending & Gallery. Direset setelah pulang ke Bumi.
const INITIAL_VISITOR = {
  name: '',                // nama penerima konstelasi
  sender: '',              // pengirim (hanya mode "For Someone Else")
  constellationPoints: [],
  finalMessage: '',
  coords: '',
  wishSubmitted: false,
  mySubmission: null,      // data kiriman wish pengunjung ini (untuk highlight "Your wish")
  skipped: false,          // true = Wishing di-skip langsung ke Gallery (tidak membuka Secret Ending)
  secretSeen: false,       // true = sinematik Secret Ending sudah selesai/di-skip (kunjungan ulang tampil langsung)
}

const HOME_END_PAD = 800 // jeda setelah pindah ke intro sebelum overlay HomeWarp dilepas
// (HomeWarp.jsx: T_LEAVE bergantung pada angka ini, lihat catatan di sana)

// Perjalanan Bumi (intro > solar): Bumi menjauh & mengecil ke posisinya di tata surya
const TRAVEL_MS = 2800 // dari tombol penuh sampai section berganti
const ARRIVE_MS = 1000 // jeda sampai UI global muncul lagi

// Transisi gelap-terang untuk perpindahan section selain intro > solar
const FADE_OUT = 700
const FADE_IN = 800

export default function App() {
  const [fact, setFact] = useState(null)
  const [message, setMessage] = useState(null)

  // Gerbang masuk: 'show' > 'leaving' (memudar) > 'done'. Tidak pernah kembali ke 'show'
  // dalam satu kunjungan, jadi HomeWarp / "Return home" tidak memunculkannya lagi.
  const [gate, setGate] = useState('show')
  const mainRef = useRef(null)

  const [activeSection, setActiveSection] = useState('intro')
  const [renderedSection, setRenderedSection] = useState('intro')
  const [fadeState, setFadeState] = useState('visible')
  const navTimer = useRef(0) // timer pergantian section (dibatalkan jika ada perpindahan baru)

  // Perjalanan intro > solar: 'idle' > 'warp' > 'idle'
  const [warp, setWarp] = useState({ phase: 'idle', startTime: 0 })
  const warpingRef = useRef(false)

  const [fadePhase, setFadePhase] = useState('idle')

  const [visitor, setVisitor] = useState(INITIAL_VISITOR)
  const updateVisitor = useCallback((patch) => setVisitor((v) => ({ ...v, ...patch })), [])
  // true = Ending langsung membuka pesan konstelasi (dari Secret Ending: "Stay with my constellation")
  const [resumeMessage, setResumeMessage] = useState(false)

  const [homeWarping, setHomeWarping] = useState(false)
  const homeTimers = useRef([])
  // true = HomeWarp sudah memberi tahu bahwa Bumi mendekat: bintang kembali terang (depth 0)
  const [homeBright, setHomeBright] = useState(false)
  const brightenStars = useCallback(() => setHomeBright(true), [])

  // Mode Comfort: hook saja, belum ada UI (Fase 12 akan memanggil setComfort).
  // Efek: starfield satu tingkat lebih redup + class body.comfort-mode (kontras teks naik).
  const [comfort] = useState(false)

  const {
    playing,
    toggleMusic,
    volume,
    setVolume,
    changeSection,
    getAudioData,
    audioProgress,
    startAudio,
    blocked: audioBlocked,
    resumeAudio,
  } = useAmbientAudio()

  // Ketukan gerbang. SEMUA yang butuh gestur (AudioContext sfx, musik) harus dipanggil
  // sinkron di sini, sebelum await / setTimeout apa pun, supaya iOS mengizinkannya.
  const handleGateEnter = useCallback((withSound) => {
    sfx.setMuted(!withSound)
    if (withSound) sfx.unlock()
    startAudio(withSound)
    setGate((g) => (g === 'show' ? 'leaving' : g))
  }, [startAudio])

  const handleGateExited = useCallback(() => setGate('done'), [])

  const handleNavigate = useCallback((id) => {
    setFact(null)
    setMessage(null)
    if (id === activeSection) return
    if (id !== 'ending') setResumeMessage(false)
    setActiveSection(id)
    setFadeState('hidden')
    changeSection(MUSIC_ALIAS[id] ?? id)

    // Perpindahan cepat beruntun: hanya tujuan terakhir yang dirender (tanpa kilatan section perantara)
    clearTimeout(navTimer.current)
    navTimer.current = setTimeout(() => {
      setRenderedSection(id)
      setFadeState('visible')
    }, 400)
  }, [activeSection, changeSection])

  // NavDots: diabaikan selama transisi / HomeWarp (tombol tetap bisa dijangkau keyboard walau tersembunyi)
  const handleNavDot = useCallback((id) => {
    if (warpingRef.current) return
    handleNavigate(id)
  }, [handleNavigate])

  // Dipanggil Intro saat tombol tahan penuh: Intro menganimasikan Bumi, App mengganti section
  const handleWarpNavigate = useCallback((id) => {
    if (warpingRef.current || id === activeSection) return
    warpingRef.current = true
    setFact(null)
    setMessage(null)
    setWarp({ phase: 'warp', startTime: Date.now() })

    setTimeout(() => {
      handleNavigate(id)
      setTimeout(() => {
        setWarp({ phase: 'idle', startTime: 0 })
        warpingRef.current = false
      }, ARRIVE_MS)
    }, TRAVEL_MS)
  }, [activeSection, handleNavigate])

  const handleFadeNavigate = useCallback((id) => {
    if (warpingRef.current || id === activeSection) return
    warpingRef.current = true
    setFact(null)
    setMessage(null)
    setFadePhase('dark')

    setTimeout(() => {
      handleNavigate(id)
      setFadePhase('light')

      setTimeout(() => {
        setFadePhase('idle')
        warpingRef.current = false
      }, FADE_IN)
    }, FADE_OUT)
  }, [activeSection, handleNavigate])

  // Pulang ke Bumi (HomeWarp): bisa dipanggil dari section mana pun (mis. Secret Ending).
  // Tidak menyentuh state gerbang, jadi gerbang tidak muncul lagi.
  const handleReturnHome = useCallback(async () => {
    if (warpingRef.current) return
    warpingRef.current = true
    setFact(null)
    setMessage(null)
    setHomeBright(false)
    try {
      const { HOME_MS } = await import('./components/HomeWarp.jsx')
      setHomeWarping(true)
      const t1 = setTimeout(() => {
        handleNavigate('intro')
        const t2 = setTimeout(() => {
          setHomeWarping(false)
          setHomeBright(false)        // Intro sudah depth 0, jadi tidak ada lompatan visual
          setVisitor(INITIAL_VISITOR) // perjalanan baru = pengunjung baru
          resetSessionCallsign()      // callsign anonim baru untuk pengunjung berikutnya
          warpingRef.current = false
        }, HOME_END_PAD)
        homeTimers.current.push(t2)
      }, HOME_MS)
      homeTimers.current.push(t1)
    } catch (_) {
      warpingRef.current = false
    }
  }, [handleNavigate])

  // Secret Ending > "Stay with my constellation"
  const handleBackToConstellation = useCallback(() => {
    setResumeMessage(true)
    handleFadeNavigate('ending')
  }, [handleFadeNavigate])

  // Wishing: wish berhasil terkirim dan animasi bintang naik selesai > Secret Ending
  const handleWishSubmitted = useCallback(() => {
    handleFadeNavigate('secret')
  }, [handleFadeNavigate])

  useEffect(() => () => {
    homeTimers.current.forEach(clearTimeout)
    clearTimeout(navTimer.current)
  }, [])

  // Selama gerbang tampil, konten di belakangnya tidak bisa difokus keyboard / pembaca layar
  useEffect(() => {
    const el = mainRef.current
    if (!el) return
    if (gate === 'show') el.setAttribute('inert', '')
    else el.removeAttribute('inert')
  }, [gate])

  const transitioning = warp.phase !== 'idle' || fadePhase !== 'idle' || homeWarping
  const hideUi = transitioning || gate !== 'done'
  useEffect(() => {
    document.body.classList.toggle('hide-global-ui', hideUi)
    return () => document.body.classList.remove('hide-global-ui')
  }, [hideUi])

  // Hook Comfort: kontras teks naik lewat token di index.css
  useEffect(() => {
    document.body.classList.toggle('comfort-mode', comfort)
    return () => document.body.classList.remove('comfort-mode')
  }, [comfort])

  // Preload Solar System supaya perpindahan dari intro tidak tersendat
  useEffect(() => {
    const t = setTimeout(() => import('./sections/SolarSystem.jsx'), 1500)
    return () => clearTimeout(t)
  }, [])

  // Preload section akhir + HomeWarp begitu pengunjung mencapai Deep Space,
  // supaya layar tidak kosong saat pindah ke Ending / Wishing / Secret / Gallery / pulang.
  const preloaded = useRef(false)
  useEffect(() => {
    if (preloaded.current || !PRELOAD_FROM.has(activeSection)) return
    const t = setTimeout(() => {
      preloaded.current = true
      const quiet = () => { }
      import('./sections/Ending.jsx').catch(quiet)
      import('./sections/Wishing.jsx').catch(quiet)
      import('./sections/SecretEnding.jsx').catch(quiet)
      import('./sections/Gallery.jsx').catch(quiet)
      import('./components/HomeWarp.jsx').catch(quiet)
    }, 1200)
    return () => clearTimeout(t)
  }, [activeSection])

  const showFact = useCallback((d) => setFact(d), [])
  const closeFact = useCallback(() => setFact(null), [])

  const showMessage = useCallback(() => {
    setMessage(UNIVERSE_MESSAGES[Math.floor(Math.random() * UNIVERSE_MESSAGES.length)])
  }, [])
  const closeMessage = useCallback(() => setMessage(null), [])

  // Secret Ending tidak punya titik sendiri di NavDots/ProgressBar, jadi ikut 'wishing'
  const navSection = activeSection === 'secret' ? 'wishing' : activeSection
  const currentIdx = SECTION_IDS.indexOf(navSection)
  const progress = Math.max(0, currentIdx / (SECTION_IDS.length - 1))

  // Depth starfield mengikuti activeSection (berubah saat navigasi dimulai; StarField
  // meng-easing selama ~2 dtk sehingga tumpang tindih dengan fade section, tanpa lompatan).
  const baseDepth = SECTION_DEPTH[activeSection] ?? 0
  const starDepth = homeBright
    ? 0
    : Math.min(1, baseDepth + (comfort ? COMFORT_DEPTH_STEP : 0))

  return (
    <>
      <StarField count={500} mouseParallax depth={starDepth} />
      <ProgressBar progress={progress} />
      <NavDots active={navSection} onNavigate={handleNavDot} />

      <MusicPlayer
        playing={playing}
        onToggle={toggleMusic}
        volume={volume}
        onVolumeChange={setVolume}
        activeSection={MUSIC_ALIAS[activeSection] ?? activeSection}
        getAudioData={getAudioData}
        audioProgress={audioProgress}
      />

      <div
        className={`fade-overlay ${fadePhase === 'dark' ? 'fade-dark' : ''} ${fadePhase === 'light' ? 'fade-light' : ''}`}
        style={{ '--fade-out': `${FADE_OUT}ms`, '--fade-in': `${FADE_IN}ms` }}
        aria-hidden="true"
      />

      <main ref={mainRef} className={`section-wrapper ${fadeState}`}>
        <Suspense fallback={null}>
          {renderedSection === 'intro' && (
            <Intro
              onNext={() => handleWarpNavigate('solar')}
              isWarping={warp.phase !== 'idle'}
              warpStartTime={warp.startTime}
              travelMs={TRAVEL_MS}
            />
          )}
          {renderedSection === 'solar' && (
            <SolarSystem onFact={showFact} onNext={() => handleFadeNavigate('phenomena')} />
          )}
          {renderedSection === 'phenomena' && (
            <Phenomena onFact={showFact} onNext={() => handleFadeNavigate('deepspace')} />
          )}
          {renderedSection === 'deepspace' && (
            <DeepSpace onFact={showFact} onNext={() => handleFadeNavigate('ending')} />
          )}
          {renderedSection === 'ending' && (
            <Ending
              visitor={visitor}
              onReveal={updateVisitor}
              onNext={() => handleFadeNavigate('wishing')}
              onBackToChoices={() => handleFadeNavigate('secret')}
              resumeMessage={resumeMessage}
            />
          )}
          {renderedSection === 'wishing' && (
            <Wishing
              visitor={visitor}
              onVisitorChange={updateVisitor}
              onGallery={() => handleFadeNavigate('gallery')}
              onSubmitted={handleWishSubmitted}
              onBackToConstellation={handleBackToConstellation}
              onReturnHome={handleReturnHome}
            />
          )}
          {renderedSection === 'secret' && (
            <SecretEnding
              visitor={visitor}
              onVisitorChange={updateVisitor}
              onStay={handleBackToConstellation}
              onGallery={() => handleFadeNavigate('gallery')}
              onReturnHome={handleReturnHome}
            />
          )}
          {renderedSection === 'gallery' && (
            <Gallery
              visitor={visitor}
              onBackToConstellation={handleBackToConstellation}
              onReturnHome={handleReturnHome}
            />
          )}
        </Suspense>
      </main>

      {homeWarping && (
        <Suspense fallback={null}>
          <HomeWarp
            active
            name={visitor.name}
            points={visitor.constellationPoints}
            onBrighten={brightenStars}
          />
        </Suspense>
      )}

      <FactPopup fact={fact} onClose={closeFact} />
      <MessageBox message={message} onClose={closeMessage} />

      {audioBlocked && gate === 'done' && (
        <div style={HINT_WRAP} role="status">
          <button type="button" style={HINT_BTN} onClick={resumeAudio}>
            Sound is paused. Tap to turn it on.
            <br />
            On iPhone, also check the silent switch.
          </button>
        </div>
      )}

      {gate !== 'done' && (
        <Suspense fallback={<div style={GATE_FALLBACK} aria-hidden="true" />}>
          <EntryGate
            onEnter={handleGateEnter}
            leaving={gate === 'leaving'}
            onExited={handleGateExited}
          />
        </Suspense>
      )}
    </>
  )
}

import { useEffect } from 'react'
import styles from './FactPopup.module.css'

export default function FactPopup({ fact, onClose }) {
  useEffect(() => {
    if (!fact) return
    const timer = setTimeout(onClose, 6000)
    return () => clearTimeout(timer)
  }, [fact, onClose])

  return (
    <div className={`${styles.popup} ${fact ? styles.show : ''}`} role="status">
      {fact && (
        <>
          <div className={styles.header}>
            <span className={styles.title}>{fact.title}</span>
            {fact.subtitle && <span className={styles.subtitle}>{fact.subtitle}</span>}
            <button className={styles.close} onClick={onClose} aria-label="Close">×</button>
          </div>
          <p className={styles.text}>{fact.text}</p>
          {fact.extra && <p className={styles.extra}>{fact.extra}</p>}
        </>
      )}
    </div>
  )
}

import { useEffect, useState } from 'react'
import styles from './MessageBox.module.css'

export default function MessageBox({ message, onClose }) {
  const [displayed, setDisplayed] = useState('')
  const [done, setDone] = useState(false)

  useEffect(() => {
    if (!message) { setDisplayed(''); setDone(false); return }
    setDisplayed('')
    setDone(false)

    let i = 0
    const interval = setInterval(() => {
      if (i < message.length) {
        setDisplayed(message.slice(0, i + 1))
        i++
      } else {
        setDone(true)
        clearInterval(interval)
      }
    }, 24)

    return () => clearInterval(interval)
  }, [message])

  const handleBackdrop = (e) => {
    if (e.target === e.currentTarget) onClose()
  }

  if (!message) return null

  return (
    <div className={styles.overlay} onClick={handleBackdrop} role="dialog" aria-modal="true" aria-label="Message from the universe">
      <div className={styles.card}>
        <div className={styles.deco}>✦</div>
        <p className={styles.label}>A message from the universe</p>
        <p className={styles.text}>
          {displayed}
          {!done && <span className={styles.cursor} />}
        </p>
        {done && (
          <button className={styles.closeBtn} onClick={onClose}>
            Return to the cosmos
          </button>
        )}
      </div>
    </div>
  )
}

const prefersReduced = () =>
  typeof window !== 'undefined' &&
  window.matchMedia('(prefers-reduced-motion: reduce)').matches

function getAudioMetrics(data) {
  if (!data || data.length === 0) return { bass: 0, avg: 0, mid: 0 }
  const bassEnd = Math.floor(data.length * 0.06)
  const midEnd = Math.floor(data.length * 0.35)
  let bass = 0, mid = 0, total = 0
  for (let i = 0; i < data.length; i++) {
    const v = data[i] / 255
    total += v
    if (i < bassEnd) bass += v
    else if (i < midEnd) mid += v
  }
  return { bass: bass / bassEnd, mid: mid / (midEnd - bassEnd), avg: total / data.length }
}

function fmt(sec) {
  if (!sec || isNaN(sec)) return '0:00'
  const m = Math.floor(sec / 60)
  const s = Math.floor(sec % 60).toString().padStart(2, '0')
  return `${m}:${s}`
}

// ── Animated artwork canvas per theme (IDLE ANIMATION) ──────────────
function ArtCanvas({ theme, color }) {
  const canvasRef = useRef(null)
  const rafRef = useRef(null)
  const t0 = useRef(Date.now())
  const particles = useRef([])

  useEffect(() => {
    const canvas = canvasRef.current
    if (!canvas) return
    const ctx = canvas.getContext('2d')
    const W = canvas.width, H = canvas.height

    const initParticles = () => {
      particles.current = []
      if (theme === 'stars' || theme === 'nebula' || theme === 'clouds') {
        for (let i = 0; i < 40; i++) {
          particles.current.push({
            x: Math.random() * W, y: Math.random() * H,
            r: Math.random() * 1.5 + 0.3,
            speed: Math.random() * 0.3 + 0.05,
            phase: Math.random() * Math.PI * 2,
            opacity: Math.random() * 0.8 + 0.2,
          })
        }
      }
    }
    initParticles()

    const draw = () => {
      const t = (Date.now() - t0.current) / 1000
      const c1 = color[0], c2 = color[1]

      ctx.clearRect(0, 0, W, H)

      if (theme === 'nebula') {
        for (let i = 0; i < 3; i++) {
          const cx = W * (0.3 + i * 0.2) + Math.sin(t * 0.3 + i) * 8
          const cy = H * (0.4 + Math.sin(t * 0.2 + i * 1.2) * 0.2)
          const r = 22 + Math.sin(t + i) * 4
          const g = ctx.createRadialGradient(cx, cy, 0, cx, cy, r)
          g.addColorStop(0, c1 + (i === 1 ? 'cc' : '88'))
          g.addColorStop(0.5, c2 + '44')
          g.addColorStop(1, 'transparent')
          ctx.fillStyle = g
          ctx.beginPath(); ctx.arc(cx, cy, r, 0, Math.PI * 2); ctx.fill()
        }
        particles.current.forEach(p => {
          p.phase += 0.02
          const a = 0.4 + Math.abs(Math.sin(p.phase)) * 0.6
          ctx.beginPath(); ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2)
          ctx.fillStyle = `rgba(200,220,255,${a})`; ctx.fill()
        })
      }

      if (theme === 'solar') {
        const cx = W * 0.5, cy = H * 0.5
        const pulseR = 14
        for (let ring = 3; ring >= 0; ring--) {
          const g = ctx.createRadialGradient(cx, cy, 0, cx, cy, pulseR + ring * 8)
          g.addColorStop(0, ring === 0 ? '#ffffff' : c1 + 'ee')
          g.addColorStop(0.4, c1 + (ring === 0 ? 'cc' : '66'))
          g.addColorStop(1, 'transparent')
          ctx.fillStyle = g
          ctx.beginPath(); ctx.arc(cx, cy, pulseR + ring * 8, 0, Math.PI * 2); ctx.fill()
        }
        for (let i = 0; i < 6; i++) {
          const angle = t * 0.4 + i * Math.PI / 3
          const len = 10 + Math.sin(t + i) * 4
          const x2 = cx + Math.cos(angle) * (pulseR + len)
          const y2 = cy + Math.sin(angle) * (pulseR + len)
          ctx.strokeStyle = c1 + '88'
          ctx.lineWidth = 1.5
          ctx.beginPath(); ctx.moveTo(cx, cy); ctx.lineTo(x2, y2); ctx.stroke()
        }
      }

      if (theme === 'blackhole') {
        const cx = W * 0.5, cy = H * 0.5
        const coreR = 10
        for (let i = 5; i >= 0; i--) {
          const rx = coreR + 6 + i * 4
          const ry = (coreR + 6 + i * 4) * 0.28
          const alpha = (0.6 - i * 0.08) * 0.7
          ctx.save()
          ctx.translate(cx, cy)
          ctx.rotate(t * 0.5)
          ctx.strokeStyle = i < 2 ? c1 + 'ff' : c2 + Math.floor(alpha * 255).toString(16).padStart(2, '0')
          ctx.lineWidth = i < 2 ? 2 : 1
          ctx.beginPath(); ctx.ellipse(0, 0, rx, ry, 0, 0, Math.PI * 2)
          ctx.stroke(); ctx.restore()
        }
        const cg = ctx.createRadialGradient(cx, cy, 0, cx, cy, coreR)
        cg.addColorStop(0, '#000000'); cg.addColorStop(0.6, '#110008'); cg.addColorStop(1, c1 + '44')
        ctx.fillStyle = cg
        ctx.beginPath(); ctx.arc(cx, cy, coreR, 0, Math.PI * 2); ctx.fill()
      }

      if (theme === 'stars') {
        particles.current.forEach(p => {
          p.x -= p.speed * 0.8
          if (p.x < 0) { p.x = W; p.y = Math.random() * H }
          p.phase += 0.015
          const a = 0.3 + Math.abs(Math.sin(p.phase)) * 0.7
          ctx.beginPath(); ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2)
          ctx.fillStyle = `rgba(180,210,255,${a})`; ctx.fill()
        })
        const g = ctx.createRadialGradient(W * 0.3, H * 0.5, 0, W * 0.3, H * 0.5, W * 0.6)
        g.addColorStop(0, c1 + '22'); g.addColorStop(1, 'transparent')
        ctx.fillStyle = g; ctx.fillRect(0, 0, W, H)
      }

      if (theme === 'clouds') {
        const cloudDefs = [
          { x: 0.2, y: 0.35, r: 18, speed: 0.20 },
          { x: 0.55, y: 0.55, r: 22, speed: 0.15 },
          { x: 0.8, y: 0.3, r: 14, speed: 0.25 },
        ]
        cloudDefs.forEach((cl, i) => {
          const cx = (cl.x + Math.sin(t * cl.speed + i) * 0.15) * W
          const cy = (cl.y + Math.cos(t * cl.speed * 0.15 + i) * 0.16) * H
          const r = cl.r
          for (let b = 3; b >= 0; b--) {
            const g = ctx.createRadialGradient(cx, cy, 0, cx, cy, r + b * 5)
            g.addColorStop(0, `rgba(200,210,255,${0.18 - b * 0.04})`)
            g.addColorStop(1, 'transparent')

          }

          p.phase += 0.008
          const a = 0.2 + Math.abs(Math.sin(p.phase)) * 0.5
          ctx.beginPath(); ctx.arc(p.x, p.y, p.r * 0.8, 0, Math.PI * 2)
          ctx.fillStyle = `rgba(220,225,255,${a})`; ctx.fill()
        })
      }

      rafRef.current = requestAnimationFrame(draw)
    }

    rafRef.current = requestAnimationFrame(draw)
    return () => cancelAnimationFrame(rafRef.current)
  }, [theme, color])

  return <canvas ref={canvasRef} width={56} height={56} className={styles.artCanvas} />
}

function BarVisualizer({ getAudioData, playing, color, onMetrics }) {
  const canvasRef = useRef(null)
  const rafRef = useRef(null)
  const t0 = useRef(Date.now())
  const freqRef = useRef(new Uint8Array(128))

  useEffect(() => {
    const canvas = canvasRef.current
    if (!canvas) return
    const ctx = canvas.getContext('2d')
    const W = canvas.width, H = canvas.height
    const BINS = 28
    const freqData = freqRef.current

    const draw = () => {
      ctx.clearRect(0, 0, W, H)
      const t = (Date.now() - t0.current) / 1000

      if (!getAudioData || !playing) {
        // Idle sine wave
        ctx.strokeStyle = color[0] + '55'
        ctx.lineWidth = 1.5
        ctx.beginPath()
        for (let x = 0; x < W; x++) {
          const y = H / 2 + Math.sin(x * 0.18 + t * 1.2) * 3.5
            + Math.sin(x * 0.08 + t * 0.7) * 2
          x === 0 ? ctx.moveTo(x, y) : ctx.lineTo(x, y)
        }
        ctx.stroke()
        if (onMetrics) onMetrics(null)
        rafRef.current = requestAnimationFrame(draw)
        return
      }

      getAudioData(freqData)
      if (onMetrics) onMetrics(freqData)

      const step = Math.floor(freqData.length / BINS)
      const barW = (W / BINS) - 1

      const grad = ctx.createLinearGradient(0, H, 0, 0)
      grad.addColorStop(0, color[0] + 'dd')
      grad.addColorStop(0.6, color[1] + 'aa')
      grad.addColorStop(1, '#ffffff44')

      for (let i = 0; i < BINS; i++) {
        let sum = 0
        for (let j = 0; j < step; j++) sum += (freqData[i * step + j] || 0)
        const v = (sum / step) / 255
        const barH = Math.max(2, v * H * 1.15)
        const x = i * (barW + 1)
        ctx.fillStyle = grad
        ctx.beginPath()
        if (ctx.roundRect) ctx.roundRect(x, H - barH, barW, barH, 2)
        else ctx.rect(x, H - barH, barW, barH)
        ctx.fill()
      }
      rafRef.current = requestAnimationFrame(draw)
    }

    rafRef.current = requestAnimationFrame(draw)
    return () => cancelAnimationFrame(rafRef.current)
  }, [getAudioData, playing, color, onMetrics])

  return <canvas ref={canvasRef} width={196} height={32} className={styles.barCanvas} />
}

import { useEffect, useRef, memo } from 'react'

// Waktu-konstan easing depth (detik). 3 × TAU ≈ 1,6 dtk → ~95% tercapai, ~2 dtk hampir tuntas.
const DEPTH_TAU = 0.55
const clamp01 = (v) => Math.min(1, Math.max(0, v))

// depth 0..1: makin tinggi → canvas lebih redup, bintang yang digambar berkurang
// (sisa kebanyakan yang redup), halo mengecil, kedip melambat. Dibaca lewat ref di loop gambar.
function StarField({ count = 500, mouseParallax = true, depth = 0 }) {
  const canvasRef = useRef(null)
  const starsRef = useRef([])
  const shootersRef = useRef([])
  const mouseRef = useRef({ x: 0, y: 0 })
  const animRef = useRef(null)

  // Depth: target (dari prop) dan nilai berjalan (di-easing di loop, tanpa state per frame)
  const depthTargetRef = useRef(clamp01(depth))
  const depthCurRef = useRef(clamp01(depth))
  const kickRef = useRef(null) // membangunkan loop (penting saat reduced motion: loop berhenti bila sudah stabil)

  useEffect(() => {
    const canvas = canvasRef.current
    if (!canvas) return
    const ctx = canvas.getContext('2d')
    const reduce = window.matchMedia('(prefers-reduced-motion: reduce)').matches

    let lastW = 0
    let lastH = 0
    let lastT = 0

    // Layar sempit (HP) cukup dengan ±55% bintang: tampilan sama, beban jauh lebih ringan
    const targetCount = () => (window.innerWidth < 768 ? Math.round(count * 0.55) : count)

    // ── Init stars ────────────────────────────────────────
    const initStars = () => {
      starsRef.current = Array.from({ length: targetCount() }, () => {
        const tier = Math.random()
        // Radius disesuaikan agar bintang terlihat jelas
        const r = tier > 0.92 ? Math.random() * 1.8 + 1.4   // Bintang terang
          : tier > 0.75 ? Math.random() * 1.0 + 0.8   // Bintang sedang
            : Math.random() * 0.6 + 0.4                 // Bintang redup
        return {
          x: Math.random() * canvas.width,
          y: Math.random() * canvas.height,
          r,
          // rank 0..1 tetap per bintang: bintang besar cenderung berrank tinggi (dibuang duluan saat depth naik)
          rank: 0.5 * ((r - 0.4) / 2.8) + 0.5 * Math.random(),
          alpha: Math.random(),
          speed: reduce ? 0 : Math.random() * 0.005 + 0.001,
          phase: Math.random() * Math.PI * 2,
          drift: reduce ? 0 : (Math.random() - 0.5) * 0.02,
          layer: Math.floor(Math.random() * 3),
          color: (() => {
            const c = Math.random()
            if (c > 0.90) return { r: 255, g: 220, b: 120 }
            if (c > 0.80) return { r: 200, g: 180, b: 255 }
            if (c > 0.70) return { r: 140, g: 200, b: 255 }
            if (c > 0.60) return { r: 255, g: 180, b: 180 }
            return { r: 210, g: 225, b: 255 }
          })(),
        }
      })
    }

    // ── Resize ────────────────────────────────────────────
    // Address bar HP naik-turun hanya mengubah TINGGI: bintang tidak dibuat ulang,
    // posisi Y cukup diskalakan. Bintang baru dibuat hanya jika LEBAR berubah.
    const resize = () => {
      const w = window.innerWidth
      const h = window.innerHeight
      const widthChanged = w !== lastW
      const oldH = lastH
      lastW = w
      lastH = h
      canvas.width = w
      canvas.height = h

      if (!starsRef.current.length || widthChanged) {
        initStars()
      } else if (oldH && h !== oldH) {
        const k = h / oldH
        starsRef.current.forEach((s) => { s.y *= k })
      }
      if (reduce) draw() // loop tidak berjalan terus di mode reduced: gambar ulang manual
    }

    // ── Shooting stars ────────────────────────────────────
    const spawnShooter = () => ({
      x: Math.random() * canvas.width * 0.7,
      y: Math.random() * canvas.height * 0.4,
      len: Math.random() * 180 + 80,
      speed: Math.random() * 14 + 8,
      angle: Math.PI / 4 + (Math.random() - 0.5) * 0.3,
      opacity: 1,
      width: Math.random() * 1.5 + 0.5,
      trail: [],
    })

    // Spawn shooting star setiap 3–7 detik (dimatikan saat reduced motion)
    const shooterInterval = reduce ? 0 : setInterval(() => {
      if (!document.hidden && shootersRef.current.length < 3) {
        shootersRef.current.push(spawnShooter())
      }
    }, 3000 + Math.random() * 4000)

    // ── Draw ──────────────────────────────────────────────
    function draw() {
      ctx.clearRect(0, 0, canvas.width, canvas.height)
      const mx = mouseRef.current.x
      const my = mouseRef.current.y

      // Parameter turunan depth (dihitung sekali per frame)
      const d = depthCurRef.current
      const dim = 1 - 0.5 * d        // kecerahan canvas: 1 → 0,5
      const keep = 1.1 - 0.65 * d    // ambang rank: 1,1 (semua) → 0,45 (sisa kebanyakan redup)
      const slow = 1 - 0.65 * d      // kecepatan kedip: 100% → 35%
      const haloK = 2.2 - 0.9 * d    // radius halo: 2,2× → 1,3× jari-jari
      const haloA = 0.22 * (1 - 0.5 * d)

      ctx.globalAlpha = dim

      // Draw stars
      starsRef.current.forEach((s) => {
        s.phase += s.speed * slow
        s.x += s.drift
        if (s.x < -2) s.x = canvas.width + 2
        if (s.x > canvas.width + 2) s.x = -2

        // Memudar halus di sekitar ambang (tidak "pop" saat bintang hilang)
        const vis = clamp01((keep - s.rank) / 0.1)
        if (vis <= 0.01) return

        const parallaxStr = [0.003, 0.009, 0.020][s.layer]
        const px = s.x + mx * parallaxStr
        const py = s.y + my * parallaxStr

        // Twinkle: bintang besar lebih dramatis (reduced: kecerahan tetap)
        const twinkle = reduce ? 0.8 : 0.3 + Math.abs(Math.sin(s.phase)) * 0.7
        const { r, g, b } = s.color
        const alpha = twinkle * (s.r > 1.2 ? 1 : 0.85) * vis

        ctx.beginPath()
        ctx.arc(px, py, s.r, 0, Math.PI * 2)
        ctx.fillStyle = `rgba(${r},${g},${b},${alpha})`
        ctx.fill()

        // Luminous halo untuk bintang terang (jauh lebih efisien dari shadowBlur CPU filter)
        if (s.r > 1.0) {
          ctx.beginPath()
          ctx.arc(px, py, s.r * haloK, 0, Math.PI * 2)
          ctx.fillStyle = `rgba(${r},${g},${b},${alpha * haloA / 0.22 * 0.22})`
          ctx.fill()
        }
      })

      // Draw shooting stars (ikut dim lewat globalAlpha)
      shootersRef.current = shootersRef.current.filter((s) => {
        s.x += Math.cos(s.angle) * s.speed
        s.y += Math.sin(s.angle) * s.speed
        s.opacity -= 0.012

        if (s.opacity <= 0 || s.x > canvas.width + 50 || s.y > canvas.height + 50) return false

        const tailX = s.x - Math.cos(s.angle) * s.len
        const tailY = s.y - Math.sin(s.angle) * s.len

        const grad = ctx.createLinearGradient(tailX, tailY, s.x, s.y)
        grad.addColorStop(0, `rgba(255,255,255,0)`)
        grad.addColorStop(0.7, `rgba(200,220,255,${s.opacity * 0.4})`)
        grad.addColorStop(1, `rgba(255,255,255,${s.opacity})`)

        ctx.beginPath()
        ctx.moveTo(tailX, tailY)
        ctx.lineTo(s.x, s.y)
        ctx.strokeStyle = grad
        ctx.lineWidth = s.width
        ctx.shadowBlur = 8
        ctx.shadowColor = `rgba(180,210,255,${s.opacity})`
        ctx.stroke()
        ctx.shadowBlur = 0

        return true
      })

      ctx.globalAlpha = 1
    }

    // ── Loop ──────────────────────────────────────────────
    // Normal: berjalan terus (dijeda saat tab tersembunyi).
    // Reduced motion: hanya berjalan selama depth masih berpindah, lalu berhenti sendiri.
    const loop = (now) => {
      const dt = lastT ? Math.min((now - lastT) / 1000, 0.1) : 0.016
      lastT = now

      const target = depthTargetRef.current
      const diff = target - depthCurRef.current
      depthCurRef.current = Math.abs(diff) < 0.0005
        ? target
        : depthCurRef.current + diff * (1 - Math.exp(-dt / DEPTH_TAU))

      draw()

      if (reduce && depthCurRef.current === target) {
        animRef.current = null
        lastT = 0
        return
      }
      animRef.current = requestAnimationFrame(loop)
    }
    const startLoop = () => {
      if (animRef.current || document.hidden) return
      lastT = 0
      animRef.current = requestAnimationFrame(loop)
    }
    const stopLoop = () => {
      cancelAnimationFrame(animRef.current)
      animRef.current = null
    }
    const onVisibility = () => (document.hidden ? stopLoop() : startLoop())

    const onMouseMove = (e) => {
      mouseRef.current = {
        x: e.clientX - window.innerWidth / 2,
        y: e.clientY - window.innerHeight / 2,
      }
    }

    kickRef.current = startLoop

    resize()
    window.addEventListener('resize', resize)
    document.addEventListener('visibilitychange', onVisibility)
    // Parallax mouse dimatikan saat reduced motion
    if (mouseParallax && !reduce) window.addEventListener('mousemove', onMouseMove, { passive: true })
    startLoop()

    return () => {
      kickRef.current = null
      window.removeEventListener('resize', resize)
      window.removeEventListener('mousemove', onMouseMove)
      document.removeEventListener('visibilitychange', onVisibility)
      stopLoop()
      if (shooterInterval) clearInterval(shooterInterval)
      shootersRef.current = []
    }
  }, [count, mouseParallax])

  // Perubahan depth hanya memperbarui target (ref) dan membangunkan loop; efek utama tidak dijalankan ulang
  useEffect(() => {
    depthTargetRef.current = clamp01(depth)
    kickRef.current?.()
  }, [depth])

  return (
    <canvas
      ref={canvasRef}
      aria-hidden="true"
      style={{
        position: 'fixed',
        top: 0, left: 0,
        width: '100%', height: '100%',
        pointerEvents: 'none',
        zIndex: 0,
      }}
    />
  )
}

export default memo(StarField)

// ── KOMPONEN EFEK MESIN KETIK (TERMINAL SCI-FI) ────────────────────────
function Typewriter({ text, speed = 25, delay = 0 }) {
  const [displayed, setDisplayed] = useState('')

  useEffect(() => {
    setDisplayed('')
    let timeout
    let interval

    timeout = setTimeout(() => {
      let i = 0
      interval = setInterval(() => {
        setDisplayed(text.substring(0, i + 1))
        i++
        if (i >= text.length) clearInterval(interval)
      }, speed)
    }, delay)

    return () => {
      clearTimeout(timeout)
      clearInterval(interval)
    }
  }, [text, speed, delay])

  return <span>{displayed}</span>
}

// ── KOMPONEN 2.5D PARALLAX CARD ────────────────────────────────────────
function TiltCard({ p, onClick }) {
  const [tilt, setTilt] = useState({ x: 0, y: 0 })
  const cardRef = useRef(null)

  const handleMouseMove = (e) => {
    if (!cardRef.current) return
    const rect = cardRef.current.getBoundingClientRect()
    const x = e.clientX - rect.left
    const y = e.clientY - rect.top
    const centerX = rect.width / 2
    const centerY = rect.height / 2
    const rotateX = ((y - centerY) / centerY) * -15
    const rotateY = ((x - centerX) / centerX) * 15
    setTilt({ x: rotateX, y: rotateY })
  }

  const handleMouseLeave = () => setTilt({ x: 0, y: 0 })

  return (
    <div className={styles.cardWrapper}>
      <div
        ref={cardRef}
        className={styles.cardHUD}
        style={{
          '--glow': p.glowColor,
          '--border': p.borderColor,
          transform: `perspective(1000px) rotateX(${tilt.x}deg) rotateY(${tilt.y}deg)`,
        }}
        onMouseMove={handleMouseMove}
        onMouseEnter={() => sfx.hover()}
        onMouseLeave={handleMouseLeave}
        onClick={() => onClick(p)}
        role="button"
        tabIndex={0}
        aria-label={`Open ${p.name}`}
        onKeyDown={(e) => {
          if (e.key === 'Enter' || e.key === ' ') {
            e.preventDefault()
            onClick(p)
          }
        }}
      >
        <div className={styles.icon3D} style={{ boxShadow: `0 0 30px ${p.glowColor}88` }}>
          {getIcon(p.id)}
        </div>
        <h3 className={styles.cardTitle3D}>{p.name}</h3>
        <p className={styles.cardDesc3D}>{p.description.substring(0, 75)}...</p>

        <div className={styles.cardFooter3D}>
          <span className={styles.discover}>Initialize Scan →</span>
        </div>
      </div>
    </div>
  )
}

    setActivePhenom(p)
    sfx.open()
    if (onFact) onFact(null)
  }

    setActivePhenom(null)
    sfx.close()

      if (e.key === 'Escape') closeOverlay()
    }
    window.addEventListener('keydown', onKey)
    return () => window.removeEventListener('keydown', onKey)
  }, [activePhenom])

  return (
